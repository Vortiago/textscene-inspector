# WI-51: SubResource/ExtResource Go to Definition

## Overview

Implement a `DefinitionProvider` for .tscn files to enable "Go to Definition" functionality for SubResource and ExtResource references. This allows developers to **Ctrl+Click** (or Cmd+Click on Mac) on resource IDs to jump directly to their definitions within the file or external resources.

**User Experience:**

```tscn
[node name="Player" type="MeshInstance3D"]
mesh = SubResource("BoxMesh_abc123")  ← Ctrl+Click here

[sub_resource type="BoxMesh" id="BoxMesh_abc123"]  ← Jumps to this line
size = Vector3(1, 2, 1)
```

**Key Benefits:**
- ✅ Fast navigation between references and definitions
- ✅ Eliminates manual searching in large files
- ✅ Works with F12 (Go to Definition), Alt+F12 (Peek Definition), Ctrl+Click
- ✅ Essential for understanding material/mesh/scene relationships
- ✅ Lightweight implementation (~150 lines)

## TSCN Resource Reference Format

### SubResource References (Internal)

SubResources are defined **within** the same .tscn file:

**Reference Patterns:**
```tscn
mesh = SubResource("BoxMesh_abc123")
material = SubResource("Material_xyz")
shape = SubResource( "Shape_123" )  # with spaces
albedo_texture = SubResource('Texture_456')  # single quotes
```

**Definition Format:**
```tscn
[sub_resource type="BoxMesh" id="BoxMesh_abc123"]
size = Vector3(1, 2, 1)

[sub_resource type="StandardMaterial3D" id="Material_xyz"]
albedo_color = Color(1, 0, 0, 1)
```

### ExtResource References (External)

ExtResources reference files external to the current .tscn:

**Reference Patterns:**
```tscn
instance = ExtResource("Door_scene")
texture = ExtResource("wood_texture")
script = ExtResource("player_script")
```

**Definition Format:**
```tscn
[ext_resource type="PackedScene" path="res://scenes/Door.tscn" id="Door_scene"]
[ext_resource type="Texture2D" path="res://textures/wood.png" id="wood_texture"]
[ext_resource type="Script" path="res://scripts/Player.cs" id="player_script"]
```

### Edge Cases to Handle

**1. Quotes Variations:**
```tscn
SubResource("id")      # Double quotes
SubResource('id')      # Single quotes
SubResource( "id" )    # Extra spaces
```

**2. Special Characters in IDs:**
```tscn
SubResource("Mesh_123-456")
SubResource("Material.Main")
SubResource("Texture@2x")
```

**3. Multiple References to Same Resource:**
```tscn
# Same resource used multiple times
mesh = SubResource("BoxMesh_1")
collision_mesh = SubResource("BoxMesh_1")
```

## Implementation Notes

### Architecture Overview

**Components:**
1. **TscnDefinitionProvider** - Implements `DefinitionProvider`
2. **Resource ID Extractor** - Parses SubResource/ExtResource calls
3. **Resource Definition Finder** - Locates [sub_resource] and [ext_resource] headings

**Data Flow:**

```
User Ctrl+Clicks on resource ID
    ↓
VS Code calls provideDefinition(document, position, token)
    ↓
Extract word at cursor position
    ↓
Check if word is inside SubResource() or ExtResource() call
    ↓
Parse resource ID from function call
    ↓
Search document for [sub_resource id="..."] or [ext_resource id="..."]
    ↓
Calculate Range of definition heading
    ↓
Return Location pointing to definition
    ↓
VS Code navigates to definition or shows peek
```

### Detecting Resource References

**Strategy:** Check if the cursor position is within a SubResource() or ExtResource() call.

```typescript
function getResourceIdAtPosition(
  document: vscode.TextDocument,
  position: vscode.Position
): { type: 'SubResource' | 'ExtResource'; id: string } | null {
  const line = document.lineAt(position).text;
  const cursorOffset = position.character;

  // Regex patterns for SubResource and ExtResource calls
  // Matches: SubResource("id"), SubResource('id'), SubResource( "id" ), etc.
  const subResourceRegex = /SubResource\s*\(\s*["']([^"']+)["']\s*\)/g;
  const extResourceRegex = /ExtResource\s*\(\s*["']([^"']+)["']\s*\)/g;

  // Check SubResource
  let match;
  subResourceRegex.lastIndex = 0;
  while ((match = subResourceRegex.exec(line)) !== null) {
    const startIndex = match.index;
    const endIndex = match.index + match[0].length;

    // Check if cursor is within this match
    if (cursorOffset >= startIndex && cursorOffset <= endIndex) {
      return { type: 'SubResource', id: match[1] };
    }
  }

  // Check ExtResource
  extResourceRegex.lastIndex = 0;
  while ((match = extResourceRegex.exec(line)) !== null) {
    const startIndex = match.index;
    const endIndex = match.index + match[0].length;

    if (cursorOffset >= startIndex && cursorOffset <= endIndex) {
      return { type: 'ExtResource', id: match[1] };
    }
  }

  return null;
}
```

### Finding Resource Definitions

**Strategy:** Search the entire document for the matching heading.

```typescript
function findResourceDefinition(
  document: vscode.TextDocument,
  resourceType: 'SubResource' | 'ExtResource',
  resourceId: string
): vscode.Location | null {
  const text = document.getText();
  const lines = text.split('\n');

  // Build heading pattern
  // [sub_resource type="..." id="resourceId"]
  // [ext_resource type="..." path="..." id="resourceId"]
  const headingPrefix = resourceType === 'SubResource'
    ? '[sub_resource'
    : '[ext_resource';

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i]!;

    // Check if line is a resource heading
    if (!line.startsWith(headingPrefix)) {
      continue;
    }

    // Extract id from heading
    const idMatch = line.match(/id\s*=\s*["']([^"']+)["']/);
    if (!idMatch) {
      continue;
    }

    const definitionId = idMatch[1];
    if (definitionId === resourceId) {
      // Found the definition!
      const range = new vscode.Range(
        new vscode.Position(i, 0),
        new vscode.Position(i, line.length)
      );

      return new vscode.Location(document.uri, range);
    }
  }

  return null;
}
```

### TscnDefinitionProvider Implementation

```typescript
// TscnDefinitionProvider.ts
import * as vscode from 'vscode';

/**
 * Provides "Go to Definition" for SubResource and ExtResource references in TSCN files.
 */
export class TscnDefinitionProvider implements vscode.DefinitionProvider {
  provideDefinition(
    document: vscode.TextDocument,
    position: vscode.Position,
    token: vscode.CancellationToken
  ): vscode.ProviderResult<vscode.Definition | vscode.LocationLink[]> {
    // Get resource reference at cursor position
    const resource = this.getResourceIdAtPosition(document, position);

    if (!resource) {
      return null;
    }

    // Find the definition
    const definition = this.findResourceDefinition(
      document,
      resource.type,
      resource.id
    );

    return definition;
  }

  private getResourceIdAtPosition(
    document: vscode.TextDocument,
    position: vscode.Position
  ): { type: 'SubResource' | 'ExtResource'; id: string } | null {
    const line = document.lineAt(position).text;
    const cursorOffset = position.character;

    // Match SubResource("id") or ExtResource("id")
    // Handles: SubResource("id"), SubResource('id'), SubResource( "id" ), etc.
    const subResourceRegex = /SubResource\s*\(\s*["']([^"']+)["']\s*\)/g;
    const extResourceRegex = /ExtResource\s*\(\s*["']([^"']+)["']\s*\)/g;

    // Check if cursor is within a SubResource call
    let match;
    subResourceRegex.lastIndex = 0;
    while ((match = subResourceRegex.exec(line)) !== null) {
      const startIndex = match.index;
      const endIndex = match.index + match[0].length;

      if (cursorOffset >= startIndex && cursorOffset <= endIndex) {
        return { type: 'SubResource', id: match[1]! };
      }
    }

    // Check if cursor is within an ExtResource call
    extResourceRegex.lastIndex = 0;
    while ((match = extResourceRegex.exec(line)) !== null) {
      const startIndex = match.index;
      const endIndex = match.index + match[0].length;

      if (cursorOffset >= startIndex && cursorOffset <= endIndex) {
        return { type: 'ExtResource', id: match[1]! };
      }
    }

    return null;
  }

  private findResourceDefinition(
    document: vscode.TextDocument,
    resourceType: 'SubResource' | 'ExtResource',
    resourceId: string
  ): vscode.Location | null {
    const text = document.getText();
    const lines = text.split('\n');

    // Determine heading prefix
    const headingPrefix =
      resourceType === 'SubResource' ? '[sub_resource' : '[ext_resource';

    // Search for matching resource definition
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i]!;

      // Check if line starts with resource heading
      if (!line.startsWith(headingPrefix)) {
        continue;
      }

      // Extract id attribute from heading
      // Matches: id="value" or id='value'
      const idMatch = line.match(/id\s*=\s*["']([^"']+)["']/);

      if (!idMatch) {
        continue;
      }

      const definitionId = idMatch[1];

      if (definitionId === resourceId) {
        // Found the definition
        const range = new vscode.Range(
          new vscode.Position(i, 0),
          new vscode.Position(i, line.length)
        );

        return new vscode.Location(document.uri, range);
      }
    }

    // Definition not found
    return null;
  }
}
```

private async findExtResourceDefinition(
  document: vscode.TextDocument,
  resourceId: string
): Promise<vscode.Location | null> {
  // Find [ext_resource] heading
  const lines = document.getText().split('\n');

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i]!;

    if (!line.startsWith('[ext_resource')) {
      continue;
    }

    const idMatch = line.match(/id\s*=\s*["']([^"']+)["']/);
    if (idMatch?.[1] === resourceId) {

      if (pathMatch) {
        const resourcePath = pathMatch[1];

        const fsPath = this.convertGodotPathToFsPath(resourcePath);
        const uri = vscode.Uri.file(fsPath);

        // Return location in external file
        return new vscode.Location(uri, new vscode.Position(0, 0));
      }
    }
  }

  return null;
}

class TscnHoverProvider implements vscode.HoverProvider {

    document: vscode.TextDocument,

    return new vscode.Hover(
