# WI-78.2: Scene Resolution & Flattening

**Parent**: WI-78 (Immutable Scene Hierarchies)
**Status**: Not Started
**Complexity**: Moderate
**Dependency**: WI-78.1 (Immutable SceneGraph Model)

## Goal

Load external scenes and flatten into single immutable SceneGraph.

## Problem

Current external scene loading mutates TscnNode:
```typescript
// BAD: Mutates node.children during loading
async addScene(instancePath: string, scenePath: string): Promise<void> {
  const externalScene = await this.loadExternalScene(scenePath);
  for (const node of externalScene.nodes) {
    parent.children.push(node);  // MUTATION!
  }
}
```

**Issues:**
- Parser output (TscnNode) is mutated at runtime
- Can't compare old vs new scene
- No record of original structure
- External scene tracking is implicit (just look at node.children)

## Solution

**SceneResolver** that:
1. Loads root scene + all external scenes
2. Keeps scenes separate (no mutation)
3. Flattens into single node array
4. Tracks external scene references explicitly
5. Builds immutable SceneGraph

## Implementation

### SceneResolver

**File**: `packages/textscene-core/src/core/SceneResolver.ts`

```typescript
import { TscnParser } from '../parser/TscnParser.js';
import { SceneGraphBuilder } from './SceneGraphBuilder.js';
import type { SceneGraph, ParsedScene, ExternalSceneRef } from './SceneGraph.js';
import type { TscnScene, TscnNode, TscnExternalResource } from '../parser/types.js';
import type { ResourceProvider } from './ResourceProvider.js';
import { logger } from '../utils/logger.js';

/**
 * Resolves external scene references and builds immutable SceneGraph.
 * Handles missing scenes with placeholders (ResourceRecoveryManager integration).
 */
export class SceneResolver {
  private parser = new TscnParser();
  private sceneCache = new Map<string, ParsedScene>();  // Cache parsed scenes

  constructor(
    private resourceProvider: ResourceProvider
  ) {}

  /**
   * Resolve root scene and all external scenes into immutable SceneGraph.
   *
   * @param rootScenePath - Root scene file path (e.g., 'res://main.tscn')
   * @returns Promise<SceneGraph> - Immutable scene graph
   */
  async resolve(rootScenePath: string): Promise<SceneGraph> {
    logger.info(`[SceneResolver] Resolving scene hierarchy for: ${rootScenePath}`);

    // Load root scene
    const rootScene = await this.loadScene(rootScenePath);

    // Build scene collection (root + all external scenes)
    const scenes = new Map<string, ParsedScene>();
    scenes.set(rootScenePath, rootScene);

    // Resolve external scenes recursively
    await this.resolveExternalScenes(rootScene, scenes);

    // Build immutable SceneGraph
    const builder = new SceneGraphBuilder()
      .setRootScene(rootScenePath);

    for (const scene of scenes.values()) {
      builder.addScene(scene);
    }

    const graph = builder.build();

    logger.info(`[SceneResolver] ✅ Resolved ${scenes.size} scenes, ${graph.flattenedNodes.length} nodes`);

    return graph;
  }

  /**
   * Load and parse scene file.
   */
  private async loadScene(scenePath: string): Promise<ParsedScene> {
    // Check cache first
    if (this.sceneCache.has(scenePath)) {
      logger.info(`[SceneResolver] Cache hit: ${scenePath}`);
      return this.sceneCache.get(scenePath)!;
    }

    logger.info(`[SceneResolver] Loading scene: ${scenePath}`);

    try {
      // Load file content
      const content = await this.resourceProvider.loadFile(scenePath);

      // Parse TSCN
      const tscnScene = this.parser.parse(content);

      // Convert to ParsedScene
      const parsedScene = this.toParsedScene(scenePath, tscnScene);

      // Cache
      this.sceneCache.set(scenePath, parsedScene);

      return parsedScene;

    } catch (error) {
      logger.error(`[SceneResolver] Failed to load scene: ${scenePath}`, error);

      // Create placeholder scene (missing file)
      return this.createPlaceholderScene(scenePath, error);
    }
  }

  /**
   * Convert TscnScene to ParsedScene (immutable).
   */
  private toParsedScene(path: string, tscnScene: TscnScene): ParsedScene {
    // Find external scene references
    const externalScenes: ExternalSceneRef[] = [];

    for (const node of tscnScene.nodes) {
      if (node.instance) {
        externalScenes.push({
          nodeName: node.name,
          scenePath: node.instance,
          resolutionStatus: 'pending'
        });
      }
    }

    return {
      path,
      nodes: Object.freeze([...tscnScene.nodes]),  // Freeze copy
      externalScenes: Object.freeze(externalScenes),
      internalResources: Object.freeze([...tscnScene.internalResources]),
      externalResources: Object.freeze([...tscnScene.externalResources])
    };
  }

  /**
   * Recursively resolve external scenes.
   */
  private async resolveExternalScenes(
    scene: ParsedScene,
    scenes: Map<string, ParsedScene>
  ): Promise<void> {
    // Find all external scene references
    const externalRefs = scene.externalScenes;

    if (externalRefs.length === 0) {
      return;  // No external scenes
    }

    logger.info(`[SceneResolver] Resolving ${externalRefs.length} external scenes from ${scene.path}`);

    // Load all external scenes in parallel
    const loadPromises = externalRefs.map(async (ref) => {
      if (scenes.has(ref.scenePath)) {
        logger.info(`[SceneResolver] External scene already loaded: ${ref.scenePath}`);
        return;  // Already loaded
      }

      const externalScene = await this.loadScene(ref.scenePath);
      scenes.set(ref.scenePath, externalScene);

      // Recurse for nested external scenes
      await this.resolveExternalScenes(externalScene, scenes);
    });

    await Promise.all(loadPromises);
  }

  /**
   * Create placeholder scene for missing file.
   * Integrates with ResourceRecoveryManager.
   */
  private createPlaceholderScene(path: string, error: unknown): ParsedScene {
    logger.warn(`[SceneResolver] Creating placeholder for missing scene: ${path}`);

    // Create placeholder node
    const placeholderNode: TscnNode = {
      name: `Missing_${path.split('/').pop()?.replace('.tscn', '') || 'Scene'}`,
      type: 'Node3D',
      properties: {
        __missing: 'true',
        __error: String(error)
      },
      children: []
    };

    return {
      path,
      nodes: Object.freeze([placeholderNode]),
      externalScenes: Object.freeze([]),
      internalResources: Object.freeze([]),
      externalResources: Object.freeze([])
    };
  }

  /**
   * Clear scene cache.
   * Called when user uploads missing file or scene is modified.
   */
  clearCache(scenePath?: string): void {
    if (scenePath) {
      this.sceneCache.delete(scenePath);
      logger.info(`[SceneResolver] Cleared cache for: ${scenePath}`);
    } else {
      this.sceneCache.clear();
      logger.info(`[SceneResolver] Cleared all scene cache`);
    }
  }
}
```

## Testing Strategy

### Unit Tests

**File**: `packages/textscene-core/src/core/SceneResolver.test.ts`

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { SceneResolver } from './SceneResolver.js';
import type { ResourceProvider } from './ResourceProvider.js';

describe('SceneResolver', () => {
  let resolver: SceneResolver;
  let mockProvider: ResourceProvider;

  beforeEach(() => {
    mockProvider = {
      loadFile: vi.fn()
    } as any;
    resolver = new SceneResolver(mockProvider);
  });

  it('resolves single scene (no external scenes)', async () => {
    const tscnContent = `
[gd_scene load_steps=1 format=3]

[node name="Root" type="Node3D"]
    `;

    mockProvider.loadFile = vi.fn().mockResolvedValue(tscnContent);

    const graph = await resolver.resolve('res://test.tscn');

    expect(graph.rootScene).toBe('res://test.tscn');
    expect(graph.scenes.size).toBe(1);
    expect(graph.flattenedNodes.length).toBe(1);
    expect(graph.flattenedNodes[0].name).toBe('Root');
  });

  it('resolves scene with external scene', async () => {
    const mainContent = `
[gd_scene load_steps=1 format=3]

[node name="Main" type="Node3D"]

[node name="Door" type="Node3D" parent="." instance=ExtResource("1_abcde")]

[ext_resource type="PackedScene" path="res://door.tscn" id="1_abcde"]
    `;

    const doorContent = `
[gd_scene load_steps=1 format=3]

[node name="Door" type="MeshInstance3D"]
    `;

    mockProvider.loadFile = vi.fn()
      .mockResolvedValueOnce(mainContent)   // res://main.tscn
      .mockResolvedValueOnce(doorContent);  // res://door.tscn

    const graph = await resolver.resolve('res://main.tscn');

    expect(graph.scenes.size).toBe(2);
    expect(graph.scenes.has('res://main.tscn')).toBe(true);
    expect(graph.scenes.has('res://door.tscn')).toBe(true);
    expect(graph.flattenedNodes.length).toBeGreaterThan(1);
  });

  it('resolves nested external scenes (3 levels)', async () => {
    const mainContent = `
[gd_scene load_steps=1 format=3]
[node name="Main" type="Node3D"]
[node name="Hallway" instance=ExtResource("1_h")]
[ext_resource type="PackedScene" path="res://hallway.tscn" id="1_h"]
    `;

    const hallwayContent = `
[gd_scene load_steps=1 format=3]
[node name="Hallway" type="Node3D"]
[node name="Door" instance=ExtResource("1_d")]
[ext_resource type="PackedScene" path="res://door.tscn" id="1_d"]
    `;

    const doorContent = `
[gd_scene load_steps=1 format=3]
[node name="Door" type="MeshInstance3D"]
    `;

    mockProvider.loadFile = vi.fn()
      .mockResolvedValueOnce(mainContent)
      .mockResolvedValueOnce(hallwayContent)
      .mockResolvedValueOnce(doorContent);

    const graph = await resolver.resolve('res://main.tscn');

    expect(graph.scenes.size).toBe(3);
    expect(graph.scenes.has('res://main.tscn')).toBe(true);
    expect(graph.scenes.has('res://hallway.tscn')).toBe(true);
    expect(graph.scenes.has('res://door.tscn')).toBe(true);
  });

  it('creates placeholder for missing scene', async () => {
    const mainContent = `
[gd_scene load_steps=1 format=3]
[node name="Main" type="Node3D"]
[node name="Door" instance=ExtResource("1_d")]
[ext_resource type="PackedScene" path="res://missing.tscn" id="1_d"]
    `;

    mockProvider.loadFile = vi.fn()
      .mockResolvedValueOnce(mainContent)
      .mockRejectedValueOnce(new Error('File not found'));

    const graph = await resolver.resolve('res://main.tscn');

    expect(graph.scenes.size).toBe(2);
    expect(graph.scenes.has('res://main.tscn')).toBe(true);
    expect(graph.scenes.has('res://missing.tscn')).toBe(true);

    const placeholder = graph.scenes.get('res://missing.tscn')!;
    expect(placeholder.nodes[0].name).toContain('Missing');
    expect(placeholder.nodes[0].properties.__missing).toBe('true');
  });

  it('caches loaded scenes', async () => {
    const content = `
[gd_scene load_steps=1 format=3]
[node name="Root" type="Node3D"]
    `;

    mockProvider.loadFile = vi.fn().mockResolvedValue(content);

    await resolver.resolve('res://test.tscn');
    await resolver.resolve('res://test.tscn');

    // loadFile called only once (cached on second call)
    expect(mockProvider.loadFile).toHaveBeenCalledTimes(1);
  });

  it('clears cache when requested', async () => {
    const content = `
[gd_scene load_steps=1 format=3]
[node name="Root" type="Node3D"]
    `;

    mockProvider.loadFile = vi.fn().mockResolvedValue(content);

    await resolver.resolve('res://test.tscn');

    // Clear cache
    resolver.clearCache('res://test.tscn');

    await resolver.resolve('res://test.tscn');

    // loadFile called twice (cache cleared)
    expect(mockProvider.loadFile).toHaveBeenCalledTimes(2);
  });

  it('handles circular scene references gracefully', async () => {
    // Main → Door → Main (circular)
    const mainContent = `
[gd_scene load_steps=1 format=3]
[node name="Main" type="Node3D"]
[node name="Door" instance=ExtResource("1_d")]
[ext_resource type="PackedScene" path="res://door.tscn" id="1_d"]
    `;

    const doorContent = `
[gd_scene load_steps=1 format=3]
[node name="Door" type="Node3D"]
[node name="BackToMain" instance=ExtResource("1_m")]
[ext_resource type="PackedScene" path="res://main.tscn" id="1_m"]
    `;

    mockProvider.loadFile = vi.fn()
      .mockImplementation((path: string) => {
        if (path === 'res://main.tscn') return Promise.resolve(mainContent);
        if (path === 'res://door.tscn') return Promise.resolve(doorContent);
        return Promise.reject(new Error('Not found'));
      });

    const graph = await resolver.resolve('res://main.tscn');

    // Should handle circular reference (already loaded check prevents infinite loop)
    expect(graph.scenes.size).toBe(2);
    expect(graph.scenes.has('res://main.tscn')).toBe(true);
    expect(graph.scenes.has('res://door.tscn')).toBe(true);
  });
});
```
