# WI-30: Real-time Property Editing

## Overview

Implement real-time property editing in the VS Code extension, allowing users to modify TSCN node properties through a GUI panel and see changes instantly reflected in the 3D preview. This feature bridges the gap between text-based TSCN editing and visual feedback, significantly improving the development workflow for Godot scene files.

## Implementation Requirements

From TODO.md Phase 8: Interactive Editing:

- Property panel UI for selected nodes
- Edit common properties (position, rotation, scale, color, etc.)
- Two-way communication between webview and extension
- Live preview updates when properties change
- Update TSCN text document when properties are modified
- Support undo/redo through VS Code's text editor
- Validate property values before applying
- Handle different property types (Vector3, Color, float, int, bool, string)
- Debounce rapid property changes
- Sync changes back to TSCN file format

### Property Editor Panel Component

```typescript
// propertyEditor.ts
import * as vscode from 'vscode';
import type { TscnNode } from '@tscn-renderer/core';

export interface PropertyChange {
  nodePath: string;
  property: string;
  value: any;
  oldValue: any;
}

export class PropertyEditorPanel {
  private panel: vscode.WebviewPanel | undefined;
  private currentDocument: vscode.TextDocument | undefined;
  private onPropertyChangeCallback?: (change: PropertyChange) => void;

  constructor(
    private readonly extensionUri: vscode.Uri,
    private readonly context: vscode.ExtensionContext
  ) {}

  show(document: vscode.TextDocument, selectedNode?: TscnNode): void {
    if (this.panel) {
      this.panel.reveal(vscode.ViewColumn.Two);
    } else {
      this.panel = this.createPanel();
      this.setupMessageHandling();
    }

    this.currentDocument = document;

    // Send selected node data to webview
    if (selectedNode) {
      this.panel.webview.postMessage({
        command: 'selectNode',
        node: this.serializeNode(selectedNode),
      });
    }
  }

  private createPanel(): vscode.WebviewPanel {
    const panel = vscode.window.createWebviewPanel(
      'tscnPropertyEditor',
      'TSCN Property Editor',
      vscode.ViewColumn.Two,
      {
        enableScripts: true,
        retainContextWhenHidden: true,
        localResourceRoots: [
          vscode.Uri.joinPath(this.extensionUri, 'media'),
        ],
      }
    );

    panel.webview.html = this.getWebviewHtml(panel.webview);

    panel.onDidDispose(() => {
      this.panel = undefined;
    });

    return panel;
  }

  private setupMessageHandling(): void {
    if (!this.panel) return;

    this.panel.webview.onDidReceiveMessage(
      async (message) => {
        switch (message.command) {
          case 'propertyChanged':
            await this.handlePropertyChange(message);
            break;
          case 'requestNodeData':
            await this.sendNodeData(message.nodePath);
            break;
        }
      },
      undefined,
      this.context.subscriptions
    );
  }

  private async handlePropertyChange(message: any): Promise<void> {
    if (!this.currentDocument) return;

    const change: PropertyChange = {
      nodePath: message.nodePath,
      property: message.property,
      value: message.value,
      oldValue: message.oldValue,
    };

    // Update TSCN document
    await this.updateTscnDocument(change);

    // Notify callback
    if (this.onPropertyChangeCallback) {
      this.onPropertyChangeCallback(change);
    }

    // Notify preview webview to update
    this.notifyPreviewUpdate(change);
  }

  private async updateTscnDocument(change: PropertyChange): Promise<void> {
    if (!this.currentDocument) return;

    const edit = new vscode.WorkspaceEdit();
    const text = this.currentDocument.getText();

    // Find the node section in TSCN
    const nodeRegex = new RegExp(
      `\\[node name="${this.escapeRegex(change.nodePath)}"[^\\]]*\\]([^\\[]*)`,
      'g'
    );

    const match = nodeRegex.exec(text);
    if (!match) return;

    const nodeSection = match[0];
    const nodeSectionStart = match.index;

    // Build updated property string
    const propertyLine = this.formatTscnProperty(change.property, change.value);

    // Check if property exists
    const propertyRegex = new RegExp(`^${change.property}\\s*=.*$`, 'm');
    const propertyMatch = propertyRegex.exec(nodeSection);

    if (propertyMatch) {
      // Replace existing property
      const propertyStart = nodeSectionStart + propertyMatch.index;
      const propertyEnd = propertyStart + propertyMatch[0].length;

      edit.replace(
        this.currentDocument.uri,
        new vscode.Range(
          this.currentDocument.positionAt(propertyStart),
          this.currentDocument.positionAt(propertyEnd)
        ),
        propertyLine
      );
    } else {
      // Add new property after node header
      const headerEndPos = this.currentDocument.positionAt(
        nodeSectionStart + match[0].indexOf(']') + 1
      );

      edit.insert(
        this.currentDocument.uri,
        headerEndPos,
        `\n${propertyLine}`
      );
    }

    await vscode.workspace.applyEdit(edit);
  }

  private formatTscnProperty(property: string, value: any): string {
    // Format value based on type
    if (Array.isArray(value)) {
      // Vector3, Color, etc.
      const formatted = value.map(v => v.toFixed(6)).join(', ');
      return `${property} = Vector3(${formatted})`;
    } else if (typeof value === 'string') {
      return `${property} = "${value}"`;
    } else if (typeof value === 'boolean') {
      return `${property} = ${value}`;
    } else if (typeof value === 'number') {
      return `${property} = ${value}`;
    }

    return `${property} = ${value}`;
  }

  private notifyPreviewUpdate(change: PropertyChange): void {
    // Send message to preview webview (requires coordination with TscnEditorProvider)
    vscode.commands.executeCommand('tscn.updatePreview', change);
  }

  private serializeNode(node: TscnNode): any {
    return {
      name: node.name,
      type: node.type,
      properties: node.properties,
      path: this.getNodePath(node),
    };
  }

  private getNodePath(node: TscnNode): string {
    const parts: string[] = [];
    let current: TscnNode | undefined = node;

    while (current) {
      parts.unshift(current.name);
      current = current.parent ? this.findNodeByName(current.parent) : undefined;
    }

    return parts.join('/');
  }

  private findNodeByName(name: string): TscnNode | undefined {
    // Implementation depends on scene cache
    return undefined;
  }

  private escapeRegex(str: string): string {
    return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  }

  private getWebviewHtml(webview: vscode.Webview): string {
    const scriptUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'propertyEditor.js')
    );

    const styleUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'propertyEditor.css')
    );

    return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <meta http-equiv="Content-Security-Policy" content="default-src 'none'; style-src ${webview.cspSource} 'unsafe-inline'; script-src ${webview.cspSource};">
  <link href="${styleUri}" rel="stylesheet">
  <title>Property Editor</title>
</head>
<body>
  <div id="property-container">
    <div id="no-selection">No node selected</div>
    <div id="property-list" style="display: none;"></div>
  </div>
  <script src="${scriptUri}"></script>
</body>
</html>`;
  }

  onPropertyChange(callback: (change: PropertyChange) => void): void {
    this.onPropertyChangeCallback = callback;
  }
}
```

### Webview Property Editor UI

```typescript
// media/propertyEditor.js
(function() {
  const vscode = acquireVsCodeApi();
  let currentNode = null;

  // Listen for messages from extension
  window.addEventListener('message', event => {
    const message = event.data;

    switch (message.command) {
      case 'selectNode':
        displayNodeProperties(message.node);
        break;
      case 'updateProperty':
        updatePropertyDisplay(message.property, message.value);
        break;
    }
  });

  function displayNodeProperties(node) {
    currentNode = node;

    const noSelection = document.getElementById('no-selection');
    const propertyList = document.getElementById('property-list');

    noSelection.style.display = 'none';
    propertyList.style.display = 'block';
    propertyList.innerHTML = '';

    // Add node header
    const header = document.createElement('div');
    header.className = 'property-header';
    header.innerHTML = `
      <h3>${node.name}</h3>
      <span class="node-type">${node.type}</span>
    `;
    propertyList.appendChild(header);

    // Add editable properties
    const properties = parseProperties(node.properties);
    properties.forEach(prop => {
      const editor = createPropertyEditor(prop);
      propertyList.appendChild(editor);
    });
  }

  function parseProperties(properties) {
    const parsed = [];

    for (const [key, value] of Object.entries(properties)) {
      const type = inferPropertyType(value);
      parsed.push({
        name: key,
        value: value,
        type: type,
        editable: isPropertyEditable(key, type),
      });
    }

    return parsed;
  }

  function inferPropertyType(value) {
    if (typeof value === 'string') {
      if (value.startsWith('Vector3(')) return 'vector3';
      if (value.startsWith('Vector2(')) return 'vector2';
      if (value.startsWith('Color(')) return 'color';
      if (value.match(/^[+-]?\d+(\.\d+)?$/)) return 'float';
      if (value.match(/^[+-]?\d+$/)) return 'int';
      if (value === 'true' || value === 'false') return 'bool';
      return 'string';
    }
    return 'unknown';
  }

  function isPropertyEditable(name, type) {
    // Some properties shouldn't be edited through UI
    const nonEditable = ['parent', 'instance', 'script'];
    return !nonEditable.includes(name);
  }

  function createPropertyEditor(prop) {
    const container = document.createElement('div');
    container.className = 'property-row';

    const label = document.createElement('label');
    label.textContent = prop.name;
    container.appendChild(label);

    let input;

    switch (prop.type) {
      case 'vector3':
        input = createVector3Editor(prop);
        break;
      case 'color':
        input = createColorEditor(prop);
        break;
      case 'float':
      case 'int':
        input = createNumberEditor(prop);
        break;
      case 'bool':
        input = createBoolEditor(prop);
        break;
      default:
        input = createStringEditor(prop);
    }

    if (!prop.editable) {
      input.disabled = true;
    }

    container.appendChild(input);
    return container;
  }

  function createVector3Editor(prop) {
    const container = document.createElement('div');
    container.className = 'vector3-editor';

    // Parse Vector3(x, y, z)
    const match = prop.value.match(/Vector3\((.*)\)/);
    const values = match ? match[1].split(',').map(v => parseFloat(v.trim())) : [0, 0, 0];

    ['x', 'y', 'z'].forEach((axis, index) => {
      const input = document.createElement('input');
      input.type = 'number';
      input.step = '0.1';
      input.value = values[index];
      input.dataset.axis = axis;
      input.dataset.property = prop.name;

      input.addEventListener('input', debounce(() => {
        handleVector3Change(prop.name, container);
      }, 300));

      const axisLabel = document.createElement('span');
      axisLabel.textContent = axis.toUpperCase();
      axisLabel.className = 'axis-label';

      container.appendChild(axisLabel);
      container.appendChild(input);
    });

    return container;
  }

  function createColorEditor(prop) {
    const input = document.createElement('input');
    input.type = 'color';

    // Parse Color(r, g, b, a) and convert to hex
    const match = prop.value.match(/Color\((.*)\)/);
    const values = match ? match[1].split(',').map(v => parseFloat(v.trim())) : [1, 1, 1, 1];
    input.value = rgbToHex(values[0], values[1], values[2]);

    input.addEventListener('input', debounce(() => {
      const hex = input.value;
      const rgb = hexToRgb(hex);
      const colorValue = `Color(${rgb.r}, ${rgb.g}, ${rgb.b}, 1)`;

      vscode.postMessage({
        command: 'propertyChanged',
        nodePath: currentNode.path,
        property: prop.name,
        value: colorValue,
        oldValue: prop.value,
      });
    }, 300));

    return input;
  }

  function createNumberEditor(prop) {
    const input = document.createElement('input');
    input.type = 'number';
    input.step = prop.type === 'int' ? '1' : '0.01';
    input.value = prop.value;

    input.addEventListener('input', debounce(() => {
      const value = prop.type === 'int' ? parseInt(input.value) : parseFloat(input.value);

      vscode.postMessage({
        command: 'propertyChanged',
        nodePath: currentNode.path,
        property: prop.name,
        value: value,
        oldValue: prop.value,
      });
    }, 300));

    return input;
  }

  function createBoolEditor(prop) {
    const input = document.createElement('input');
    input.type = 'checkbox';
    input.checked = prop.value === 'true';

    input.addEventListener('change', () => {
      vscode.postMessage({
        command: 'propertyChanged',
        nodePath: currentNode.path,
        property: prop.name,
        value: input.checked,
        oldValue: prop.value,
      });
    });

    return input;
  }

  function createStringEditor(prop) {
    const input = document.createElement('input');
    input.type = 'text';
    input.value = prop.value;

    input.addEventListener('input', debounce(() => {
      vscode.postMessage({
        command: 'propertyChanged',
        nodePath: currentNode.path,
        property: prop.name,
        value: input.value,
        oldValue: prop.value,
      });
    }, 500));

    return input;
  }

  function handleVector3Change(propertyName, container) {
    const inputs = container.querySelectorAll('input');
    const values = Array.from(inputs).map(input => parseFloat(input.value));
    const vectorValue = `Vector3(${values.join(', ')})`;

    vscode.postMessage({
      command: 'propertyChanged',
      nodePath: currentNode.path,
      property: propertyName,
      value: vectorValue,
      oldValue: currentNode.properties[propertyName],
    });
  }

  function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
      const later = () => {
        clearTimeout(timeout);
        func(...args);
      };
      clearTimeout(timeout);
      timeout = setTimeout(later, wait);
    };
  }

  function rgbToHex(r, g, b) {
    const toHex = (c) => {
      const hex = Math.round(c * 255).toString(16);
      return hex.length === 1 ? '0' + hex : hex;
    };
    return `#${toHex(r)}${toHex(g)}${toHex(b)}`;
  }

  function hexToRgb(hex) {
    const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    return result ? {
      r: (parseInt(result[1], 16) / 255).toFixed(3),
      g: (parseInt(result[2], 16) / 255).toFixed(3),
      b: (parseInt(result[3], 16) / 255).toFixed(3),
    } : { r: 1, g: 1, b: 1 };
  }

  function updatePropertyDisplay(property, value) {
    // Update UI when property changes externally
    // (e.g., from undo/redo or direct text editing)
  }
})();
```

### Preview Webview Update Handler

```typescript
// previewUpdateHandler.ts
export class PreviewUpdateHandler {
  constructor(
    private scene: THREE.Scene,
    private nodeCache: Map<string, THREE.Object3D>
  ) {}

  handlePropertyChange(change: PropertyChange): void {
    const object = this.nodeCache.get(change.nodePath);
    if (!object) return;

    switch (change.property) {
      case 'transform':
        this.updateTransform(object, change.value);
        break;
      case 'position':
        this.updatePosition(object, change.value);
        break;
      case 'rotation':
        this.updateRotation(object, change.value);
        break;
      case 'scale':
        this.updateScale(object, change.value);
        break;
      case 'albedo_color':
        this.updateMaterialColor(object, change.value);
        break;
      default:
        this.updateGenericProperty(object, change.property, change.value);
    }

    // Trigger re-render
    this.requestRender();
  }

  private updatePosition(object: THREE.Object3D, value: string): void {
    // Parse Vector3(x, y, z)
    const match = value.match(/Vector3\((.*)\)/);
    if (!match) return;

    const [x, y, z] = match[1].split(',').map(v => parseFloat(v.trim()));
    object.position.set(x, y, z);
  }

  private updateRotation(object: THREE.Object3D, value: string): void {
    // Parse Vector3(x, y, z) in degrees, convert to radians
    const match = value.match(/Vector3\((.*)\)/);
    if (!match) return;

    const [x, y, z] = match[1].split(',').map(v => parseFloat(v.trim()) * (Math.PI / 180));
    object.rotation.set(x, y, z);
  }

  private updateScale(object: THREE.Object3D, value: string): void {
    const match = value.match(/Vector3\((.*)\)/);
    if (!match) return;

    const [x, y, z] = match[1].split(',').map(v => parseFloat(v.trim()));
    object.scale.set(x, y, z);
  }

  private updateMaterialColor(object: THREE.Object3D, value: string): void {
    if (!(object instanceof THREE.Mesh)) return;
    if (!object.material) return;

    // Parse Color(r, g, b, a)
    const match = value.match(/Color\((.*)\)/);
    if (!match) return;

    const [r, g, b] = match[1].split(',').map(v => parseFloat(v.trim()));

    const material = object.material as THREE.MeshStandardMaterial;
    material.color.setRGB(r, g, b);
  }

  private updateTransform(object: THREE.Object3D, value: string): void {
    // Parse Transform3D(...) and update matrix
    // Implementation depends on Transform3D format
  }

  private updateGenericProperty(
    object: THREE.Object3D,
    property: string,
    value: any
  ): void {
    // Handle other property types
    console.log(`Updating ${property} to ${value}`);
  }

  private requestRender(): void {
    // Trigger render loop
    window.dispatchEvent(new CustomEvent('scene-updated'));
```
