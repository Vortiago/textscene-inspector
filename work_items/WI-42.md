# WI-42: Animation Preview Support

## Overview

Implement animation playback and preview capabilities for Godot AnimationPlayer nodes, enabling developers to view and test animations directly in the VS Code preview without running the game. Support timeline scrubbing, playback controls, animation blending visualization, and performance profiling for animations.

## Implementation Requirements

From TODO.md Phase 11: Animation and Performance:

- Parse AnimationPlayer nodes and animations
- Add playback controls (play, pause, scrub timeline)
- Preview animations in 3D viewport
- Support animation blending visualization
- Display animation tracks and keyframes
- Show animation curves
- Support animation speed control
- Handle multiple AnimationPlayer nodes
- Visualize blend trees (AnimationTree support)

## TSCN Animation Format

### AnimationPlayer Node

```ini
[node name="AnimationPlayer" type="AnimationPlayer" parent="."]
libraries = {
"": SubResource("AnimationLibrary_abc123")
}
```

### Animation Library SubResource

```ini
[sub_resource type="AnimationLibrary" id="AnimationLibrary_abc123"]
_data = {
"idle": SubResource("Animation_idle"),
"walk": SubResource("Animation_walk"),
"run": SubResource("Animation_run")
}
```

### Animation SubResource

```ini
[sub_resource type="Animation" id="Animation_idle"]
length = 1.0
loop_mode = 1
tracks/0/type = "value"
tracks/0/imported = false
tracks/0/enabled = true
tracks/0/path = NodePath("Sprite:position")
tracks/0/interp = 1
tracks/0/loop_wrap = true
tracks/0/keys = {
"times": PackedFloat32Array(0, 0.5, 1),
"transitions": PackedFloat32Array(1, 1, 1),
"update": 0,
"values": [Vector2(0, 0), Vector2(0, -10), Vector2(0, 0)]
}
```

## Implementation Notes

### Animation Parser

```typescript
// animationParser.ts
export interface AnimationTrack {
  type: 'value' | 'transform' | 'method' | 'bezier' | 'audio' | 'animation';
  path: string;
  interp: number;
  enabled: boolean;
  keys: AnimationKeyframe[];
}

export interface AnimationKeyframe {
  time: number;
  value: any;
  transition: number;
}

export interface Animation {
  name: string;
  length: number;
  loopMode: number;
  step: number;
  tracks: AnimationTrack[];
}

export interface AnimationLibrary {
  animations: Map<string, Animation>;
}

export class AnimationParser {
  parseAnimationPlayer(node: TscnNode, sceneData: TscnData): AnimationLibrary | null {
    const librariesProp = node.properties.get('libraries');
    if (!librariesProp) return null;

    // Parse library reference
    const libraryMatch = /SubResource\("([^"]+)"\)/.exec(librariesProp);
    if (!libraryMatch) return null;

    const libraryId = libraryMatch[1];
    const libraryResource = sceneData.subResources.get(libraryId);

    if (!libraryResource || libraryResource.type !== 'AnimationLibrary') {
      return null;
    }

    return this.parseAnimationLibrary(libraryResource);
  }

  private parseAnimationLibrary(resource: SubResource): AnimationLibrary {
    const library: AnimationLibrary = {
      animations: new Map(),
    };

    const dataProp = resource.properties.get('_data');
    if (!dataProp) return library;

    // Parse animation references from _data
    const animRefs = this.parseAnimationReferences(dataProp);

    for (const [name, animId] of animRefs) {
      const animResource = this.sceneData.subResources.get(animId);
      if (animResource && animResource.type === 'Animation') {
        library.animations.set(name, this.parseAnimation(name, animResource));
      }
    }

    return library;
  }

  private parseAnimation(name: string, resource: SubResource): Animation {
    const animation: Animation = {
      name,
      length: parseFloat(resource.properties.get('length') || '1.0'),
      loopMode: parseInt(resource.properties.get('loop_mode') || '0'),
      step: parseFloat(resource.properties.get('step') || '0.1'),
      tracks: [],
    };

    // Parse tracks
    let trackIndex = 0;
    while (true) {
      const trackType = resource.properties.get(`tracks/${trackIndex}/type`);
      if (!trackType) break;

      const track: AnimationTrack = {
        type: trackType as any,
        path: resource.properties.get(`tracks/${trackIndex}/path`) || '',
        interp: parseInt(resource.properties.get(`tracks/${trackIndex}/interp`) || '1'),
        enabled: resource.properties.get(`tracks/${trackIndex}/enabled`) !== 'false',
        keys: this.parseTrackKeys(resource, trackIndex),
      };

      animation.tracks.push(track);
      trackIndex++;
    }

    return animation;
  }

  private parseTrackKeys(resource: SubResource, trackIndex: number): AnimationKeyframe[] {
    const keysProp = resource.properties.get(`tracks/${trackIndex}/keys`);
    if (!keysProp) return [];

    const keys: AnimationKeyframe[] = [];

    // Parse PackedFloat32Array for times
    const timesMatch = /"times": PackedFloat32Array\(([^)]+)\)/.exec(keysProp);
    const transitionsMatch = /"transitions": PackedFloat32Array\(([^)]+)\)/.exec(keysProp);
    const valuesMatch = /"values": \[([^\]]+)\]/.exec(keysProp);

    if (!timesMatch || !valuesMatch) return [];

    const times = timesMatch[1].split(',').map(s => parseFloat(s.trim()));
    const transitions = transitionsMatch
      ? transitionsMatch[1].split(',').map(s => parseFloat(s.trim()))
      : times.map(() => 1.0);

    const values = this.parseValueArray(valuesMatch[1]);

    for (let i = 0; i < times.length; i++) {
      keys.push({
        time: times[i],
        value: values[i],
        transition: transitions[i] || 1.0,
      });
    }

    return keys;
  }

  private parseValueArray(valuesStr: string): any[] {
    const values: any[] = [];

    // Handle different value types
    const vectorMatches = valuesStr.matchAll(/Vector[23]\(([^)]+)\)/g);
    for (const match of vectorMatches) {
      const components = match[1].split(',').map(s => parseFloat(s.trim()));
      values.push(components);
    }

    // Handle simple numeric values
    if (values.length === 0) {
      return valuesStr.split(',').map(s => s.trim());
    }

    return values;
  }

  private parseAnimationReferences(dataStr: string): Map<string, string> {
    const refs = new Map<string, string>();

    // Match: "animation_name": SubResource("id")
    const regex = /"([^"]+)":\s*SubResource\("([^"]+)"\)/g;
    let match;

    while ((match = regex.exec(dataStr)) !== null) {
      refs.set(match[1], match[2]);
    }

    return refs;
  }
}
```

### Animation Player Component

```typescript
// animationPlayer.ts
import * as THREE from 'three';

export class AnimationPlayerComponent {
  private mixer: THREE.AnimationMixer | null = null;
  private actions: Map<string, THREE.AnimationAction> = new Map();
  private currentAction: THREE.AnimationAction | null = null;
  private clock: THREE.Clock = new THREE.Clock();

  constructor(
    private rootObject: THREE.Object3D,
    private library: AnimationLibrary
  ) {
    this.setupAnimations();
  }

  private setupAnimations(): void {
    this.mixer = new THREE.AnimationMixer(this.rootObject);

    for (const [name, animation] of this.library.animations) {
      const clip = this.createAnimationClip(animation);
      const action = this.mixer.clipAction(clip);
      this.actions.set(name, action);
    }
  }

  private createAnimationClip(animation: Animation): THREE.AnimationClip {
    const tracks: THREE.KeyframeTrack[] = [];

    for (const track of animation.tracks) {
      if (track.type === 'value') {
        const threeTrack = this.createValueTrack(track);
        if (threeTrack) tracks.push(threeTrack);
      } else if (track.type === 'transform') {
        const threeTrack = this.createTransformTrack(track);
        if (threeTrack) tracks.push(threeTrack);
      }
    }

    return new THREE.AnimationClip(animation.name, animation.length, tracks);
  }

  private createValueTrack(track: AnimationTrack): THREE.KeyframeTrack | null {
    // Extract node path and property
    const pathMatch = /NodePath\("([^"]+):([^"]+)"\)/.exec(track.path);
    if (!pathMatch) return null;

    const nodePath = pathMatch[1];
    const property = pathMatch[2];

    // Find target object
    const targetObject = this.findObjectByPath(nodePath);
    if (!targetObject) return null;

    // Prepare keyframe data
    const times = track.keys.map(k => k.time);
    const values: number[] = [];

    for (const key of track.keys) {
      if (Array.isArray(key.value)) {
        values.push(...key.value);
      } else {
        values.push(parseFloat(key.value));
      }
    }

    // Create appropriate track type
    if (property === 'position') {
      return new THREE.VectorKeyframeTrack(
        `${targetObject.name}.position`,
        times,
        values
      );
    } else if (property === 'rotation') {
      return new THREE.QuaternionKeyframeTrack(
        `${targetObject.name}.quaternion`,
        times,
        values
      );
    } else if (property === 'scale') {
      return new THREE.VectorKeyframeTrack(
        `${targetObject.name}.scale`,
        times,
        values
      );
    }

    return null;
  }

  private createTransformTrack(track: AnimationTrack): THREE.KeyframeTrack | null {
    const pathMatch = /NodePath\("([^"]+)"\)/.exec(track.path);
    if (!pathMatch) return null;

    const nodePath = pathMatch[1];
    const targetObject = this.findObjectByPath(nodePath);
    if (!targetObject) return null;

    const times = track.keys.map(k => k.time);
    const values: number[] = [];

    for (const key of track.keys) {
      // Godot Transform3D has 12 components
      if (Array.isArray(key.value) && key.value.length === 12) {
        // Convert to position (translation)
        values.push(key.value[9], key.value[10], key.value[11]);
      }
    }

    return new THREE.VectorKeyframeTrack(
      `${targetObject.name}.position`,
      times,
      values
    );
  }

  private findObjectByPath(path: string): THREE.Object3D | null {
    const parts = path.split('/');
    let current = this.rootObject;

    for (const part of parts) {
      if (part === '.') continue;
      const child = current.children.find(c => c.name === part);
      if (!child) return null;
      current = child;
    }

    return current;
  }

  play(animationName?: string): void {
    if (animationName) {
      this.currentAction?.stop();
      this.currentAction = this.actions.get(animationName) || null;
    }

    if (this.currentAction) {
      this.currentAction.reset();
      this.currentAction.play();
    }
  }

  pause(): void {
    this.currentAction?.paused = true;
  }

  resume(): void {
    if (this.currentAction) {
      this.currentAction.paused = false;
    }
  }

  stop(): void {
    this.currentAction?.stop();
  }

  setTime(time: number): void {
    if (this.currentAction) {
      this.currentAction.time = time;
      this.mixer?.update(0); // Force update without advancing time
    }
  }

  setSpeed(speed: number): void {
    if (this.currentAction) {
      this.currentAction.timeScale = speed;
    }
  }

  update(): void {
    if (this.mixer) {
      const delta = this.clock.getDelta();
      this.mixer.update(delta);
    }
  }

  getAnimationNames(): string[] {
    return Array.from(this.actions.keys());
  }

  getCurrentTime(): number {
    return this.currentAction?.time || 0;
  }

  getDuration(): number {
    return this.currentAction?.getClip().duration || 0;
  }

  isPlaying(): boolean {
    return this.currentAction?.isRunning() || false;
  }
}
```

```typescript
        const speed = parseFloat(e.target.value);
        document.getElementById('speed-value').textContent = speed.toFixed(1) + 'x';
        vscode.postMessage({ type: 'setSpeed', speed });

      document.getElementById('loop-checkbox').addEventListener('change', (e) => {
        vscode.postMessage({ type: 'setLoop', loop: e.target.checked });
      });

      // Timeline scrubbing
      let isScrubbing = false;
      const timelineSlider = document.getElementById('timeline-slider');

      timelineSlider.addEventListener('mousedown', () => {
        isScrubbing = true;
        vscode.postMessage({ type: 'pauseAnimation' });
      });

      timelineSlider.addEventListener('input', (e) => {
        const time = parseFloat(e.target.value);
        document.getElementById('current-time').textContent = time.toFixed(2) + 's';
        vscode.postMessage({ type: 'scrubTimeline', time });
      });

      timelineSlider.addEventListener('mouseup', () => {
        isScrubbing = false;
      });

      // Update timeline from renderer
      window.addEventListener('message', (event) => {
        const message = event.data;

        if (message.type === 'updateTimeline') {
          if (!isScrubbing) {
            timelineSlider.value = message.time;
            document.getElementById('current-time').textContent =
              message.time.toFixed(2) + 's';
          }

          const playBtn = document.getElementById('play-btn');
          playBtn.textContent = message.isPlaying ? '⏸' : '▶';
        }
      });
    `;
  }
}
```
