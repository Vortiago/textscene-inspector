# WI-35: Signal Connection Visualization

## Overview

Implement visualization of signal connections between nodes in TSCN scenes, helping developers understand the event flow and inter-node communication. Display signal connections in the scene tree view, provide navigation to signal handlers, and show connection details including signal names, target nodes, and methods.

## Implementation Requirements

From TODO.md Phase 10: Advanced Navigation:

- Parse signal connections from TSCN files
- Display signal connections in scene tree view
- Visual indicators for nodes with signals (connected/disconnected)
- Navigate from signal to connected method
- Show signal connection details panel
- List all signals for a node
- Show which nodes are connected to which signals
- Highlight signal flow paths
- Support for custom signals
- Connection validation (detect broken connections)
- Quick actions to find signal definitions

## Implementation Notes

### TSCN Connection Parser

```typescript
// signalConnectionParser.ts
export interface SignalConnection {
  signal: string;
  from: string;        // Source node path
  to: string;          // Target node path
  method: string;      // Method name to call
  binds?: any[];       // Optional bound arguments
  flags?: number;      // Connection flags
}

export class SignalConnectionParser {
  parseConnections(tscnContent: string): SignalConnection[] {
    const connections: SignalConnection[] = [];
    const lines = tscnContent.split('\n');

    for (const line of lines) {
      if (line.trim().startsWith('[connection')) {
        const connection = this.parseConnectionLine(line);
        if (connection) {
          connections.push(connection);
        }
      }
    }

    return connections;
  }

  private parseConnectionLine(line: string): SignalConnection | null {
    // Parse: [connection signal="pressed" from="Button" to="." method="_on_button_pressed"]
    const signalMatch = /signal="([^"]+)"/.exec(line);
    const fromMatch = /from="([^"]+)"/.exec(line);
    const toMatch = /to="([^"]+)"/.exec(line);
    const methodMatch = /method="([^"]+)"/.exec(line);

    if (!signalMatch || !fromMatch || !toMatch || !methodMatch) {
      return null;
    }

    const connection: SignalConnection = {
      signal: signalMatch[1],
      from: fromMatch[1],
      to: toMatch[1],
      method: methodMatch[1],
    };

    // Parse optional binds
    const bindsMatch = /binds=\[(.*?)\]/.exec(line);
    if (bindsMatch) {
      connection.binds = this.parseBinds(bindsMatch[1]);
    }

    // Parse optional flags
    const flagsMatch = /flags=(\d+)/.exec(line);
    if (flagsMatch) {
      connection.flags = parseInt(flagsMatch[1]);
    }

    return connection;
  }

  private parseBinds(bindsStr: string): any[] {
    // Parse bind values: "value1", 123, true, etc.
    return bindsStr.split(',').map(s => {
      s = s.trim();
      if (s.startsWith('"') && s.endsWith('"')) {
        return s.slice(1, -1); // String
      }
      if (s === 'true') return true;
      if (s === 'false') return false;
      if (!isNaN(Number(s))) return Number(s);
      return s;
    });
  }

  getConnectionsForNode(
    connections: SignalConnection[],
    nodePath: string
  ): { outgoing: SignalConnection[]; incoming: SignalConnection[] } {
    return {
      outgoing: connections.filter(c => this.normalizeNodePath(c.from) === nodePath),
      incoming: connections.filter(c => this.normalizeNodePath(c.to) === nodePath),
    };
  }

  private normalizeNodePath(path: string): string {
    // Normalize "." to root and resolve relative paths
    if (path === '.') return '';
    return path;
  }
}
```

### Signal Visualization in Tree View

```typescript
// signalTreeDecorator.ts
import * as vscode from 'vscode';
import { SignalConnection } from './signalConnectionParser';

export class SignalTreeDecorator {
  private readonly _onDidChangeTreeData = new vscode.EventEmitter<void>();
  readonly onDidChangeTreeData = this._onDidChangeTreeData.event;

  constructor(
    private connections: SignalConnection[]
  ) {}

  updateConnections(connections: SignalConnection[]): void {
    this.connections = connections;
    this._onDidChangeTreeData.fire();
  }

  decorateTreeItem(item: vscode.TreeItem, nodePath: string): void {
    const { outgoing, incoming } = this.getNodeConnections(nodePath);

    // Add icon overlay for nodes with signals
    if (outgoing.length > 0 || incoming.length > 0) {
      const signalCount = outgoing.length + incoming.length;

      // Add badge
      item.description = `${item.description || ''} [${signalCount} signals]`;

      // Update tooltip
      const tooltip = new vscode.MarkdownString();
      tooltip.appendMarkdown(`**${item.label}**\n\n`);

      if (outgoing.length > 0) {
        tooltip.appendMarkdown(`**Outgoing Signals (${outgoing.length}):**\n`);
        outgoing.forEach(conn => {
          tooltip.appendMarkdown(`- \`${conn.signal}\` → ${conn.to}.${conn.method}\n`);
        });
        tooltip.appendMarkdown('\n');
      }

      if (incoming.length > 0) {
        tooltip.appendMarkdown(`**Incoming Signals (${incoming.length}):**\n`);
        incoming.forEach(conn => {
          tooltip.appendMarkdown(`- \`${conn.signal}\` from ${conn.from}\n`);
        });
      }

      item.tooltip = tooltip;

      // Add context value for commands
      item.contextValue = 'nodeWithSignals';
    }
  }

  private getNodeConnections(nodePath: string): {
    outgoing: SignalConnection[];
    incoming: SignalConnection[];
  } {
    return {
      outgoing: this.connections.filter(c => c.from === nodePath),
      incoming: this.connections.filter(c => c.to === nodePath),
    };
  }
}
```

```typescript
// signalConnectionPanel.ts
import * as vscode from 'vscode';
import { SignalConnection, SignalConnectionParser } from './signalConnectionParser';

export class SignalConnectionPanel {
  private panel: vscode.WebviewPanel | undefined;
  private connections: SignalConnection[] = [];

  constructor(
    private readonly extensionUri: vscode.Uri,
    private readonly context: vscode.ExtensionContext
  ) {}

  show(connections: SignalConnection[], selectedNode?: string): void {
    if (this.panel) {
      this.panel.reveal(vscode.ViewColumn.Two);
    } else {
      this.panel = this.createPanel();
    }

    this.connections = connections;
    this.updateContent(connections, selectedNode);
  }

  private createPanel(): vscode.WebviewPanel {
    const panel = vscode.window.createWebviewPanel(
      'signalConnections',
      'Signal Connections',
      vscode.ViewColumn.Two,
      {
        enableScripts: true,
        retainContextWhenHidden: true,
      }
    );

    panel.webview.html = this.getHtmlContent(panel.webview);

    panel.webview.onDidReceiveMessage(
      async (message) => {
        switch (message.command) {
          case 'navigateToMethod':
            await this.navigateToMethod(message.nodePath, message.method);
            break;
          case 'navigateToSignal':
            await this.navigateToSignal(message.nodePath, message.signal);
            break;
        }
      },
      undefined,
      this.context.subscriptions
    );

    panel.onDidDispose(() => {
      this.panel = undefined;
    });

    return panel;
  }

  private updateContent(connections: SignalConnection[], selectedNode?: string): void {
    if (!this.panel) return;

    // Group connections by type
    const grouped = this.groupConnections(connections, selectedNode);

    this.panel.webview.postMessage({
      command: 'showConnections',
      data: {
        all: connections,
        selected: selectedNode,
        grouped: grouped,
      },
    });
  }

  private groupConnections(
    connections: SignalConnection[],
    selectedNode?: string
  ): Record<string, SignalConnection[]> {
    const grouped: Record<string, SignalConnection[]> = {
      all: connections,
      outgoing: [],
      incoming: [],
      internal: [],
    };

    if (selectedNode) {
      grouped.outgoing = connections.filter(c => c.from === selectedNode);
      grouped.incoming = connections.filter(c => c.to === selectedNode);
    }

    // Internal connections (within same subtree)
    grouped.internal = connections.filter(c => {
      return c.from.startsWith(selectedNode || '') && c.to.startsWith(selectedNode || '');
    });

    return grouped;
  }

  private async navigateToMethod(nodePath: string, methodName: string): Promise<void> {
    // Find script file for node
    const scriptUri = await this.findScriptForNode(nodePath);
    if (!scriptUri) {
      vscode.window.showWarningMessage(`Could not find script for node: ${nodePath}`);
      return;
    }

    // Open script and navigate to method
    const document = await vscode.workspace.openTextDocument(scriptUri);
    const editor = await vscode.window.showTextDocument(document);

    // Find method definition
    const text = document.getText();
    const methodRegex = new RegExp(`(func|private void)\\s+${methodName}\\s*\\(`, 'g');
    const match = methodRegex.exec(text);

    if (match) {
      const position = document.positionAt(match.index);
      editor.selection = new vscode.Selection(position, position);
      editor.revealRange(new vscode.Range(position, position));
    }
  }

  private async navigateToSignal(nodePath: string, signalName: string): Promise<void> {
    // Find script file and navigate to signal definition
    const scriptUri = await this.findScriptForNode(nodePath);
    if (!scriptUri) return;

    const document = await vscode.workspace.openTextDocument(scriptUri);
    const editor = await vscode.window.showTextDocument(document);

    const text = document.getText();
    const signalRegex = new RegExp(`signal\\s+${signalName}`, 'g');
    const match = signalRegex.exec(text);

    if (match) {
      const position = document.positionAt(match.index);
      editor.selection = new vscode.Selection(position, position);
      editor.revealRange(new vscode.Range(position, position));
    }
  }

  private async findScriptForNode(nodePath: string): Promise<vscode.Uri | undefined> {
    // Search for script files

  }

  private getHtmlContent(webview: vscode.Webview): string {
    const scriptUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'signalPanel.js')
    );

    const styleUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'signalPanel.css')
    );
```

### Webview Script

```javascript
// media/signalPanel.js
(function() {
  const vscode = acquireVsCodeApi();
  let currentData = null;
  let currentFilter = 'all';

  window.addEventListener('message', event => {
    const message = event.data;

    switch (message.command) {
      case 'showConnections':
        currentData = message.data;
        renderConnections(currentFilter);
        break;
    }
  });

  function renderConnections(filter) {
    const container = document.getElementById('connections-list');
    const noConnectionsMsg = document.getElementById('no-connections');

    if (!currentData) return;

    const connections = currentData.grouped[filter] || currentData.all;

    if (connections.length === 0) {
      container.style.display = 'none';
      noConnectionsMsg.style.display = 'block';
      return;
    }

    container.style.display = 'block';
    noConnectionsMsg.style.display = 'none';
    container.innerHTML = '';

    connections.forEach(conn => {
      const card = createConnectionCard(conn);
      container.appendChild(card);
    });
  }

  function createConnectionCard(conn) {
    const card = document.createElement('div');
    card.className = 'connection-card';

    card.innerHTML = `
      <div class="connection-header">
        <span class="signal-name">${conn.signal}</span>
        <span class="connection-arrow">→</span>
      </div>
      <div class="connection-body">
        <div class="connection-row">
          <span class="label">From:</span>
          <a href="#" class="node-link" data-node="${conn.from}" data-signal="${conn.signal}">${conn.from || 'Root'}</a>
        </div>
        <div class="connection-row">
          <span class="label">To:</span>
          <a href="#" class="node-link" data-node="${conn.to}">${conn.to || 'Root'}</a>
        </div>
        <div class="connection-row">
          <span class="label">Method:</span>
          <a href="#" class="method-link" data-node="${conn.to}" data-method="${conn.method}">${conn.method}</a>
        </div>
        ${conn.binds ? `
          <div class="connection-row">
            <span class="label">Binds:</span>
            <span class="binds-value">${JSON.stringify(conn.binds)}</span>
          </div>
        ` : ''}
      </div>
    `;

    // Add click handlers
    card.querySelectorAll('.method-link').forEach(link => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        vscode.postMessage({
          command: 'navigateToMethod',
          nodePath: e.target.dataset.node,
          method: e.target.dataset.method,
        });
      });
    });

    card.querySelectorAll('.node-link').forEach(link => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        if (e.target.dataset.signal) {
          vscode.postMessage({
            command: 'navigateToSignal',
            nodePath: e.target.dataset.node,
            signal: e.target.dataset.signal,
          });
        }
      });
    });

    return card;
  }

  // Filter buttons
  document.querySelectorAll('.filter-btn').forEach(btn => {
    btn.addEventListener('click', (e) => {
      document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
      e.target.classList.add('active');

      currentFilter = e.target.id.replace('filter-', '');
      renderConnections(currentFilter);
    });
  });
})();
```

```typescript
// Register commands
vscode.commands.registerCommand('tscn.showSignalConnections', async (nodePath?: string) => {
  const connections = connectionParser.parseConnections(documentContent);
  signalPanel.show(connections, nodePath);
});

vscode.commands.registerCommand('tscn.validateSignalConnections', async () => {
  const connections = connectionParser.parseConnections(documentContent);
  const broken = await validateConnections(connections);

  if (broken.length > 0) {
    vscode.window.showWarningMessage(
      `Found ${broken.length} broken signal connections`,
      'Show Details'
    ).then(action => {
      if (action === 'Show Details') {
        signalPanel.show(broken);
      }
    });
  } else {
    vscode.window.showInformationMessage('All signal connections are valid');
  }
});
```
