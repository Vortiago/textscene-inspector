# WI-13: Implement ExtResource Loading System

## Overview
Implement a system to parse and load external resources referenced in TSCN files. This enables scenes to reference external files like scripts, textures, materials, and other scenes, which is fundamental for scene composition and asset reuse in Godot.

## Implementation Requirements
- Parse [ext_resource] headings from TSCN files
- Build resource registry (uid → path mapping)
- Create resource loader for PackedScene references
- **Frequency**: 20+ external scene instances in typical scenes
- **Critical**: Enables scene reuse and composition

## Godot Documentation

### TSCN File Format - External Resources

**[ext_resource] Format:**
```text
[gd_scene load_steps=4 format=3 uid="uid://..."]

[ext_resource type="Script" path="res://player.gd" id="1_abcde"]
[ext_resource type="Texture2D" uid="uid://..." path="res://icon.png" id="2_xyzpd"]
[ext_resource type="PackedScene" path="res://scenes/enemy.tscn" id="3_fghij"]

[node name="Player" type="Node2D"]
script = ExtResource("1_abcde")

[node name="Sprite" type="Sprite2D" parent="."]
texture = ExtResource("2_xyzpd")
```

**Key Components:**
- `type`: Resource type (Script, Texture2D, PackedScene, Material, etc.)
- `path`: File path using `res://` protocol (project resources)
- `uid`: Globally unique identifier for the resource
- `id`: Local identifier used within the scene file (e.g., "1_abcde")

**Resource Reference Pattern:**
```text
property_name = ExtResource("local_id")
```

### Resource Loading in Godot

**Dynamic Loading:**
```gdscript
# Load resource at runtime
var texture = load("res://icon.png")

# Load scene
var scene = load("res://scenes/my_level.tscn")
var instance = scene.instantiate()
add_child(instance)
```

**Compile-time Preloading:**
```gdscript
# Load resource at compile-time (more efficient)
const player_scene = preload("res://scenes/Player.tscn")

func spawn_player():
    var instance = player_scene.instantiate()
    add_child(instance)
```

**Asynchronous Loading:**
```gdscript
# Load large resources in background thread
var loading_thread_rid = ResourceLoader.load_threaded_request("res://scenes/MyLargeScene.tscn")

func _process(delta):
    var status = ResourceLoader.load_threaded_get_status(loading_thread_rid)

    if status == ResourceLoader.THREAD_LOAD_LOADED:
        var loaded_resource = ResourceLoader.load_threaded_get(loading_thread_rid)
        var instance = loaded_resource.instantiate()
        add_child(instance)
        loading_thread_rid = RID()
    elif status == ResourceLoader.THREAD_LOAD_FAILED:
        print("Error: Resource loading failed.")
```

### Resource Path Types

**Project Resources (res://):**
- Files within the project directory
- Exported with the game
- Example: `res://assets/images/player.png`

**User Data (user://):**
- Persistent storage for save files, settings
- Platform-specific location
- Example: `user://savegame.dat`

**Absolute Paths:**
```gdscript
# Convert res:// to absolute path
var absolute_path = ProjectSettings.globalize_path("res://icon.png")
```

### Resource Types Commonly Referenced

1. **PackedScene** - External scene files (.tscn)
2. **Script** - GDScript (.gd) or C# (.cs) files
3. **Texture2D** - Image files (.png, .jpg, .svg)
4. **Material** - Material resources (.tres, .material)
5. **AudioStream** - Audio files (.wav, .ogg, .mp3)
6. **Shader** - Shader files (.gdshader)
7. **FontFile** - Font files (.ttf, .otf)

```typescript
  path: string;        // File path (e.g., "res://scenes/player.tscn")
  uid?: string;        // Global unique ID (e.g., "uid://abc123")
  id: string;          // Local ID used in file (e.g., "1_abcde")
}
```

### Resource Registry Pattern

```typescript
class ResourceRegistry {
  private resources: Map<string, ExtResource> = new Map();
  private loadedResources: Map<string, any> = new Map();

  // Register external resource from parsed TSCN
  register(resource: ExtResource): void {
    this.resources.set(resource.id, resource);
    if (resource.uid) {
      this.resources.set(resource.uid, resource);
    }
  }

  // Get resource metadata
  get(idOrUid: string): ExtResource | undefined {
    return this.resources.get(idOrUid);
  }

  // Load and cache resource
  async loadResource(id: string): Promise<any> {
    // Check cache first
    if (this.loadedResources.has(id)) {
      return this.loadedResources.get(id);
    }

    const resource = this.resources.get(id);
    if (!resource) {
      throw new Error(`Resource not found: ${id}`);
    }

    // Load resource based on type
    const loaded = await this.loadByType(resource);
    this.loadedResources.set(id, loaded);

    return loaded;
  }

  private async loadByType(resource: ExtResource): Promise<any> {
    switch (resource.type) {
      case 'PackedScene':
        return await this.loadScene(resource.path);
      case 'Texture2D':
        return await this.loadTexture(resource.path);
      case 'Script':
        return await this.loadScript(resource.path);
      default:
        console.warn(`Unsupported resource type: ${resource.type}`);
        return null;
    }
  }

  private async loadScene(path: string): Promise<TscnScene> {
    // Load and parse .tscn file
    const content = await fetch(path).then(r => r.text());
    const parser = new TscnParser();
    return parser.parse(content);
  }

  private async loadTexture(path: string): Promise<THREE.Texture> {
    // Load texture using three.js TextureLoader
    const loader = new THREE.TextureLoader();
    return new Promise((resolve, reject) => {
      loader.load(
        path,
        (texture) => resolve(texture),
        undefined,
        (error) => reject(error)
      );
    });
  }

  private async loadScript(path: string): Promise<string> {
    // Load script content
    return await fetch(path).then(r => r.text());
  }
}
```

### Parsing [ext_resource] Sections

```typescript
function parseExtResources(tscnContent: string): ExtResource[] {
  const resources: ExtResource[] = [];
  const lines = tscnContent.split('\n');

  for (const line of lines) {
    // Match [ext_resource ...] pattern
    if (line.startsWith('[ext_resource')) {
      const resource = parseExtResourceLine(line);
      if (resource) {
        resources.push(resource);
      }
    }
  }

  return resources;
}

function parseExtResourceLine(line: string): ExtResource | null {
  // Example: [ext_resource type="Script" path="res://player.gd" id="1_abcde"]
  const typeMatch = line.match(/type="([^"]+)"/);
  const pathMatch = line.match(/path="([^"]+)"/);
  const uidMatch = line.match(/uid="([^"]+)"/);
  const idMatch = line.match(/id="([^"]+)"/);

  if (!typeMatch || !pathMatch || !idMatch) {
    return null;
  }

  return {
    type: typeMatch[1],
    path: pathMatch[1],
    uid: uidMatch?.[1],
    id: idMatch[1],
  };
}
```

### Key Considerations

1. **Path Resolution**:
   - Convert `res://` paths to actual file system paths
   - Handle relative vs absolute paths
   - Support different environments (web, desktop, VS Code extension)

2. **Resource Caching**:
   - Cache loaded resources to avoid re-loading
   - Implement cache invalidation strategy
   - Consider memory management for large scenes

3. **Asynchronous Loading**:
   - All resource loading should be async
   - Show loading states in UI
   - Handle loading errors gracefully

4. **Resource Types**:
   - Not all resource types need to be loaded for preview
   - Priority: PackedScene, Texture2D, Material
   - Scripts can be loaded but not executed

5. **UID System**:
   - UIDs are globally unique across project
   - Can be used for resource lookup
   - Not all resources have UIDs (older formats)

6. **Circular Dependencies**:
   - Detect scene A → scene B → scene A loops
   - Track loading stack to prevent infinite recursion
   - Show error or warning for circular references

### Complete Resource Registry Implementation

```typescript
// packages/textscene-renderer/src/resources/ResourceRegistry.ts
import { TscnParser } from '../parser/TscnParser';
import * as THREE from 'three';

export interface ExtResource {
  type: string;
  path: string;
  uid?: string;
  id: string;
}

export class ResourceRegistry {
  private resources: Map<string, ExtResource> = new Map();
  private loadedResources: Map<string, any> = new Map();
  private loadingPromises: Map<string, Promise<any>> = new Map();
  private loadingStack: Set<string> = new Set();

  register(resource: ExtResource): void {
    this.resources.set(resource.id, resource);
    if (resource.uid) {
      this.resources.set(resource.uid, resource);
    }
  }

  get(idOrUid: string): ExtResource | undefined {
    return this.resources.get(idOrUid);
  }

  async loadResource(id: string): Promise<any> {
    // Check cache
    if (this.loadedResources.has(id)) {
      return this.loadedResources.get(id);
    }

    // Check if already loading (avoid duplicate loads)
    if (this.loadingPromises.has(id)) {
      return await this.loadingPromises.get(id);
    }

    // Check for circular dependencies
    if (this.loadingStack.has(id)) {
      throw new Error(`Circular dependency detected: ${id}`);
    }

    const resource = this.resources.get(id);
    if (!resource) {
      throw new Error(`Resource not found: ${id}`);
    }

    // Track loading
    this.loadingStack.add(id);

    // Create loading promise
    const loadingPromise = this.loadByType(resource)
      .then(loaded => {
        this.loadedResources.set(id, loaded);
        this.loadingStack.delete(id);
        this.loadingPromises.delete(id);
        return loaded;
      })
      .catch(error => {
        this.loadingStack.delete(id);
        this.loadingPromises.delete(id);
        throw error;
      });

    this.loadingPromises.set(id, loadingPromise);
    return await loadingPromise;
  }

  private async loadByType(resource: ExtResource): Promise<any> {
    switch (resource.type) {
      case 'PackedScene':
        return await this.loadScene(resource.path);
      case 'Texture2D':
      case 'CompressedTexture2D':
        return await this.loadTexture(resource.path);
      case 'Script':
        return await this.loadScript(resource.path);
      case 'Material':
      case 'StandardMaterial3D':
      case 'ShaderMaterial':
        return await this.loadMaterial(resource.path);
      default:
        console.warn(`Unsupported resource type: ${resource.type}`);
        return null;
    }
  }

  private async loadScene(path: string): Promise<any> {
    const resolvedPath = this.resolvePath(path);
    const content = await fetch(resolvedPath).then(r => r.text());
    const parser = new TscnParser();
    return parser.parse(content);
  }

  private async loadTexture(path: string): Promise<THREE.Texture> {
    const resolvedPath = this.resolvePath(path);
    const loader = new THREE.TextureLoader();

    return new Promise((resolve, reject) => {
      loader.load(
        resolvedPath,
        (texture) => {
          texture.colorSpace = THREE.SRGBColorSpace;
          resolve(texture);
        },
        undefined,
        (error) => reject(new Error(`Failed to load texture: ${path}`))
      );
    });
  }

  private async loadScript(path: string): Promise<string> {
    const resolvedPath = this.resolvePath(path);
    return await fetch(resolvedPath).then(r => r.text());
  }

  private async loadMaterial(path: string): Promise<any> {
    // Load material resource file (.tres)
    const resolvedPath = this.resolvePath(path);
    const content = await fetch(resolvedPath).then(r => r.text());
    // Parse material resource format (similar to TSCN)
    // Return material data
    return content;
  }

  private resolvePath(godotPath: string): string {
    // Convert res:// to actual path
    // This depends on the environment (web, VS Code, etc.)

    if (godotPath.startsWith('res://')) {
      // Remove res:// prefix
      const relativePath = godotPath.substring(6);

      // In web environment
      if (typeof window !== 'undefined') {
        return `/project/${relativePath}`;
      }

      // In Node environment (VS Code extension)
      // Use workspace path
      return relativePath;
    }

    return godotPath;
  }

  clear(): void {
    this.resources.clear();
    this.loadedResources.clear();
    this.loadingPromises.clear();
    this.loadingStack.clear();
  }
}
```

### ExtResource Parser

```typescript
// packages/textscene-renderer/src/parser/ExtResourceParser.ts
import type { ExtResource } from '../resources/ResourceRegistry';

export function parseExtResources(content: string): ExtResource[] {
  const resources: ExtResource[] = [];
  const lines = content.split('\n');

  for (const line of lines) {
    if (line.trim().startsWith('[ext_resource')) {
      const resource = parseExtResourceLine(line);
      if (resource) {
        resources.push(resource);
      }
    }
  }

  return resources;
}

function parseExtResourceLine(line: string): ExtResource | null {
  // Parse: [ext_resource type="PackedScene" path="res://scene.tscn" id="1_abc"]
  const typeMatch = line.match(/type="([^"]+)"/);
  const pathMatch = line.match(/path="([^"]+)"/);
  const uidMatch = line.match(/uid="([^"]+)"/);
  const idMatch = line.match(/id="([^"]+)"/);

  if (!typeMatch || !pathMatch || !idMatch) {
    console.warn('Invalid ext_resource line:', line);
    return null;
  }

  return {
    type: typeMatch[1],
    path: pathMatch[1],
    uid: uidMatch?.[1],
    id: idMatch[1],
  };
}

export function parseExtResourceReference(value: string): string | null {
  // Parse: ExtResource("1_abc") → "1_abc"
  const match = value.match(/ExtResource\("([^"]+)"\)/);
  return match ? match[1] : null;
}
```
