# WI-31: Node Transformation Gizmos

## Overview

Implement interactive 3D transformation gizmos using three.js TransformControls, allowing users to visually manipulate node positions, rotations, and scales directly in the preview window. This feature provides intuitive visual editing similar to Unity, Unreal Engine, and Godot's own editor, making scene composition faster and more natural.

## Implementation Requirements

From TODO.md Phase 8: Interactive Editing:

- Visual gizmos for translate, rotate, and scale operations
- Mouse-based interaction for dragging and manipulating nodes
- Mode switching (translate/rotate/scale) via keyboard or UI
- Support for world-space and local-space transformations
- Snap-to-grid functionality for precise positioning
- Visual feedback during transformation
- Sync transformations back to TSCN document
- Coordinate with OrbitControls (disable camera during gizmo drag)
- Handle multiple selected nodes
- Display current transformation values

## Implementation Notes

### TransformGizmoManager Class

```typescript
// transformGizmoManager.ts
import * as THREE from 'three';
import { TransformControls } from 'three/addons/controls/TransformControls.js';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

export type TransformMode = 'translate' | 'rotate' | 'scale';
export type TransformSpace = 'world' | 'local';

export interface TransformChangeEvent {
  nodePath: string;
  property: 'position' | 'rotation' | 'scale';
  value: THREE.Vector3 | THREE.Euler;
  oldValue: THREE.Vector3 | THREE.Euler;
}

export class TransformGizmoManager {
  private transformControls: TransformControls;
  private currentObject: THREE.Object3D | null = null;
  private currentNodePath: string | null = null;
  private orbitControls: OrbitControls;
  private snapEnabled = false;
  private onChangeCallback?: (event: TransformChangeEvent) => void;

  // Store initial transform for change detection
  private initialPosition: THREE.Vector3 = new THREE.Vector3();
  private initialRotation: THREE.Euler = new THREE.Euler();
  private initialScale: THREE.Vector3 = new THREE.Vector3();

  constructor(
    camera: THREE.Camera,
    renderer: THREE.WebGLRenderer,
    scene: THREE.Scene,
    orbitControls: OrbitControls
  ) {
    this.orbitControls = orbitControls;

    // Create transform controls
    this.transformControls = new TransformControls(camera, renderer.domElement);
    scene.add(this.transformControls);

    // Default configuration
    this.transformControls.setMode('translate');
    this.transformControls.setSpace('world');
    this.transformControls.size = 1.2;

    // Setup event handlers
    this.setupEventHandlers();

    // Setup keyboard shortcuts
    this.setupKeyboardShortcuts(renderer.domElement);
  }

  private setupEventHandlers(): void {
    // Disable orbit controls while dragging gizmo
    this.transformControls.addEventListener('dragging-changed', (event) => {
      this.orbitControls.enabled = !event.value;

      if (event.value) {
        // Dragging started - store initial values
        if (this.currentObject) {
          this.initialPosition.copy(this.currentObject.position);
          this.initialRotation.copy(this.currentObject.rotation);
          this.initialScale.copy(this.currentObject.scale);
        }
      } else {
        // Dragging ended - emit change event
        if (this.currentObject && this.currentNodePath) {
          this.emitTransformChange();
        }
      }
    });

    // Listen for transform changes
    this.transformControls.addEventListener('objectChange', () => {
      // Real-time update (can be used for live preview)
      this.updateTransformDisplay();
    });
  }

  private setupKeyboardShortcuts(domElement: HTMLElement): void {
    window.addEventListener('keydown', (event) => {
      // Only handle if gizmo is active
      if (!this.currentObject) return;

      switch (event.key.toLowerCase()) {
        case 'w':
          this.setMode('translate');
          break;
        case 'e':
          this.setMode('rotate');
          break;
        case 'r':
          this.setMode('scale');
          break;
        case 'q':
          this.toggleSpace();
          break;
        case 'x':
          if (event.shiftKey) {
            this.toggleSnapMode();
          }
          break;
        case '+':
        case '=':
          this.transformControls.size += 0.1;
          break;
        case '-':
          this.transformControls.size = Math.max(0.1, this.transformControls.size - 0.1);
          break;
        case 'escape':
          this.detach();
          break;
      }
    });
  }

  attachToObject(object: THREE.Object3D, nodePath: string): void {
    this.currentObject = object;
    this.currentNodePath = nodePath;

    // Store initial transform
    this.initialPosition.copy(object.position);
    this.initialRotation.copy(object.rotation);
    this.initialScale.copy(object.scale);

    // Attach gizmo
    this.transformControls.attach(object);

    // Show transform info
    this.updateTransformDisplay();
  }

  detach(): void {
    this.transformControls.detach();
    this.currentObject = null;
    this.currentNodePath = null;
  }

  setMode(mode: TransformMode): void {
    this.transformControls.setMode(mode);
    this.updateTransformDisplay();
  }

  getMode(): TransformMode {
    return this.transformControls.mode as TransformMode;
  }

  setSpace(space: TransformSpace): void {
    this.transformControls.setSpace(space);
  }

  toggleSpace(): void {
    const currentSpace = this.transformControls.space;
    this.setSpace(currentSpace === 'world' ? 'local' : 'world');
  }

  setSnapEnabled(enabled: boolean): void {
    this.snapEnabled = enabled;

    if (enabled) {
      // Set snap values
      this.transformControls.setTranslationSnap(0.5);
      this.transformControls.setRotationSnap(Math.PI / 12); // 15 degrees
      this.transformControls.setScaleSnap(0.1);
    } else {
      // Disable snap
      this.transformControls.setTranslationSnap(null);
      this.transformControls.setRotationSnap(null);
      this.transformControls.setScaleSnap(null);
    }
  }

  toggleSnapMode(): void {
    this.setSnapEnabled(!this.snapEnabled);
  }

  isSnapEnabled(): boolean {
    return this.snapEnabled;
  }

  private emitTransformChange(): void {
    if (!this.currentObject || !this.currentNodePath) return;

    const mode = this.getMode();
    let property: 'position' | 'rotation' | 'scale';
    let value: THREE.Vector3 | THREE.Euler;
    let oldValue: THREE.Vector3 | THREE.Euler;

    switch (mode) {
      case 'translate':
        property = 'position';
        value = this.currentObject.position.clone();
        oldValue = this.initialPosition.clone();
        break;
      case 'rotate':
        property = 'rotation';
        value = this.currentObject.rotation.clone();
        oldValue = this.initialRotation.clone();
        break;
      case 'scale':
        property = 'scale';
        value = this.currentObject.scale.clone();
        oldValue = this.initialScale.clone();
        break;
    }

    if (this.onChangeCallback) {
      this.onChangeCallback({
        nodePath: this.currentNodePath,
        property,
        value,
        oldValue,
      });
    }
  }

  private updateTransformDisplay(): void {
    if (!this.currentObject) return;

    // Dispatch custom event for UI update
    window.dispatchEvent(new CustomEvent('transform-update', {
      detail: {
        mode: this.getMode(),
        space: this.transformControls.space,
        position: this.currentObject.position.toArray(),
        rotation: this.currentObject.rotation.toArray().slice(0, 3),
        scale: this.currentObject.scale.toArray(),
        snap: this.snapEnabled,
      }
    }));
  }

  onChange(callback: (event: TransformChangeEvent) => void): void {
    this.onChangeCallback = callback;
  }

  dispose(): void {
    this.transformControls.dispose();
  }
}
```

### Integration with Custom Editor

```typescript
// tscnEditorProvider.ts (additions)
import { TransformGizmoManager } from './transformGizmoManager';

export class TscnEditorProvider implements vscode.CustomTextEditorProvider {
  private gizmoManager?: TransformGizmoManager;

  async resolveCustomTextEditor(
    document: vscode.TextDocument,
    webviewPanel: vscode.WebviewPanel,
    token: vscode.CancellationToken
  ): Promise<void> {
    // ... existing setup code ...

    // Setup message handling for gizmo
    webviewPanel.webview.onDidReceiveMessage(async (message) => {
      switch (message.command) {
        case 'gizmoInitialized':
          // Gizmo manager created in webview
          break;

        case 'nodeSelected':
          // User selected a node in the scene tree
          webviewPanel.webview.postMessage({
            command: 'attachGizmo',
            nodePath: message.nodePath,
          });
          break;

        case 'transformChanged':
          // Gizmo transformation completed
          await this.updateTscnTransform(document, message);
          break;

        case 'gizmoModeChanged':
          // Mode switched (translate/rotate/scale)
          this.updateGizmoUI(message.mode);
          break;
      }
    });
  }

  private async updateTscnTransform(
    document: vscode.TextDocument,
    change: any
  ): Promise<void> {
    const edit = new vscode.WorkspaceEdit();
    const text = document.getText();

    // Find node section
    const nodeRegex = new RegExp(
      `\\[node name="${change.nodePath}"[^\\]]*\\]([^\\[]*)`,
      'g'
    );

    const match = nodeRegex.exec(text);
    if (!match) return;

    // Build transform property string
    let propertyLine: string;

    switch (change.property) {
      case 'position':
        const [x, y, z] = change.value;
        propertyLine = `transform = Transform3D(1, 0, 0, 0, 1, 0, 0, 0, 1, ${x}, ${y}, ${z})`;
        break;
      case 'rotation':
        // Convert Euler to rotation matrix
        propertyLine = this.buildTransformWithRotation(change.value);
        break;
      case 'scale':
        propertyLine = this.buildTransformWithScale(change.value);
        break;
    }

    // Update document
    // ... similar to WI-30 property editing logic ...

    await vscode.workspace.applyEdit(edit);
  }

  private buildTransformWithRotation(euler: number[]): string {
    // Convert Euler angles to rotation matrix for TSCN format
    // Implementation depends on Godot's Transform3D format
    return `transform = Transform3D(...)`;
  }

  private buildTransformWithScale(scale: number[]): string {
    // Build Transform3D with scale values
    return `transform = Transform3D(${scale[0]}, 0, 0, 0, ${scale[1]}, 0, 0, 0, ${scale[2]}, 0, 0, 0)`;
  }
}
```

### Webview Integration

```typescript
// media/preview.js (additions)
let gizmoManager;

function initializeGizmo() {
  gizmoManager = new TransformGizmoManager(
    camera,
    renderer,
    scene,
    orbitControls
  );

  // Handle transform changes
  gizmoManager.onChange((event) => {
    // Send to extension
    vscode.postMessage({
      command: 'transformChanged',
      nodePath: event.nodePath,
      property: event.property,
      value: event.value.toArray ? event.value.toArray() : [event.value.x, event.value.y, event.value.z],
      oldValue: event.oldValue.toArray ? event.oldValue.toArray() : [event.oldValue.x, event.oldValue.y, event.oldValue.z],
    });
  });

  // Listen for transform UI updates
  window.addEventListener('transform-update', (event) => {
    updateTransformUI(event.detail);
  });

  vscode.postMessage({ command: 'gizmoInitialized' });
}

function updateTransformUI(detail) {
  // Update toolbar UI with current mode, space, values
  const modeDisplay = document.getElementById('current-mode');
  if (modeDisplay) {
    modeDisplay.textContent = `Mode: ${detail.mode} | Space: ${detail.space} | Snap: ${detail.snap ? 'ON' : 'OFF'}`;
  }

  // Update position/rotation/scale display
  const posDisplay = document.getElementById('position-display');
  if (posDisplay) {
    posDisplay.textContent = `Position: (${detail.position.map(v => v.toFixed(2)).join(', ')})`;
  }
}

// Handle messages from extension
window.addEventListener('message', (event) => {
  const message = event.data;

  switch (message.command) {
    case 'attachGizmo':
      const object = findObjectByPath(message.nodePath);
      if (object) {
        gizmoManager.attachToObject(object, message.nodePath);
      }
      break;

    case 'detachGizmo':
      gizmoManager.detach();
      break;

    case 'setGizmoMode':
      gizmoManager.setMode(message.mode);
      break;

    case 'toggleGizmoSnap':
      gizmoManager.toggleSnapMode();
      break;
  }
});

function findObjectByPath(path) {
  // Find three.js object by node path
  return nodeCache.get(path);
}
```
