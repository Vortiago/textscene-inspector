# WI-32: Resource Thumbnail Previews

## Overview

Implement thumbnail preview generation for TSCN resources including textures, materials, meshes, and scenes. Display these thumbnails in VS Code's file explorer, custom tree views, and resource inspector panels, providing visual feedback that helps developers quickly identify assets without opening them.

## Implementation Requirements

From TODO.md Phase 9: Enhanced Visualization:

- Generate thumbnails for texture resources (PNG, JPG, etc.)
- Render material previews showing PBR properties
- Render mesh previews as 3D thumbnails
- Generate scene file thumbnails (preview of root nodes)
- Cache thumbnails to disk for performance
- Display thumbnails in custom TreeView
- Integrate with VS Code FileDecorationProvider
- Support thumbnail sizes (small, medium, large)
- Handle thumbnail regeneration on file changes
- Show placeholder icons for unsupported resource types
- Optimize thumbnail generation performance

## Implementation Notes

### Thumbnail Generator Class

```typescript
// thumbnailGenerator.ts
import * as THREE from 'three';
import * as vscode from 'vscode';
import * as fs from 'fs';
import * as path from 'path';
import * as crypto from 'crypto';

export enum ThumbnailSize {
  Small = 64,
  Medium = 128,
  Large = 256,
}

export interface ThumbnailOptions {
  size: ThumbnailSize;
  background?: string; // Hex color
  lighting?: boolean;
}

export class ThumbnailGenerator {
  private renderer: THREE.WebGLRenderer | null = null;
  private scene: THREE.Scene;
  private camera: THREE.PerspectiveCamera;
  private cacheDir: string;

  constructor(private readonly context: vscode.ExtensionContext) {
    // Create cache directory
    this.cacheDir = path.join(context.globalStorageUri.fsPath, 'thumbnails');
    if (!fs.existsSync(this.cacheDir)) {
      fs.mkdirSync(this.cacheDir, { recursive: true });
    }

    // Setup reusable scene
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(45, 1, 0.1, 1000);
  }

  async generateMeshThumbnail(
    meshPath: string,
    options: ThumbnailOptions = { size: ThumbnailSize.Medium, lighting: true }
  ): Promise<string> {
    // Check cache first
    const cacheKey = this.getCacheKey(meshPath, options);
    const cachedPath = await this.getCachedThumbnail(cacheKey);
    if (cachedPath) return cachedPath;

    // Load mesh
    const mesh = await this.loadMesh(meshPath);
    if (!mesh) {
      return this.getPlaceholderIcon('mesh');
    }

    // Setup scene
    this.scene.clear();
    this.scene.background = new THREE.Color(
      options.background || '#1e1e1e'
    );

    // Center and scale mesh
    const box = new THREE.Box3().setFromObject(mesh);
    const center = box.getCenter(new THREE.Vector3());
    const size = box.getSize(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);

    mesh.position.sub(center);
    mesh.scale.setScalar(2 / maxDim);

    this.scene.add(mesh);

    // Setup lighting
    if (options.lighting) {
      this.setupLighting();
    }

    // Position camera
    this.camera.position.set(2, 2, 2);
    this.camera.lookAt(0, 0, 0);

    // Render thumbnail
    const dataUrl = await this.renderToDataUrl(options.size);

    // Save to cache
    const thumbnailPath = await this.saveThumbnail(cacheKey, dataUrl);

    return thumbnailPath;
  }

  async generateMaterialThumbnail(
    material: THREE.Material,
    options: ThumbnailOptions = { size: ThumbnailSize.Medium, lighting: true }
  ): Promise<string> {
    this.scene.clear();
    this.scene.background = new THREE.Color(options.background || '#1e1e1e');

    // Create sphere to show material
    const geometry = new THREE.SphereGeometry(1, 32, 32);
    const mesh = new THREE.Mesh(geometry, material);
    this.scene.add(mesh);

    // Setup lighting
    if (options.lighting) {
      this.setupLighting();
    }

    // Position camera
    this.camera.position.set(0, 0, 3);
    this.camera.lookAt(0, 0, 0);

    // Render
    const dataUrl = await this.renderToDataUrl(options.size);

    // Generate cache key from material properties
    const materialId = this.getMaterialId(material);
    const cacheKey = this.getCacheKey(`material_${materialId}`, options);

    const thumbnailPath = await this.saveThumbnail(cacheKey, dataUrl);

    return thumbnailPath;
  }

  async generateTextureThumbnail(
    texturePath: string,
    options: ThumbnailOptions = { size: ThumbnailSize.Medium, lighting: false }
  ): Promise<string> {
    // Check cache
    const cacheKey = this.getCacheKey(texturePath, options);
    const cachedPath = await this.getCachedThumbnail(cacheKey);
    if (cachedPath) return cachedPath;

    // For textures, we can use Canvas API directly
    try {
      const canvas = document.createElement('canvas');
      canvas.width = options.size;
      canvas.height = options.size;
      const ctx = canvas.getContext('2d');

      const img = new Image();
      await new Promise<void>((resolve, reject) => {
        img.onload = () => resolve();
        img.onerror = reject;
        img.src = texturePath;
      });

      // Draw image scaled to fit
      const scale = Math.min(
        options.size / img.width,
        options.size / img.height
      );
      const width = img.width * scale;
      const height = img.height * scale;
      const x = (options.size - width) / 2;
      const y = (options.size - height) / 2;

      if (ctx) {
        ctx.fillStyle = options.background || '#1e1e1e';
        ctx.fillRect(0, 0, options.size, options.size);
        ctx.drawImage(img, x, y, width, height);
      }

      const dataUrl = canvas.toDataURL('image/png');
      const thumbnailPath = await this.saveThumbnail(cacheKey, dataUrl);

      return thumbnailPath;
    } catch (error) {
      return this.getPlaceholderIcon('texture');
    }
  }

  async generateSceneThumbnail(
    scenePath: string,
    options: ThumbnailOptions = { size: ThumbnailSize.Large, lighting: true }
  ): Promise<string> {
    // Parse TSCN file and render root node tree
    const cacheKey = this.getCacheKey(scenePath, options);
    const cachedPath = await this.getCachedThumbnail(cacheKey);
    if (cachedPath) return cachedPath;

    // Load and parse scene
    const sceneData = await this.loadTscnScene(scenePath);
    if (!sceneData) {
      return this.getPlaceholderIcon('scene');
    }

    // Render scene preview
    this.scene.clear();
    this.scene.background = new THREE.Color(options.background || '#1e1e1e');

    // Add scene nodes to three.js scene
    await this.buildSceneGraph(sceneData, this.scene);

    // Setup lighting
    if (options.lighting) {
      this.setupLighting();
    }

    // Auto-frame camera
    this.frameCamera();

    // Render
    const dataUrl = await this.renderToDataUrl(options.size);
    const thumbnailPath = await this.saveThumbnail(cacheKey, dataUrl);

    return thumbnailPath;
  }

  private setupLighting(): void {
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
    const directionalLight1 = new THREE.DirectionalLight(0xffffff, 0.6);
    directionalLight1.position.set(5, 5, 5);

    const directionalLight2 = new THREE.DirectionalLight(0xffffff, 0.3);
    directionalLight2.position.set(-5, 3, -5);

    this.scene.add(ambientLight, directionalLight1, directionalLight2);
  }

  private frameCamera(): void {
    // Calculate bounding box of all objects
    const box = new THREE.Box3();
    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh) {
        box.expandByObject(object);
      }
    });

    if (box.isEmpty()) {
      this.camera.position.set(2, 2, 2);
      this.camera.lookAt(0, 0, 0);
      return;
    }

    const center = box.getCenter(new THREE.Vector3());
    const size = box.getSize(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);

    // Position camera to fit scene
    const distance = maxDim / Math.tan((this.camera.fov * Math.PI) / 360);
    const direction = new THREE.Vector3(1, 1, 1).normalize();

    this.camera.position.copy(center).add(direction.multiplyScalar(distance * 1.5));
    this.camera.lookAt(center);
  }

  private async renderToDataUrl(size: number): Promise<string> {
    if (!this.renderer) {
      const canvas = document.createElement('canvas');
      canvas.width = size;
      canvas.height = size;

      this.renderer = new THREE.WebGLRenderer({
        canvas,
        alpha: true,
        antialias: true,
        preserveDrawingBuffer: true,
      });
    }

    this.renderer.setSize(size, size);
    this.renderer.render(this.scene, this.camera);

    return this.renderer.domElement.toDataURL('image/png');
  }

  private getCacheKey(resourcePath: string, options: ThumbnailOptions): string {
    const stats = fs.statSync(resourcePath);
    const data = `${resourcePath}:${stats.mtimeMs}:${options.size}:${options.background}`;
    return crypto.createHash('md5').update(data).digest('hex');
  }

  private async getCachedThumbnail(cacheKey: string): Promise<string | null> {
    const thumbnailPath = path.join(this.cacheDir, `${cacheKey}.png`);
    if (fs.existsSync(thumbnailPath)) {
      return thumbnailPath;
    }
    return null;
  }

  private async saveThumbnail(cacheKey: string, dataUrl: string): Promise<string> {
    const thumbnailPath = path.join(this.cacheDir, `${cacheKey}.png`);

    // Convert data URL to buffer
    const base64Data = dataUrl.replace(/^data:image\/png;base64,/, '');
    const buffer = Buffer.from(base64Data, 'base64');

    fs.writeFileSync(thumbnailPath, buffer);

    return thumbnailPath;
  }

  private getPlaceholderIcon(type: string): string {
    // Return path to placeholder icon
    return path.join(this.context.extensionPath, 'media', `${type}-placeholder.png`);
  }

  private getMaterialId(material: THREE.Material): string {
    // Generate unique ID from material properties
    const props = JSON.stringify({
      type: material.type,
      color: (material as any).color?.getHex(),
      metalness: (material as any).metalness,
      roughness: (material as any).roughness,
    });
    return crypto.createHash('md5').update(props).digest('hex').substring(0, 8);
  }

  private async loadMesh(meshPath: string): Promise<THREE.Mesh | null> {

  dispose(): void {

      this.renderer.dispose();
      this.renderer = null;
    }
  }
}
```

### Resource TreeView with Thumbnails

```typescript
// resourceTreeView.ts
import * as vscode from 'vscode';
import { ThumbnailGenerator, ThumbnailSize } from './thumbnailGenerator';

class ResourceTreeItem extends vscode.TreeItem {
  constructor(
    public readonly resourceUri: vscode.Uri,
    public readonly resourceType: 'texture' | 'material' | 'mesh' | 'scene',
    public readonly label: string,
    public readonly collapsibleState: vscode.TreeItemCollapsibleState,
    private thumbnailPath?: string
  ) {
    super(label, collapsibleState);

    this.resourceUri = resourceUri;
    this.contextValue = resourceType;

    // Set icon to thumbnail if available
    if (thumbnailPath) {
      this.iconPath = vscode.Uri.file(thumbnailPath);
    } else {
      this.iconPath = new vscode.ThemeIcon(this.getThemeIcon(resourceType));
    }

    this.tooltip = new vscode.MarkdownString(`![thumbnail](${thumbnailPath})\n\n**${label}**`);
    this.command = {
      command: 'tscn.previewResource',
      title: 'Preview Resource',
      arguments: [resourceUri],
    };
  }

  private getThemeIcon(type: string): string {
    const iconMap: Record<string, string> = {
      texture: 'file-media',
      material: 'symbol-color',
      mesh: 'symbol-misc',
      scene: 'file-code',
    };
    return iconMap[type] || 'file';
  }
}

class ResourceTreeDataProvider implements vscode.TreeDataProvider<ResourceTreeItem> {
  private _onDidChangeTreeData = new vscode.EventEmitter<ResourceTreeItem | undefined | void>();
  readonly onDidChangeTreeData = this._onDidChangeTreeData.event;

  constructor(
    private readonly workspaceRoot: string,
    private readonly thumbnailGenerator: ThumbnailGenerator
  ) {}

  refresh(): void {
    this._onDidChangeTreeData.fire();
  }

  getTreeItem(element: ResourceTreeItem): vscode.TreeItem {
    return element;
  }

  async getChildren(element?: ResourceTreeItem): Promise<ResourceTreeItem[]> {
    if (!element) {
      // Root level - show resource categories
      return [
        new ResourceTreeItem(
          vscode.Uri.file(path.join(this.workspaceRoot, 'textures')),
          'texture',
          'Textures',
          vscode.TreeItemCollapsibleState.Collapsed
        ),
        new ResourceTreeItem(
          vscode.Uri.file(path.join(this.workspaceRoot, 'materials')),
          'material',
          'Materials',
          vscode.TreeItemCollapsibleState.Collapsed
        ),
        new ResourceTreeItem(
          vscode.Uri.file(path.join(this.workspaceRoot, 'meshes')),
          'mesh',
          'Meshes',
          vscode.TreeItemCollapsibleState.Collapsed
        ),
        new ResourceTreeItem(
          vscode.Uri.file(path.join(this.workspaceRoot, 'scenes')),
          'scene',
          'Scenes',
          vscode.TreeItemCollapsibleState.Collapsed
        ),
      ];
    } else {
      // Show files in category
      return this.getResourceFiles(element.resourceUri, element.resourceType);
    }
  }

  private async getResourceFiles(
    dirUri: vscode.Uri,
    resourceType: string
  ): Promise<ResourceTreeItem[]> {
    const items: ResourceTreeItem[] = [];

    try {
      const files = await vscode.workspace.fs.readDirectory(dirUri);

      for (const [name, type] of files) {
        if (type === vscode.FileType.File && this.isResourceFile(name, resourceType)) {
          const fileUri = vscode.Uri.joinPath(dirUri, name);

          // Generate thumbnail
          const thumbnailPath = await this.generateThumbnail(
            fileUri.fsPath,
            resourceType
          );

          items.push(
            new ResourceTreeItem(
              fileUri,
              resourceType as any,
              name,
              vscode.TreeItemCollapsibleState.None,
              thumbnailPath
            )
          );
        }
      }
    } catch (error) {
      // Directory doesn't exist
    }

    return items;
  }

  private isResourceFile(name: string, resourceType: string): boolean {
    const extensions: Record<string, string[]> = {
      texture: ['.png', '.jpg', '.jpeg', '.webp'],
      material: ['.material', '.tres'],
      mesh: ['.mesh', '.obj', '.gltf', '.glb'],
      scene: ['.tscn', '.scn'],
    };

    const validExtensions = extensions[resourceType] || [];
    return validExtensions.some(ext => name.endsWith(ext));
  }

  private async generateThumbnail(
    filePath: string,
    resourceType: string
  ): Promise<string> {
    try {
      switch (resourceType) {
        case 'texture':
          return await this.thumbnailGenerator.generateTextureThumbnail(filePath, {
            size: ThumbnailSize.Small,
            lighting: false,
          });
        case 'scene':
          return await this.thumbnailGenerator.generateSceneThumbnail(filePath, {
            size: ThumbnailSize.Medium,
            lighting: true,
          });
        default:
          return '';
      }
    } catch (error) {
      console.error(`Failed to generate thumbnail for ${filePath}:`, error);
      return '';
    }
  }
}

export function registerResourceTreeView(
  context: vscode.ExtensionContext,
  thumbnailGenerator: ThumbnailGenerator
): void {
  const workspaceRoot = vscode.workspace.workspaceFolders?.[0].uri.fsPath;
  if (!workspaceRoot) return;

  const treeDataProvider = new ResourceTreeDataProvider(workspaceRoot, thumbnailGenerator);

  const treeView = vscode.window.createTreeView('tscnResources', {
    treeDataProvider,
    showCollapseAll: true,
  });

  // Watch for file changes and refresh
  const fileWatcher = vscode.workspace.createFileSystemWatcher('**/*.{png,jpg,tscn}');
  fileWatcher.onDidCreate(() => treeDataProvider.refresh());
  fileWatcher.onDidChange(() => treeDataProvider.refresh());
  fileWatcher.onDidDelete(() => treeDataProvider.refresh());

  context.subscriptions.push(treeView, fileWatcher);
}
```
