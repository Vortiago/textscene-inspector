# WI-37: Node Path Copy/Generation

## Overview

Implement utilities for copying and generating node paths in various formats, helping developers quickly reference nodes in their scripts. Support multiple path formats (relative, absolute, C# GetNode format), clipboard integration, and code snippet generation.

## Implementation Requirements

From TODO.md Phase 10: Advanced Navigation:

- Copy node path to clipboard
- Generate paths in multiple formats (GDScript, C#, absolute, relative)
- Support NodePath string format
- Generate GetNode<T>() calls with correct types
- Path validation and autocomplete
- Convert between path formats
- Generate code snippets for common operations
- Support unique node names (%)
- Handle scene root references
- Quick path insertion in active editor

## Implementation Notes

### Path Format Examples

**GDScript:**
```gdscript
# Relative path
$"Player/Camera3D"
get_node("Player/Camera3D")

# Absolute path
get_node("/root/MainScene/Player")

# NodePath
var path = NodePath("Player/Camera3D")

# Unique node
%UniquePlayer
```

**C#:**
```csharp
// Relative path
GetNode<Camera3D>("Player/Camera3D")

// Absolute path
GetTree().Root.GetNode<Player>("/root/MainScene/Player")

// NodePath
var path = new NodePath("Player/Camera3D");
```

### Node Path Generator

```typescript
// nodePathGenerator.ts
import * as vscode from 'vscode';

export enum PathFormat {
  Relative = 'relative',
  Absolute = 'absolute',
  GDScriptShorthand = 'gdscript-short',
  GDScriptGetNode = 'gdscript-getnode',
  CSharpGetNode = 'csharp-getnode',
  NodePathLiteral = 'nodepath-literal',
}

export interface NodePathOptions {
  format: PathFormat;
  fromNode?: string;  // Calculate relative path from this node
  includeType?: boolean;  // For GetNode<T>
  useUniqueNames?: boolean;
}

export class NodePathGenerator {
  generatePath(
    targetNodePath: string,
    options: NodePathOptions
  ): string {
    switch (options.format) {
      case PathFormat.GDScriptShorthand:
        return this.generateGDScriptShorthand(targetNodePath);

      case PathFormat.GDScriptGetNode:
        return this.generateGDScriptGetNode(targetNodePath, options);

      case PathFormat.CSharpGetNode:
        return this.generateCSharpGetNode(targetNodePath, options);

      case PathFormat.NodePathLiteral:
        return this.generateNodePathLiteral(targetNodePath);

      case PathFormat.Absolute:
        return this.generateAbsolutePath(targetNodePath);

      case PathFormat.Relative:
      default:
        return this.generateRelativePath(targetNodePath, options.fromNode);
    }
  }

  private generateGDScriptShorthand(path: string): string {
    return `$"${path}"`;
  }

  private generateGDScriptGetNode(
    path: string,
    options: NodePathOptions
  ): string {
    return `get_node("${path}")`;
  }

  private generateCSharpGetNode(
    path: string,
    options: NodePathOptions
  ): string {
    if (options.includeType) {
      const nodeType = this.inferNodeType(path);
      return `GetNode<${nodeType}>("${path}")`;
    }
    return `GetNode("${path}")`;
  }

  private generateNodePathLiteral(path: string): string {
    return `NodePath("${path}")`;
  }

  private generateAbsolutePath(path: string): string {
    // Add /root prefix if not present
    if (!path.startsWith('/root')) {
      return `/root/${path}`;
    }
    return path;
  }

  private generateRelativePath(
    targetPath: string,
    fromPath?: string
  ): string {
    if (!fromPath) return targetPath;

    const targetParts = targetPath.split('/');
    const fromParts = fromPath.split('/');

    // Find common ancestor
    let commonLength = 0;
    while (
      commonLength < targetParts.length &&
      commonLength < fromParts.length &&
      targetParts[commonLength] === fromParts[commonLength]
    ) {
      commonLength++;
    }

    // Build relative path
    const upLevels = fromParts.length - commonLength;
    const downPath = targetParts.slice(commonLength);

    const relativeParts = [
      ...Array(upLevels).fill('..'),
      ...downPath,
    ];

    return relativeParts.join('/');
  }

  private inferNodeType(path: string): string {
    // Try to infer type from node name
    // This would ideally look up the actual node type from scene data
    const name = path.split('/').pop() || '';

    if (name.includes('Camera')) return 'Camera3D';
    if (name.includes('Light')) return 'Light3D';
    if (name.includes('Mesh')) return 'MeshInstance3D';
    if (name.includes('Player')) return 'CharacterBody3D';

    return 'Node'; // Default fallback
  }

  generateCodeSnippet(
    nodePath: string,
    operation: 'reference' | 'modify' | 'connect'
  ): string {
    switch (operation) {
      case 'reference':
        return this.generateReferenceSnippet(nodePath);
      case 'modify':
        return this.generateModifySnippet(nodePath);
      case 'connect':
        return this.generateConnectSnippet(nodePath);
      default:
        return '';
    }
  }

  private generateReferenceSnippet(path: string): string {
    const type = this.inferNodeType(path);
    const varName = this.pathToVariableName(path);

    return `var ${varName} = GetNode<${type}>("${path}");`;
  }

  private generateModifySnippet(path: string): string {
    const type = this.inferNodeType(path);

    return `var node = GetNode<${type}>("${path}");
node.Position = new Vector3(0, 0, 0);`;
  }

  private generateConnectSnippet(path: string): string {
    return `var node = GetNode("${path}");
node.Connect("signal_name", Callable.From(this.OnSignalReceived));`;
  }

  private pathToVariableName(path: string): string {
    const name = path.split('/').pop() || 'node';
    // Convert to camelCase
    return name.charAt(0).toLowerCase() + name.slice(1).replace(/[^a-zA-Z0-9]/g, '');
  }
}
```

### VS Code Integration

```typescript
// pathCommands.ts
import * as vscode from 'vscode';
import { NodePathGenerator, PathFormat, NodePathOptions } from './nodePathGenerator';

export class PathCommands {
  constructor(private generator: NodePathGenerator) {}

  registerCommands(context: vscode.ExtensionContext): void {
    // Copy path commands
    context.subscriptions.push(
      vscode.commands.registerCommand('tscn.copyNodePath', async (nodePath: string) => {
        await this.copyPath(nodePath, PathFormat.Relative);
      })
    );

    context.subscriptions.push(
      vscode.commands.registerCommand('tscn.copyNodePathAs', async (nodePath: string) => {
        await this.copyPathWithFormatSelection(nodePath);
      })
    );

    // Generate snippet commands
    context.subscriptions.push(
      vscode.commands.registerCommand('tscn.generateGetNodeSnippet', async (nodePath: string) => {
        await this.generateAndInsertSnippet(nodePath, 'reference');
      })
    );

    // Convert path format
    context.subscriptions.push(
      vscode.commands.registerCommand('tscn.convertPathFormat', async () => {
        await this.convertPathFormat();
      })
    );
  }

  private async copyPath(nodePath: string, format: PathFormat): Promise<void> {
    const options: NodePathOptions = {
      format,
      includeType: true,
    };

    const path = this.generator.generatePath(nodePath, options);
    await vscode.env.clipboard.writeText(path);

    vscode.window.setStatusBarMessage(`Copied: ${path}`, 3000);
  }

  private async copyPathWithFormatSelection(nodePath: string): Promise<void> {
    const formats = [
      { label: 'Relative Path', value: PathFormat.Relative },
      { label: 'Absolute Path', value: PathFormat.Absolute },
      { label: 'GDScript $"path"', value: PathFormat.GDScriptShorthand },
      { label: 'GDScript get_node("path")', value: PathFormat.GDScriptGetNode },
      { label: 'C# GetNode<T>("path")', value: PathFormat.CSharpGetNode },
      { label: 'NodePath("path")', value: PathFormat.NodePathLiteral },
    ];

    const selected = await vscode.window.showQuickPick(formats, {
      placeHolder: 'Select path format',
    });

    if (selected) {
      await this.copyPath(nodePath, selected.value);
    }
  }

  private async generateAndInsertSnippet(
    nodePath: string,
    operation: 'reference' | 'modify' | 'connect'
  ): Promise<void> {
    const snippet = this.generator.generateCodeSnippet(nodePath, operation);

    const editor = vscode.window.activeTextEditor;
    if (!editor) {
      vscode.window.showErrorMessage('No active editor');
      return;
    }

    await editor.insertSnippet(new vscode.SnippetString(snippet));
  }

  private async convertPathFormat(): Promise<void> {
    const editor = vscode.window.activeTextEditor;
    if (!editor) return;

    const selection = editor.selection;
    const selectedText = editor.document.getText(selection);

    // Try to extract path from selected text
    const pathMatch = /"([^"]+)"/.exec(selectedText);
    if (!pathMatch) {
      vscode.window.showErrorMessage('No valid path found in selection');
      return;
    }

    const originalPath = pathMatch[1];

    // Show format options
    const formats = [
      { label: 'C# GetNode<T>', value: PathFormat.CSharpGetNode },
      { label: 'GDScript $"path"', value: PathFormat.GDScriptShorthand },
      { label: 'GDScript get_node', value: PathFormat.GDScriptGetNode },
      { label: 'Relative', value: PathFormat.Relative },
      { label: 'Absolute', value: PathFormat.Absolute },
    ];

    const selected = await vscode.window.showQuickPick(formats, {
      placeHolder: 'Convert to format',
    });

    if (selected) {
      const converted = this.generator.generatePath(originalPath, {
        format: selected.value,
        includeType: true,
      });

      await editor.edit(editBuilder => {
        editBuilder.replace(selection, converted);
      });
    }
  }
}
```

### Code Actions Provider

```typescript
// pathCodeActionProvider.ts
export class PathCodeActionProvider implements vscode.CodeActionProvider {
  provideCodeActions(
    document: vscode.TextDocument,
    range: vscode.Range | vscode.Selection,
    context: vscode.CodeActionContext,
    token: vscode.CancellationToken
  ): vscode.CodeAction[] | undefined {
    const actions: vscode.CodeAction[] = [];

    // Check if cursor is on a node path string
    const line = document.lineAt(range.start.line);
    const pathMatch = /"([^"]+)"/.exec(line.text);

    if (pathMatch) {
      // Add "Copy Node Path" action
      const copyAction = new vscode.CodeAction(
        'Copy Node Path',
        vscode.CodeActionKind.RefactorRewrite
      );
      copyAction.command = {
        command: 'tscn.copyNodePath',
        title: 'Copy Node Path',
        arguments: [pathMatch[1]],
      };
      actions.push(copyAction);

      // Add "Convert Path Format" action
      const convertAction = new vscode.CodeAction(
        'Convert Path Format',
        vscode.CodeActionKind.RefactorRewrite
      );
      convertAction.command = {
        command: 'tscn.convertPathFormat',
        title: 'Convert Path Format',
      };
      actions.push(convertAction);
    }

    return actions;
  }
}
```
