# WI-41: Live Godot Editor Synchronization

## Overview

**Experimental Feature**: Implement bidirectional synchronization between VS Code TSCN previewer and a running Godot editor instance. Enable real-time property updates, scene changes, and runtime state inspection by connecting to Godot's remote debug protocol or implementing a custom bridge.

## Implementation Requirements

From TODO.md Phase 11: Animation and Performance:

- Connect to running Godot editor via remote debug protocol
- Bidirectional property editing (changes in VS Code → Godot and vice versa)
- Show runtime node states during game execution
- Sync camera positions between editors
- Reflect scene tree changes in real-time
- Support debugging workflows
- Handle connection lifecycle (connect, disconnect, reconnect)
- Provide connection status indicators

### Godot Debug Server Connection

```typescript
// godotDebugConnection.ts
import { WebSocket } from 'ws';

export interface GodotConnectionConfig {
  host: string;
  port: number;
  autoReconnect: boolean;
  reconnectInterval: number;
}

export interface RemoteNode {
  path: string;
  type: string;
  properties: Map<string, any>;
  runtimeState?: any;
}

export class GodotDebugConnection {
  private ws: WebSocket | null = null;
  private isConnected = false;
  private messageHandlers: Map<string, (data: any) => void> = new Map();
  private reconnectTimer: NodeJS.Timeout | null = null;

  constructor(private config: GodotConnectionConfig) {}

  async connect(): Promise<boolean> {
    try {
      const url = `ws://${this.config.host}:${this.config.port}`;
      this.ws = new WebSocket(url);

      this.ws.on('open', () => {
        this.isConnected = true;
        this.onConnected();
      });

      this.ws.on('message', (data: Buffer) => {
        this.handleMessage(data);
      });

      this.ws.on('error', (error) => {
        console.error('Godot connection error:', error);
        this.handleDisconnect();
      });

      this.ws.on('close', () => {
        this.handleDisconnect();
      });

      return true;
    } catch (error) {
      console.error('Failed to connect to Godot:', error);
      return false;
    }
  }

  disconnect(): void {
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer);
      this.reconnectTimer = null;
    }

    if (this.ws) {
      this.ws.close();
      this.ws = null;
    }

    this.isConnected = false;
  }

  private handleDisconnect(): void {
    this.isConnected = false;

    if (this.config.autoReconnect) {
      this.reconnectTimer = setTimeout(() => {
        this.connect();
      }, this.config.reconnectInterval);
    }
  }

  private onConnected(): void {
    // Send handshake
    this.send({
      type: 'handshake',
      client: 'vscode-tscn-previewer',
      version: '1.0.0',
    });

    // Request initial scene state
    this.send({
      type: 'request_scene_state',
    });
  }

  private handleMessage(data: Buffer): void {
    try {
      const message = JSON.parse(data.toString());
      const handler = this.messageHandlers.get(message.type);

      if (handler) {
        handler(message);
      }
    } catch (error) {
      console.error('Failed to parse message from Godot:', error);
    }
  }

  on(messageType: string, handler: (data: any) => void): void {
    this.messageHandlers.set(messageType, handler);
  }

  send(message: any): void {
    if (this.ws && this.isConnected) {
      this.ws.send(JSON.stringify(message));
    }
  }

  // Remote method calls
  async getSceneTree(): Promise<RemoteNode[]> {
    return new Promise((resolve) => {
      this.on('scene_tree_response', (data) => {
        resolve(data.nodes);
      });

      this.send({ type: 'get_scene_tree' });
    });
  }

  async getNodeProperty(nodePath: string, property: string): Promise<any> {
    return new Promise((resolve) => {
      this.on('property_value', (data) => {
        if (data.nodePath === nodePath && data.property === property) {
          resolve(data.value);
        }
      });

      this.send({
        type: 'get_property',
        nodePath,
        property,
      });
    });
  }

  async setNodeProperty(nodePath: string, property: string, value: any): Promise<void> {
    this.send({
      type: 'set_property',
      nodePath,
      property,
      value,
    });
  }

  subscribeToNodeUpdates(nodePath: string): void {
    this.send({
      type: 'subscribe',
      nodePath,
    });
  }

  unsubscribeFromNodeUpdates(nodePath: string): void {
    this.send({
      type: 'unsubscribe',
      nodePath,
    });
  }
}
```

### Godot Editor Plugin (GDScript)

```gdscript
# addons/vscode_sync/vscode_sync_server.gd
extends Node

var server: TCPServer
var clients: Array = []
var port: int = 6007
var is_running: bool = false

func _ready():
	start_server()

func start_server():
	server = TCPServer.new()
	var err = server.listen(port)

	if err != OK:
		push_error("Failed to start VSCode sync server on port %d" % port)
		return

	is_running = true
	print("VSCode sync server started on port %d" % port)

func _process(_delta):
	if not is_running:
		return

	# Accept new connections
	if server.is_connection_available():
		var peer = server.take_connection()
		clients.append({
			"peer": peer,
			"buffer": PackedByteArray()
		})
		print("VSCode client connected")

	# Process existing connections
	for client in clients:
		var peer: StreamPeerTCP = client.peer

		if peer.get_status() != StreamPeerTCP.STATUS_CONNECTED:
			clients.erase(client)
			continue

		# Read available data
		var available = peer.get_available_bytes()
		if available > 0:
			var data = peer.get_data(available)
			if data[0] == OK:
				handle_message(client, data[1])

func handle_message(client, data: PackedByteArray):
	var message_str = data.get_string_from_utf8()
	var json = JSON.new()
	var parse_result = json.parse(message_str)

	if parse_result != OK:
		push_error("Failed to parse message from VSCode")
		return

	var message = json.data

	match message.type:
		"handshake":
			send_response(client, {
				"type": "handshake_response",
				"status": "ok"
			})

		"get_scene_tree":
			var tree_data = get_scene_tree_data()
			send_response(client, {
				"type": "scene_tree_response",
				"nodes": tree_data
			})

		"get_property":
			var node = get_node_or_null(message.nodePath)
			if node:
				var value = node.get(message.property)
				send_response(client, {
					"type": "property_value",
					"nodePath": message.nodePath,
					"property": message.property,
					"value": var_to_str(value)
				})

		"set_property":
			var node = get_node_or_null(message.nodePath)
			if node:
				var value = str_to_var(message.value)
				node.set(message.property, value)

				# Notify all clients of the change
				broadcast({
					"type": "property_changed",
					"nodePath": message.nodePath,
					"property": message.property,
					"value": message.value
				})

		"subscribe":
			# Add node to watch list
			pass

		"unsubscribe":
			# Remove node from watch list
			pass

func get_scene_tree_data() -> Array:
	var root = get_tree().root
	var nodes = []

	_collect_node_data(root, nodes)

	return nodes

func _collect_node_data(node: Node, nodes: Array):
	nodes.append({
		"path": str(node.get_path()),
		"type": node.get_class(),
		"name": node.name
	})

	for child in node.get_children():
		_collect_node_data(child, nodes)

func send_response(client, data: Dictionary):
	var json = JSON.stringify(data)
	var bytes = json.to_utf8_buffer()
	client.peer.put_data(bytes)

func broadcast(data: Dictionary):
	for client in clients:
		send_response(client, data)

func _exit_tree():
	for client in clients:
		client.peer.disconnect_from_host()

	if server:
		server.stop()

	is_running = false
```

### VS Code Integration

```typescript
// liveSyncManager.ts
import * as vscode from 'vscode';
import { GodotDebugConnection } from './godotDebugConnection';

export class LiveSyncManager {
  private connection: GodotDebugConnection | null = null;
  private statusBarItem: vscode.StatusBarItem;
  private syncEnabled = false;

  constructor(private context: vscode.ExtensionContext) {
    this.statusBarItem = vscode.window.createStatusBarItem(
      vscode.StatusBarAlignment.Right,
      100
    );
    this.statusBarItem.command = 'tscn.toggleLiveSync';
    this.updateStatusBar();
    this.statusBarItem.show();

    context.subscriptions.push(this.statusBarItem);
  }

  registerCommands(): void {
    this.context.subscriptions.push(
      vscode.commands.registerCommand('tscn.connectToGodot', async () => {
        await this.connectToGodot();
      })
    );

    this.context.subscriptions.push(
      vscode.commands.registerCommand('tscn.disconnectFromGodot', () => {
        this.disconnect();
      })
    );

    this.context.subscriptions.push(
      vscode.commands.registerCommand('tscn.toggleLiveSync', async () => {
        await this.toggleSync();
      })
    );
  }

  private async connectToGodot(): Promise<void> {
    const host = await vscode.window.showInputBox({
      prompt: 'Enter Godot editor host',
      value: 'localhost',
    });

    if (!host) return;

    const portStr = await vscode.window.showInputBox({
      prompt: 'Enter Godot editor port',
      value: '6007',
    });

    if (!portStr) return;

    const port = parseInt(portStr);

    this.connection = new GodotDebugConnection({
      host,
      port,
      autoReconnect: true,
      reconnectInterval: 5000,
    });

    const success = await this.connection.connect();

    if (success) {
      vscode.window.showInformationMessage('Connected to Godot editor');
      this.setupEventHandlers();
      this.syncEnabled = true;
      this.updateStatusBar();
    } else {
      vscode.window.showErrorMessage('Failed to connect to Godot editor');
    }
  }

  private disconnect(): void {
    if (this.connection) {
      this.connection.disconnect();
      this.connection = null;
      this.syncEnabled = false;
      this.updateStatusBar();
      vscode.window.showInformationMessage('Disconnected from Godot editor');
    }
  }

  private async toggleSync(): Promise<void> {
    if (this.syncEnabled) {
      this.disconnect();
    } else {
      await this.connectToGodot();
    }
  }

  private setupEventHandlers(): void {
    if (!this.connection) return;

    // Handle property changes from Godot
    this.connection.on('property_changed', (data) => {
      this.handleRemotePropertyChange(data);
    });

    // Handle scene tree changes
    this.connection.on('scene_tree_changed', (data) => {
      this.handleRemoteSceneTreeChange(data);
    });
  }

  private handleRemotePropertyChange(data: any): void {
    // Update local TSCN file or preview
    vscode.window.showInformationMessage(
      `Property changed in Godot: ${data.nodePath}.${data.property} = ${data.value}`
    );

    // Optionally update the .tscn file
    // this.updateTscnFile(data.nodePath, data.property, data.value);
  }

  private handleRemoteSceneTreeChange(data: any): void {
    // Refresh scene tree view
    vscode.commands.executeCommand('tscn.refreshSceneTree');
  }

  async syncPropertyToGodot(
    nodePath: string,
    property: string,
    value: any
  ): Promise<void> {
    if (this.connection && this.syncEnabled) {
      await this.connection.setNodeProperty(nodePath, property, value);
    }
  }

  async getRemoteProperty(nodePath: string, property: string): Promise<any> {
    if (this.connection && this.syncEnabled) {
      return await this.connection.getNodeProperty(nodePath, property);
    }
    return null;
  }

  private updateStatusBar(): void {
    if (this.syncEnabled && this.connection) {
      this.statusBarItem.text = '$(link) Godot Connected';
      this.statusBarItem.tooltip = 'Click to disconnect from Godot editor';
      this.statusBarItem.backgroundColor = new vscode.ThemeColor(
        'statusBarItem.prominentBackground'
      );
    } else {
      this.statusBarItem.text = '$(debug-disconnect) Godot';
      this.statusBarItem.tooltip = 'Click to connect to Godot editor';
      this.statusBarItem.backgroundColor = undefined;
    }
  }

  isConnected(): boolean {
    return this.syncEnabled && this.connection !== null;
  }

  dispose(): void {
    this.disconnect();
    this.statusBarItem.dispose();
  }
}
```

## Implementation Challenges

### 1. Protocol Research
- Reverse-engineer Godot's remote debug protocol
- Or implement custom Godot editor plugin for communication
- Handle version compatibility across Godot 3.x and 4.x

### 2. State Synchronization
- Reconcile runtime state with file-based .tscn representation
- Handle conflicts when both editors modify same properties
- Implement operational transformation or CRDT for conflict resolution

### 3. Performance
- Minimize network overhead for frequent updates
- Debounce rapid changes
- Use delta encoding for property updates

### 4. Security
- Validate all remote commands
- Implement authentication/authorization
- Sandbox remote operations
