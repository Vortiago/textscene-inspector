# WI-78.5: Reconciliation Engine

**Parent**: WI-78 (Immutable Scene Hierarchies)
**Status**: Not Started
**Complexity**: Moderate
**Dependencies**: WI-78.1 (SceneGraph), WI-77.2 (PropertyDiffer)

## Goal

Diff two SceneGraphs and generate minimal patch list (React reconciliation algorithm).

## Scale Requirements

**Real-world Godot scene sizes (from research):**

| Scene Type | Node Count | Example |
|------------|------------|---------|
| Simple prop | 5-20 | Chair, lamp |
| Complex prop | 20-100 | Vehicle |
| Room/area | 100-500 | Game level area |
| Game level | 500-2,000 | Complete level |
| Complex scene | 2,000-5,000 | Open world section |
| Animation FSM | 900+ | Causes editor lag |
| Large game | 5,000-10,000+ | Full game |

**Performance Targets:**

| Operation | 100 nodes | 1,000 nodes | 5,000 nodes |
|-----------|-----------|-------------|-------------|
| Build node map | < 1ms | < 10ms | < 50ms |
| Diff (no changes) | < 1ms | < 5ms | < 25ms |
| Diff (10 changes) | < 2ms | < 10ms | < 50ms |
| Diff (100 changes) | < 5ms | < 20ms | < 100ms |

**Why O(n) matters:**
- Naive tree diff: O(n³) - 1,000 nodes = 1,000,000,000 comparisons
- React-style: O(n) - 1,000 nodes = 1,000 comparisons
- This is why React can handle thousands of components smoothly

## Algorithm

**React's Reconciliation:**
1. Build node maps (key → node) for old and new
2. Find added nodes (in new, not in old)
3. Find removed nodes (in old, not in new)
4. Find updated nodes (in both, check if changed)
5. For updated: Determine minimal update type
6. Return patch list

**Our adaptation:**
- Key = node path (e.g., "Main/Hallway/Door")
- Node = SceneNode (from flattened array)
- Update types: 'add', 'remove', 'update-transform', 'update-material', 'update-visibility'

## Implementation

**File**: `packages/textscene-core/src/core/ReconciliationEngine.ts`

```typescript
import type { SceneGraph, SceneNode } from './SceneGraph.js';
import { PropertyDiffer, type PropertyChangeset } from '../utils/PropertyDiffer.js';
import { logger } from '../utils/logger.js';

/**
 * Patch operations for updating THREE.js scene.
 */
export type PatchType =
  | 'add'                // Node added (create THREE.js object)
  | 'remove'             // Node removed (destroy THREE.js object)
  | 'update-transform'   // Transform changed (update position/rotation/scale)
  | 'update-material'    // Material changed (swap material)
  | 'update-visibility'  // Visibility changed (show/hide)
  | 'update-full';       // Structural change (destroy + recreate)

/**
 * Patch operation.
 */
export interface Patch {
  type: PatchType;
  nodePath: string;
  node?: SceneNode;      // For 'add' and 'update-full'
  data?: unknown;        // Type-specific data
}

/**
 * React-style reconciliation engine.
 * Diffs two SceneGraphs and generates minimal patch list.
 */
export class ReconciliationEngine {
  private differ = new PropertyDiffer();

  /**
   * Diff two scene graphs and return minimal patches.
   */
  diff(oldGraph: SceneGraph, newGraph: SceneGraph): Patch[] {
    logger.info(`[Reconciliation] Diffing v${oldGraph.version} → v${newGraph.version}`);

    const patches: Patch[] = [];

    // Build node maps for O(1) lookup
    const oldNodes = this.buildNodeMap(oldGraph.flattenedNodes);
    const newNodes = this.buildNodeMap(newGraph.flattenedNodes);

    // 1. Find added nodes (in new, not in old)
    for (const [path, node] of newNodes) {
      if (!oldNodes.has(path)) {
        patches.push({ type: 'add', nodePath: path, node });
      }
    }

    // 2. Find removed nodes (in old, not in new)
    for (const [path] of oldNodes) {
      if (!newNodes.has(path)) {
        patches.push({ type: 'remove', nodePath: path });
      }
    }

    // 3. Find updated nodes (in both)
    for (const [path, newNode] of newNodes) {
      const oldNode = oldNodes.get(path);
      if (oldNode) {
        const updatePatch = this.diffNode(oldNode, newNode);
        if (updatePatch) {
          patches.push(updatePatch);
        }
      }
    }

    logger.info(`[Reconciliation] Generated ${patches.length} patches`);
    this.logPatchSummary(patches);

    return patches;
  }

  /**
   * Diff single node and determine minimal update type.
   */
  private diffNode(oldNode: SceneNode, newNode: SceneNode): Patch | null {
    // Type changed? → full rebuild
    if (oldNode.data.type !== newNode.data.type) {
      return {
        type: 'update-full',
        nodePath: newNode.path,
        node: newNode
      };
    }

    // Check what changed
    const changeset = this.differ.diff(oldNode.data, newNode.data);

    if (changeset.structuralChange) {
      // Children added/removed → full rebuild
      return {
        type: 'update-full',
        nodePath: newNode.path,
        node: newNode
      };
    }

    // Determine minimal update type
    if (changeset.visibilityChanged) {
      return {
        type: 'update-visibility',
        nodePath: newNode.path,
        data: { visible: newNode.data.properties.visible !== 'false' }
      };
    }

    if (changeset.transformChanged) {
      return {
        type: 'update-transform',
        nodePath: newNode.path,
        data: { transform: newNode.data.properties.transform }
      };
    }

    if (changeset.materialChanged) {
      return {
        type: 'update-material',
        nodePath: newNode.path,
        data: { materialOverride: newNode.data.properties.material_override }
      };
    }

    // Other properties changed → full rebuild (fallback)
    if (changeset.propertiesChanged.size > 0) {
      return {
        type: 'update-full',
        nodePath: newNode.path,
        node: newNode
      };
    }

    // No changes
    return null;
  }

  /**
   * Build map: node path → node (for O(1) lookup).
   */
  private buildNodeMap(nodes: readonly SceneNode[]): Map<string, SceneNode> {
    const map = new Map<string, SceneNode>();
    for (const node of nodes) {
      map.set(node.path, node);
    }
    return map;
  }

  /**
   * Log patch summary (for debugging).
   */
  private logPatchSummary(patches: Patch[]): void {
    const summary = patches.reduce((acc, patch) => {
      acc[patch.type] = (acc[patch.type] || 0) + 1;
      return acc;
    }, {} as Record<PatchType, number>);

    logger.info('[Reconciliation] Patch summary:', summary);
  }
}
```

      externalResources: []

describe('ReconciliationEngine Performance', () => {
  /**
   * Generate scene with N nodes for performance testing.
   */
  function generateScene(nodeCount: number): ParsedScene {
    const nodes = [];
    for (let i = 0; i < nodeCount; i++) {
      nodes.push({
        name: `Node_${i}`,
        type: 'Node3D',
        children: [],
        properties: { transform: `Transform3D(1,0,0,0,1,0,0,0,1,${i},0,0)` }
      });
    }
    return {
      path: 'res://perf-test.tscn',
      nodes,
      externalScenes: [],
      internalResources: [],
      externalResources: []
    };
  }

  it('diffs 1000-node scene (no changes) in < 10ms', () => {
    const scene = generateScene(1000);
    const graph = new SceneGraphBuilder()
      .setRootScene('res://perf-test.tscn')
      .addScene(scene)
      .build();

    const start = performance.now();
    const patches = engine.diff(graph, graph);
    const duration = performance.now() - start;

    expect(patches.length).toBe(0);
    expect(duration).toBeLessThan(10);
    console.log(`1000-node diff (no changes): ${duration.toFixed(2)}ms`);
  });

  it('diffs 1000-node scene (10 changes) in < 15ms', () => {
    const scene1 = generateScene(1000);
    const scene2 = generateScene(1000);

    // Modify 10 nodes
    for (let i = 0; i < 10; i++) {
      scene2.nodes[i * 100].properties.transform = 'Transform3D(1,0,0,0,1,0,0,0,1,999,999,999)';
    }

    const graph1 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene1).build();
    const graph2 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene2).build();

    const start = performance.now();
    const patches = engine.diff(graph1, graph2);
    const duration = performance.now() - start;

    expect(patches.length).toBe(10);
    expect(duration).toBeLessThan(15);
    console.log(`1000-node diff (10 changes): ${duration.toFixed(2)}ms, ${patches.length} patches`);
  });

  it('diffs 5000-node scene in < 100ms', () => {
    const scene = generateScene(5000);
    const graph = new SceneGraphBuilder()
      .setRootScene('res://perf-test.tscn')
      .addScene(scene)
      .build();

    const start = performance.now();
    const patches = engine.diff(graph, graph);
    const duration = performance.now() - start;

    expect(patches.length).toBe(0);
    expect(duration).toBeLessThan(100);
    console.log(`5000-node diff: ${duration.toFixed(2)}ms`);
  });

  it('handles 100 adds efficiently', () => {
    const scene1 = generateScene(900);
    const scene2 = generateScene(1000);  // 100 more nodes

    const graph1 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene1).build();
    const graph2 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene2).build();

    const start = performance.now();
    const patches = engine.diff(graph1, graph2);
    const duration = performance.now() - start;

    expect(patches.filter(p => p.type === 'add').length).toBe(100);
    expect(duration).toBeLessThan(20);
    console.log(`100 adds: ${duration.toFixed(2)}ms`);
  });

  it('handles 100 removes efficiently', () => {
    const scene1 = generateScene(1000);
    const scene2 = generateScene(900);  // 100 fewer nodes

    const graph1 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene1).build();
    const graph2 = new SceneGraphBuilder().setRootScene('res://perf-test.tscn').addScene(scene2).build();

    const start = performance.now();
    const patches = engine.diff(graph1, graph2);
    const duration = performance.now() - start;

    expect(patches.filter(p => p.type === 'remove').length).toBe(100);
    expect(duration).toBeLessThan(20);
    console.log(`100 removes: ${duration.toFixed(2)}ms`);
  });
});
