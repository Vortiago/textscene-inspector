# WI-26: GetNode() Path Validation

## Overview

Implement VS Code diagnostics for validating Godot GetNode() calls in C# scripts against the actual scene tree structure. This feature analyzes C# code to detect GetNode<T>() calls, validates that the node paths exist in the associated .tscn file, and provides real-time error/warning feedback to prevent runtime errors caused by invalid node paths.

## Implementation Requirements

From TODO.md Phase 7: C# Integration:

- Parse C# files to find GetNode() and GetNode<T>() calls
- Extract node paths from GetNode() arguments
- Load associated .tscn file for the C# script
- Validate node paths against scene tree structure
- Create VS Code diagnostics for invalid paths
- Support absolute and relative node paths
- Handle string literals and string interpolation
- Provide clear error messages with suggestions
- Support % (unique name) and $ (absolute path) prefixes
- Update diagnostics on file changes (C# or TSCN)

## Godot GetNode() Pattern

### Node Path Syntax

Godot uses a hierarchical path system for node access:

**Absolute Paths (from scene root):**
```csharp
GetNode<Label>("/root/Main/UI/Label");
GetNode<Sprite2D>("$/Player/Sprite");  // $ prefix = absolute from scene root
```

**Relative Paths (from current node):**
```csharp
GetNode<Label>("UI/Label");           // Child of current node
GetNode<Button>("../Button");         // Sibling
GetNode<Timer>("../../Timer");        // Parent's sibling
GetNode<Node2D>("./Child");           // Explicit relative
```

**Unique Name Paths:**
```csharp
GetNode<Player>("%Player");           // % prefix = unique node name (searches entire tree)
```

**Common Patterns:**

```csharp
public partial class Main : Node
{
    private Label label;
    private Button button;

    public override void _Ready()
    {
        // Direct child access
        label = GetNode<Label>("UI/Label");

        // Absolute path
        button = GetNode<Button>("/root/Main/UI/Button");

        // Unique name
        var player = GetNode<Player>("%Player");

        // Parent access
        var sibling = GetNode<Node>("../Sibling");
    }
}
```

### Path Validation

**Validation Rules:**

1. **Absolute Paths**: Start from scene root
2. **Relative Paths**: Start from script's attached node
3. **Unique Names**: Search entire tree for @export var with unique name
4. **Type Checking**: Verify node type matches generic parameter (WI-28)

**Path Resolution:**

```typescript
interface NodePathValidator {
  validatePath(path: string, fromNode: string): ValidationResult;
}

interface ValidationResult {
  valid: boolean;
  targetNode?: TscnNode;
  error?: string;
  suggestions?: string[];
}

function validateNodePath(
  path: string,
  sceneTree: TscnNode,
  currentNode: TscnNode
): ValidationResult {
  // Remove leading $ or %
  let cleanPath = path;
  let isAbsolute = false;
  let isUnique = false;

  if (path.startsWith('$') || path.startsWith('/root')) {
    isAbsolute = true;
    cleanPath = path.replace(/^\$/, '').replace(/^\/root/, '');
  } else if (path.startsWith('%')) {
    isUnique = true;
    cleanPath = path.substring(1);
  }

  if (isUnique) {
    return validateUniqueName(cleanPath, sceneTree);
  }

  if (isAbsolute) {
    return validateAbsolutePath(cleanPath, sceneTree);
  }

  return validateRelativePath(cleanPath, currentNode);
}
```

### Scene-Script Association

**Finding Associated Scene:**

1. Check C# file name matches .tscn file name
2. Parse .tscn file for `script` property pointing to the C# file
3. User configuration (workspace settings)

```typescript
async function findSceneForScript(
  scriptUri: vscode.Uri
): Promise<vscode.Uri | undefined> {
  const scriptPath = scriptUri.fsPath;
  const scriptName = path.basename(scriptPath, '.cs');

  // Strategy 1: Same-named .tscn file
  const sameName = scriptPath.replace('.cs', '.tscn');
  if (await fs.pathExists(sameName)) {
    return vscode.Uri.file(sameName);
  }

  // Strategy 2: Search workspace for .tscn files with matching script
  const tscnFiles = await vscode.workspace.findFiles('**/*.tscn');

  for (const tscnUri of tscnFiles) {
    const content = await fs.readFile(tscnUri.fsPath, 'utf-8');
    if (content.includes(`script = ExtResource`) &&
        content.includes(path.basename(scriptPath))) {
      return tscnUri;
    }
  }

  return undefined;
}
```

```typescript
  type: string;
  path: string;
  range: vscode.Range;
}

export class GetNodeValidator {
  private diagnosticCollection: vscode.DiagnosticCollection;
  private sceneCache: Map<string, any> = new Map();

  constructor(diagnosticCollection: vscode.DiagnosticCollection) {
    this.diagnosticCollection = diagnosticCollection;
  }

  async validateDocument(document: vscode.TextDocument): Promise<void> {
    if (!document.fileName.endsWith('.cs')) {
      return;
    }

    // Find associated scene file
    const sceneUri = await this.findSceneForScript(document.uri);
    if (!sceneUri) {
      // No scene file found, clear diagnostics
      this.diagnosticCollection.delete(document.uri);
      return;
    }

    // Parse scene
    const scene = await this.loadScene(sceneUri);
    if (!scene) {
      return;
    }

    // Extract GetNode calls from C# code
    const getNodeCalls = this.extractGetNodeCalls(document);

    // Validate each call
    const diagnostics: vscode.Diagnostic[] = [];

    for (const call of getNodeCalls) {
      const result = this.validatePath(call.path, scene);

      if (!result.valid) {
        const diagnostic = new vscode.Diagnostic(
          call.range,
          result.error || `Node path '${call.path}' not found in scene`,
          vscode.DiagnosticSeverity.Error
        );

        diagnostic.source = 'TSCN Validator';
        diagnostic.code = 'invalid-node-path';

        // Add related information (scene file location)
        diagnostic.relatedInformation = [
          new vscode.DiagnosticRelatedInformation(
            new vscode.Location(
              sceneUri,
              new vscode.Range(0, 0, 0, 0)
            ),
            `Scene file: ${path.basename(sceneUri.fsPath)}`
          ),
        ];

        // Add suggestions if available
        if (result.suggestions && result.suggestions.length > 0) {
          diagnostic.message += `\n\nDid you mean: ${result.suggestions.join(', ')}?`;
        }

        diagnostics.push(diagnostic);

    this.diagnosticCollection.set(document.uri, diagnostics);
```

### Path Validation Logic

```typescript
// pathValidator.ts
import type { TscnNode } from '@tscn-renderer/core';

export function validateNodePath(
  path: string,
  sceneRoot: TscnNode,
  currentNode?: TscnNode
): {
  valid: boolean;
  targetNode?: TscnNode;
  error?: string;
} {
  // Handle unique name (@export var)
  if (path.startsWith('%')) {
    const uniqueName = path.substring(1);
    const node = findNodeByUniqueName(uniqueName, sceneRoot);

    if (!node) {
      return {
        valid: false,
        error: `No node with unique name '${uniqueName}' found`,
      };
    }

    return { valid: true, targetNode: node };
  }

  // Handle absolute paths
  if (path.startsWith('$') || path.startsWith('/root')) {
    const cleanPath = path.replace(/^\$/, '').replace(/^\/root\//, '');
    const node = resolveAbsolutePath(cleanPath, sceneRoot);

    if (!node) {
      return {
        valid: false,
        error: `Absolute path '${path}' not found`,
      };
    }

    return { valid: true, targetNode: node };
  }

  // Handle relative paths
  if (!currentNode) {
    return {
      valid: false,
      error: 'Cannot resolve relative path without current node context',
    };
  }

  const node = resolveRelativePath(path, currentNode);

  if (!node) {
    return {
      valid: false,
      error: `Relative path '${path}' not found from current node`,
    };
  }

  return { valid: true, targetNode: node };
}

function resolveAbsolutePath(path: string, root: TscnNode): TscnNode | null {
  const parts = path.split('/').filter(p => p.length > 0);
  let current = root;

  for (const part of parts) {
    const child = current.children?.find(c => c.name === part);
    if (!child) {
      return null;
    }
    current = child;
  }

  return current;
}

function resolveRelativePath(path: string, from: TscnNode): TscnNode | null {
  const parts = path.split('/');
  let current = from;

  for (const part of parts) {
    if (part === '..') {
      current = current.parent || current;
    } else if (part === '.' || part === '') {
      // Stay at current
    } else {
      const child = current.children?.find(c => c.name === part);
      if (!child) {
        return null;
      }
      current = child;
    }
  }

  return current;
}

function findNodeByUniqueName(name: string, root: TscnNode): TscnNode | null {
  // Search entire tree for node with unique name
  if (root.uniqueName === name) {
    return root;
  }

  for (const child of root.children || []) {
    const found = findNodeByUniqueName(name, child);
    if (found) {
      return found;
    }
  }

  return null;
}
```
