**Status**: Not Started

## Goal

Track which scenes reference which sub-scenes (bidirectional graph).

## Use Cases

```
Scenario: User edits door.tscn
Question: Which scenes use door.tscn?
Answer: dependencyGraph.getDependentScenes('res://door.tscn')
        → ['res://hallway.tscn', 'res://main.tscn']
```

This enables targeted invalidation (only update affected scenes).

## Implementation

**File**: `packages/textscene-core/src/core/DependencyGraph.ts`

```typescript
/**
 * Immutable bidirectional dependency graph.
 * Tracks which parent scenes reference which sub-scenes.
 */
export class DependencyGraph {
  // Forward: parent → sub-scenes it uses
  readonly dependencies: ReadonlyMap<string, ReadonlySet<string>>;

  // Reverse: sub-scene → parents using it
  readonly dependents: ReadonlyMap<string, ReadonlySet<string>>;

  constructor(
    dependencies: Map<string, Set<string>> = new Map(),
    dependents: Map<string, Set<string>> = new Map()
  ) {
    // Deep freeze
    this.dependencies = new Map(
      Array.from(dependencies.entries()).map(([k, v]) => [k, new Set(v)])
    );
    this.dependents = new Map(
      Array.from(dependents.entries()).map(([k, v]) => [k, new Set(v)])
    );
  }

  /**
   * Get parent scenes that use a sub-scene.
   */
  getDependentScenes(subScene: string): ReadonlySet<string> {
    return this.dependents.get(subScene) || new Set();
  }

  /**
   * Get sub-scenes used by a parent.
   */
  getSubScenes(parent: string): ReadonlySet<string> {
    return this.dependencies.get(parent) || new Set();
  }

  /**
   * Check if scene uses a sub-scene (direct or transitive).
   */
  uses(parent: string, subScene: string): boolean {
    const visited = new Set<string>();
    return this.usesRecursive(parent, subScene, visited);
  }

  private usesRecursive(parent: string, target: string, visited: Set<string>): boolean {
    if (visited.has(parent)) return false;
    visited.add(parent);

    const subs = this.dependencies.get(parent);
    if (!subs) return false;

    if (subs.has(target)) return true;

    for (const sub of subs) {
      if (this.usesRecursive(sub, target, visited)) return true;
    }

    return false;
  }
}

/**
 * Builder for DependencyGraph.
 */
export class DependencyGraphBuilder {
  private dependencies = new Map<string, Set<string>>();
  private dependents = new Map<string, Set<string>>();

  /**
   * Record dependency: parent uses subScene.
   */
  addDependency(parent: string, subScene: string): this {
    // Forward
    if (!this.dependencies.has(parent)) {
      this.dependencies.set(parent, new Set());
    }
    this.dependencies.get(parent)!.add(subScene);

    // Reverse
    if (!this.dependents.has(subScene)) {
      this.dependents.set(subScene, new Set());
    }
    this.dependents.get(subScene)!.add(parent);

    return this;
  }

  /**
   * Build immutable DependencyGraph.
   */
  build(): DependencyGraph {
    return new DependencyGraph(this.dependencies, this.dependents);
  }
}
```

```typescript
// In SceneResolver.resolve()
async resolve(rootScenePath: string): Promise<SceneGraph> {
  // ... load scenes ...

  // Build dependency graph
  const depBuilder = new DependencyGraphBuilder();
  for (const scene of scenes.values()) {
    for (const ref of scene.externalScenes) {
      depBuilder.addDependency(scene.path, ref.scenePath);
    }
  }
  const dependencies = depBuilder.build();

  // Build SceneGraph
  const builder = new SceneGraphBuilder()
    .setRootScene(rootScenePath)
    .setDependencies(dependencies);  // Add dependencies

  for (const scene of scenes.values()) {
    builder.addScene(scene);
  }

  return builder.build();
}
```

## Testing

**File**: `packages/textscene-core/src/core/DependencyGraph.test.ts`

```typescript
import { describe, it, expect } from 'vitest';
import { DependencyGraphBuilder } from './DependencyGraph.js';

describe('DependencyGraph', () => {
  it('tracks dependencies bidirectionally', () => {
    const graph = new DependencyGraphBuilder()
      .addDependency('res://main.tscn', 'res://hallway.tscn')
      .addDependency('res://hallway.tscn', 'res://door.tscn')
      .build();

    // Forward
    expect(graph.getSubScenes('res://main.tscn')).toContain('res://hallway.tscn');
    expect(graph.getSubScenes('res://hallway.tscn')).toContain('res://door.tscn');

    // Reverse
    expect(graph.getDependentScenes('res://hallway.tscn')).toContain('res://main.tscn');
    expect(graph.getDependentScenes('res://door.tscn')).toContain('res://hallway.tscn');
  });

  it('detects transitive dependencies', () => {
    const graph = new DependencyGraphBuilder()
      .addDependency('res://main.tscn', 'res://hallway.tscn')
      .addDependency('res://hallway.tscn', 'res://door.tscn')
      .build();

    expect(graph.uses('res://main.tscn', 'res://door.tscn')).toBe(true);
    expect(graph.uses('res://door.tscn', 'res://main.tscn')).toBe(false);
  });

  it('handles circular dependencies', () => {
    const graph = new DependencyGraphBuilder()
      .addDependency('res://a.tscn', 'res://b.tscn')
      .addDependency('res://b.tscn', 'res://a.tscn')
      .build();

    // Detects circular (visited check prevents infinite loop)
    expect(graph.uses('res://a.tscn', 'res://b.tscn')).toBe(true);
    expect(graph.uses('res://b.tscn', 'res://a.tscn')).toBe(true);
  });
});
```
