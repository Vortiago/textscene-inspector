# WI-56: Material Override Support

**Status**: Not Started
**Priority**: Critical
**Estimated Effort**: Moderate Complexity
**Impact**: 207 instances in ld-58 (critical for proper material rendering)

## Overview

### Current State

MeshInstance3D nodes support mesh rendering with materials, but **material override properties are not implemented**:
- ✅ Mesh sub-resources with embedded materials render correctly
- ❌ `material_override` property is ignored (applies to all surfaces)
- ❌ `surface_material_override/0`, `/1`, etc. are ignored (per-surface overrides)
- ❌ Materials default to mesh's embedded material only

### What's Missing

Material override properties allow scene designers to replace mesh materials without modifying the original mesh resource. This is essential for:
- **Reusable meshes with different materials** (e.g., same door mesh with wood/metal materials)
- **Per-surface material control** (e.g., multi-material objects like doors with handle and frame)
- **Material variants** (e.g., red vs. blue versions of the same object)

### Evidence from ld-58

**Example 1: Door.tscn (line 66)** - Surface material override on instanced scene:
```tscn
[node name="doormesh" parent="." instance=ExtResource("1_23sbp")]

[node name="doormesh" parent="doormesh" index="0"]
surface_material_override/0 = ExtResource("2_a1o0s")
```

**Example 2: WallSection.tscn (line 61)** - Material override on inline mesh:
```tscn
[node name="ChairRail" type="MeshInstance3D" parent="."]
transform = Transform3D(1, 0, 0, 0, 1, 0, 0, 0, 1, 0.015, 0.9, 0)
material_override = SubResource("StandardMaterial3D_yrd62")
```

## Implementation Requirements

### 1. Parser Changes (MeshInstance3D)

**File**: `packages/textscene-core/src/nodes/3d/meshinstance3d/parser.ts`

Add material override properties to the parser:

```typescript
export interface MeshInstance3DProperties extends Node3DProperties {
  mesh?: string; // Existing
  skeleton?: NodePath; // Existing
  material_override?: string; // NEW: Applies to all surfaces
  surface_material_override?: Record<number, string>; // NEW: Per-surface overrides
}

export function parseMeshInstance3D(section: Section): MeshInstance3DProperties {
  const baseProps = parseNode3DSection(section);
  const props: MeshInstance3DProperties = { ...baseProps };

  props.mesh = extractProperty(section, 'mesh');
  props.skeleton = extractProperty(section, 'skeleton');
  props.material_override = extractProperty(section, 'material_override'); // NEW

  // Parse indexed surface_material_override properties
  const surfaceOverrides: Record<number, string> = {};
  for (const [key, value] of Object.entries(section.properties)) {
    const match = key.match(/^surface_material_override\/(\d+)$/);
    if (match) {
      const surfaceIndex = parseInt(match[1], 10);
      surfaceOverrides[surfaceIndex] = value;
    }
  }
  if (Object.keys(surfaceOverrides).length > 0) {
    props.surface_material_override = surfaceOverrides;
  }

  return props;
}
```

### 2. Renderer Changes (MeshInstance3D)

**File**: `packages/textscene-core/src/nodes/3d/meshinstance3d/renderer.ts`

Apply material overrides after mesh loading:

```typescript
export async function createMeshInstance3D(
  props: MeshInstance3DProperties,
  context: RenderContext
): Promise<THREE.Object3D> {
  const object = createNode3D(props, context);

  if (props.mesh) {
    try {
      const meshResult = await resolveResource<THREE.Mesh>(
        props.mesh,
        context,
        MeshRegistry
      );

      if (meshResult) {
        object.add(meshResult);

        // Apply material overrides AFTER mesh is loaded
        await applyMaterialOverrides(meshResult, props, context);
      }
    } catch (error) {
      logger.error(`Failed to resolve mesh resource: ${props.mesh}`, error);
    }
  }

  return object;
}

/**
 * Apply material_override and surface_material_override properties.
 */
async function applyMaterialOverrides(
  mesh: THREE.Mesh,
  props: MeshInstance3DProperties,
  context: RenderContext
): Promise<void> {
  // Apply material_override (all surfaces)
  if (props.material_override) {
    const material = await resolveMaterialResource(props.material_override, context);
    if (material) {
      mesh.material = material;
      logger.info(
        `[MeshInstance3D] Applied material_override: ${props.material_override}`
      );
    }
  }

  // Apply surface_material_override (per-surface)
  if (props.surface_material_override) {
    const materials = Array.isArray(mesh.material) ? mesh.material : [mesh.material];

    for (const [surfaceIndex, materialRef] of Object.entries(
      props.surface_material_override
    )) {
      const index = parseInt(surfaceIndex, 10);
      const material = await resolveMaterialResource(materialRef, context);

      if (material && materials[index]) {
        materials[index] = material;
        logger.info(
          `[MeshInstance3D] Applied surface_material_override/${index}: ${materialRef}`
        );
      } else if (!materials[index]) {
        logger.warn(
          `[MeshInstance3D] Surface index ${index} does not exist on mesh`
        );
      }
    }

    // Update mesh material if we modified it
    if (materials.length === 1) {
      mesh.material = materials[0];
    } else {
      mesh.material = materials;
    }
  }
}

/**
 * Resolve a material reference (SubResource or ExtResource).
 */
async function resolveMaterialResource(
  materialRef: string,
  context: RenderContext
): Promise<THREE.Material | null> {
  try {
    return await resolveResource<THREE.Material>(
      materialRef,
      context,
      MaterialRegistry
    );
  } catch (error) {
    logger.error(`Failed to resolve material resource: ${materialRef}`, error);
    return null;
  }
}
```

## Linter Requirements

### 1. Validation Rules

**File**: `packages/textscene-core/src/nodes/3d/meshinstance3d/linter.ts`

```typescript
import { LintRule, LintContext } from '../../../linter/types';
import { ruleRegistry } from '../../../linter/RuleRegistry';
import { isMeshInstance3DNode } from './linterParser';

const materialOverrideValidationRule: LintRule = {
  meta: {
    name: 'valid-material-override',
    description: 'Validates material_override and surface_material_override references',
  },
  check: (context: LintContext) => {
    const { node, parsedScene } = context;

    if (!isMeshInstance3DNode(node)) return [];

    const diagnostics = [];

    // Validate material_override reference
    if (node.properties.material_override) {
      const exists = context.resourceExists(node.properties.material_override);
      if (!exists) {
        diagnostics.push({
          severity: 'error',
          message: `Material override reference not found: ${node.properties.material_override}`,
          nodeName: node.name,
          line: node.line,
        });
      }
    }

    // Validate surface_material_override references
    for (const [key, value] of Object.entries(node.properties)) {
      const match = key.match(/^surface_material_override\/(\d+)$/);
      if (match) {
        const exists = context.resourceExists(value);
        if (!exists) {
          diagnostics.push({
            severity: 'error',
            message: `Surface material override reference not found: ${value} (surface ${match[1]})`,
            nodeName: node.name,
            line: node.line,
          });
        }
      }
    }

    return diagnostics;
  },
};

ruleRegistry.register(materialOverrideValidationRule);
```

```typescript
export interface MeshInstance3DNode extends Node3DNode {
  properties: {
    mesh?: string;
    skeleton?: string;
    material_override?: string; // NEW
    [key: string]: unknown; // Captures surface_material_override/N
  };
}

export function isMeshInstance3DNode(node: Node): node is MeshInstance3DNode {
  return node.type === 'MeshInstance3D';
}

export function parseMeshInstance3DNode(section: Section): MeshInstance3DNode {
  const baseNode = parseNode3DSection(section);

  return {
    ...baseNode,
    properties: {
      ...baseNode.properties,
      mesh: section.properties.mesh,
      skeleton: section.properties.skeleton,
      material_override: section.properties.material_override, // NEW
      // surface_material_override/* properties are captured by [key: string]: unknown
    },
  };
}
```

```typescript
describe('MeshInstance3D Parser - Material Overrides', () => {
  test('parses material_override property', () => {
    const section: Section = {
      heading: '[node name="Door" type="MeshInstance3D" parent="."]',
      properties: {
        mesh: 'SubResource("BoxMesh_abc123")',
        material_override: 'SubResource("StandardMaterial3D_wood")',
      },
      line: 10,
      headingEndLine: 10,
    };

    const result = parseMeshInstance3D(section);

    expect(result.material_override).toBe('SubResource("StandardMaterial3D_wood")');
  });

  test('parses surface_material_override indexed properties', () => {
    const section: Section = {
      heading: '[node name="MultiMat" type="MeshInstance3D" parent="."]',
      properties: {
        mesh: 'SubResource("BoxMesh_abc")',
        'surface_material_override/0': 'ExtResource("1_metal")',
        'surface_material_override/1': 'SubResource("StandardMaterial3D_glass")',
      },
      line: 20,
      headingEndLine: 20,
    };

    const result = parseMeshInstance3D(section);

    expect(result.surface_material_override).toEqual({
      0: 'ExtResource("1_metal")',
      1: 'SubResource("StandardMaterial3D_glass")',
    });
  });

  test('handles missing material override properties', () => {
    const section: Section = {
      heading: '[node name="Plain" type="MeshInstance3D" parent="."]',
      properties: {
        mesh: 'SubResource("SphereMesh_xyz")',
      },
      line: 30,
      headingEndLine: 30,
    };

    const result = parseMeshInstance3D(section);

    expect(result.material_override).toBeUndefined();
    expect(result.surface_material_override).toBeUndefined();
  });
});
```
