# WI-36: Enhanced Scene Instance Tracking

## Overview

Implement comprehensive tracking and visualization of scene instancing relationships in TSCN files. Help developers understand how scenes are composed through instancing, track instance modifications (editable children), and navigate between parent scenes and their instances.

## Implementation Requirements

From TODO.md Phase 10: Advanced Navigation:

- Detect and parse scene instance nodes
- Track instance-to-source scene relationships
- Show instance modifications and overrides
- Display editable children indicators
- Navigate to source scene from instance
- Show all locations where a scene is instanced
- Detect circular instance dependencies
- Highlight instance-modified properties
- Support instance placeholders
- Generate instance dependency graph
- Quick actions to open instance sources

## TSCN Instance Format

### Scene Instance Node

```ini
[node name="PlayerInstance" parent="." instance=ExtResource("1_abc123")]

[node name="ModifiedChild" parent="PlayerInstance/OriginalChild"]
transform = Transform3D(1, 0, 0, 0, 1, 0, 0, 0, 1, 5, 0, 0)
```

### Editable Instances

```ini
[editable path="PlayerInstance"]
[editable path="EnemySpawner/EnemyInstance"]
```

## Implementation Notes

### Scene Instance Tracker

```typescript
// sceneInstanceTracker.ts
import * as vscode from 'vscode';
import * as path from 'path';

export interface SceneInstance {
  nodeName: string;
  nodePath: string;
  instancePath: string;      // Path to instanced scene
  resourceId: string;         // ExtResource ID
  isEditable: boolean;
  modifications: InstanceModification[];
  children: SceneInstance[];
}

export interface InstanceModification {
  property: string;
  originalValue: any;
  modifiedValue: any;
  nodePath: string;
}

export interface InstanceDependency {
  sourceScene: string;
  instancedIn: string[];
  depth: number;
}

export class SceneInstanceTracker {
  private instances: Map<string, SceneInstance[]> = new Map();
  private dependencies: Map<string, InstanceDependency> = new Map();

  parseScene(tscnUri: vscode.Uri, content: string): void {
    const instances = this.extractInstances(content);
    const editablePaths = this.extractEditablePaths(content);

    // Mark editable instances
    instances.forEach(inst => {
      inst.isEditable = editablePaths.includes(inst.nodePath);
    });

    // Parse instance modifications
    instances.forEach(inst => {
      inst.modifications = this.findInstanceModifications(content, inst);
    });

    this.instances.set(tscnUri.toString(), instances);

    // Build dependency graph
    this.updateDependencyGraph(tscnUri.toString(), instances);
  }

  private extractInstances(content: string): SceneInstance[] {
    const instances: SceneInstance[] = [];
    const lines = content.split('\n');
    const nodeRegex = /\[node name="([^"]+)"(?:[^\]]*parent="([^"]*)")?(?:[^\]]*instance=ExtResource\("([^"]+)"\))?/;

    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      const match = nodeRegex.exec(line);

      if (match && match[3]) {
        // This is an instance node
        const nodeName = match[1];
        const parent = match[2] || '';
        const resourceId = match[3];

        instances.push({
          nodeName,
          nodePath: parent ? `${parent}/${nodeName}` : nodeName,
          instancePath: '', // Resolved later from resources
          resourceId,
          isEditable: false,
          modifications: [],
          children: [],
        });
      }
    }

    return instances;
  }

  private extractEditablePaths(content: string): string[] {
    const editablePaths: string[] = [];
    const regex = /\[editable path="([^"]+)"\]/g;
    let match;

    while ((match = regex.exec(content)) !== null) {
      editablePaths.push(match[1]);
    }

    return editablePaths;
  }

  private findInstanceModifications(
    content: string,
    instance: SceneInstance
  ): InstanceModification[] {
    const modifications: InstanceModification[] = [];

    // Find nodes that are children of the instance and have property overrides
    const lines = content.split('\n');
    let currentNode: string | null = null;
    let inInstanceSubtree = false;

    for (const line of lines) {
      if (line.startsWith('[node')) {
        const nameMatch = /name="([^"]+)"/.exec(line);
        const parentMatch = /parent="([^"]+)"/.exec(line);

        if (nameMatch) {
          currentNode = nameMatch[1];
          const parent = parentMatch ? parentMatch[1] : '';
          inInstanceSubtree = parent.startsWith(instance.nodePath);
        }
      } else if (inInstanceSubtree && currentNode && line.includes('=')) {
        // Property override
        const [property, value] = line.split('=').map(s => s.trim());
        if (property && value) {
          modifications.push({
            property,
            originalValue: null, // Would need to load source scene
            modifiedValue: value,
            nodePath: currentNode,
          });
        }
      }
    }

    return modifications;
  }

  private updateDependencyGraph(
    sceneUri: string,
    instances: SceneInstance[]
  ): void {
    instances.forEach(inst => {
      if (!this.dependencies.has(inst.instancePath)) {
        this.dependencies.set(inst.instancePath, {
          sourceScene: inst.instancePath,
          instancedIn: [],
          depth: 0,
        });
      }

      const dep = this.dependencies.get(inst.instancePath)!;
      dep.instancedIn.push(sceneUri);
    });
  }

  getInstancesInScene(sceneUri: vscode.Uri): SceneInstance[] {
    return this.instances.get(sceneUri.toString()) || [];
  }

  getSceneUsages(scenePath: string): string[] {
    const dep = this.dependencies.get(scenePath);
    return dep ? dep.instancedIn : [];
  }

  detectCircularDependencies(): string[][] {
    const cycles: string[][] = [];
    const visited = new Set<string>();
    const recursionStack = new Set<string>();

    const dfs = (scene: string, path: string[]): void => {
      visited.add(scene);
      recursionStack.add(scene);
      path.push(scene);

      const instances = this.instances.get(scene) || [];
      for (const inst of instances) {
        if (!visited.has(inst.instancePath)) {
          dfs(inst.instancePath, path);
        } else if (recursionStack.has(inst.instancePath)) {
          // Found cycle
          const cycleStart = path.indexOf(inst.instancePath);
          cycles.push(path.slice(cycleStart).concat([inst.instancePath]));
        }
      }

      path.pop();
      recursionStack.delete(scene);
    };

    for (const scene of this.instances.keys()) {
      if (!visited.has(scene)) {
        dfs(scene, []);
      }
    }

    return cycles;
  }

  generateDependencyGraph(): string {
    // Generate DOT format for visualization
    let dot = 'digraph SceneInstances {\n';
    dot += '  rankdir=LR;\n';

    for (const [scene, instances] of this.instances) {
      const sceneName = path.basename(scene, '.tscn');

      instances.forEach(inst => {
        const instName = path.basename(inst.instancePath, '.tscn');
        dot += `  "${sceneName}" -> "${instName}";\n`;
      });
    }

    dot += '}\n';
    return dot;
  }
}
```

### Instance Navigation Provider

```typescript
// instanceNavigationProvider.ts
import * as vscode from 'vscode';
import { SceneInstanceTracker } from './sceneInstanceTracker';

export class InstanceNavigationProvider {
  constructor(private tracker: SceneInstanceTracker) {}

  async navigateToInstanceSource(nodePath: string, currentScene: vscode.Uri): Promise<void> {
    const instances = this.tracker.getInstancesInScene(currentScene);
    const instance = instances.find(i => i.nodePath === nodePath);

    if (!instance) {
      vscode.window.showWarningMessage('Node is not an instance');
      return;
    }

    // Open source scene
    const sourceUri = vscode.Uri.file(instance.instancePath);
    await vscode.commands.executeCommand('vscode.open', sourceUri);
  }

  async showInstanceUsages(scenePath: string): Promise<void> {
    const usages = this.tracker.getSceneUsages(scenePath);

    if (usages.length === 0) {
      vscode.window.showInformationMessage('Scene is not instanced anywhere');
      return;
    }

    // Show quick pick
    const items = usages.map(uri => ({
      label: path.basename(uri),
      description: uri,
      uri: vscode.Uri.parse(uri),
    }));

    const selected = await vscode.window.showQuickPick(items, {
      placeHolder: `Scene is instanced in ${usages.length} location(s)`,
    });

    if (selected) {
      await vscode.commands.executeCommand('vscode.open', selected.uri);
    }
  }

  async showCircularDependencies(): Promise<void> {
    const cycles = this.tracker.detectCircularDependencies();

    if (cycles.length === 0) {
      vscode.window.showInformationMessage('No circular dependencies detected');
      return;
    }

    // Show warning with cycles
    const message = `Found ${cycles.length} circular dependency cycle(s)`;
    const action = await vscode.window.showWarningMessage(
      message,
      'Show Details'
    );

    if (action === 'Show Details') {
      const panel = vscode.window.createWebviewPanel(
        'circularDeps',
        'Circular Dependencies',
        vscode.ViewColumn.Two,
        {}
      );

      panel.webview.html = this.getCircularDepsHtml(cycles);
    }
  }

  private getCircularDepsHtml(cycles: string[][]): string {
    let html = '<html><body><h1>Circular Dependencies</h1><ul>';

    cycles.forEach((cycle, index) => {
      html += `<li><strong>Cycle ${index + 1}:</strong><br>`;
      html += cycle.map(s => path.basename(s)).join(' → ');
      html += '</li>';
    });

    html += '</ul></body></html>';
    return html;
  }
}
```

### Instance Tree Item Decorator

```typescript
// instanceTreeDecorator.ts
export class InstanceTreeDecorator {
  decorateTreeItem(
    item: vscode.TreeItem,
    instance?: SceneInstance
  ): void {
    if (!instance) return;

    // Add instance icon
    item.iconPath = new vscode.ThemeIcon(
      'package',
      new vscode.ThemeColor('symbolIcon.packageForeground')
    );

    // Add modification indicator
    if (instance.modifications.length > 0) {
      item.description = `${item.description || ''} [${instance.modifications.length} mods]`;
    }

    // Add editable indicator
    if (instance.isEditable) {
      item.contextValue = 'editableInstance';

      const tooltip = new vscode.MarkdownString();
      tooltip.appendMarkdown(`**Instance:** ${path.basename(instance.instancePath)}\n\n`);
      tooltip.appendMarkdown(`**Editable:** Yes\n\n`);

      if (instance.modifications.length > 0) {
        tooltip.appendMarkdown(`**Modifications (${instance.modifications.length}):**\n`);
        instance.modifications.slice(0, 5).forEach(mod => {
          tooltip.appendMarkdown(`- \`${mod.property}\` on ${mod.nodePath}\n`);
        });
        if (instance.modifications.length > 5) {
          tooltip.appendMarkdown(`- ... and ${instance.modifications.length - 5} more\n`);
        }
      }

      item.tooltip = tooltip;
    }
  }
}
```
