# WI-34: Texture/Material Inspector Panel

## Overview

Implement a dedicated inspector panel for examining texture and material properties in detail. This panel provides developers with comprehensive information about material settings (metalness, roughness, albedo), texture details (resolution, format, memory usage), and visual previews, enabling better asset optimization and debugging.

## Implementation Requirements

From TODO.md Phase 9: Enhanced Visualization:

- Inspector panel showing material properties
- Display texture information (dimensions, format, size)
- Show material PBR properties (metalness, roughness, etc.)
- Visual preview of materials and textures
- Memory usage statistics for textures
- List all textures used by a material
- Support for StandardMaterial3D properties
- Display shader parameters (if custom shaders)
- Show texture compression settings
- Editable properties (link to WI-30)
- Export material/texture information

## Implementation Notes

### Material Inspector Panel Class

```typescript
// materialInspectorPanel.ts
import * as vscode from 'vscode';
import * as THREE from 'three';

export interface MaterialInfo {
  name: string;
  type: string;
  properties: {
    color?: string;
    metalness?: number;
    roughness?: number;
    emissive?: string;
    emissiveIntensity?: number;
    transparent?: boolean;
    opacity?: number;
    side?: string;
  };
  textures: TextureInfo[];
}

export interface TextureInfo {
  type: 'map' | 'normalMap' | 'metalnessMap' | 'roughnessMap' | 'aoMap';
  path: string;
  width: number;
  height: number;
  format: string;
  memoryMB: number;
  wrapS: string;
  wrapT: string;
  filter: string;
}

export class MaterialInspectorPanel {
  private panel: vscode.WebviewPanel | undefined;
  private currentMaterial: MaterialInfo | null = null;

  constructor(
    private readonly extensionUri: vscode.Uri,
    private readonly context: vscode.ExtensionContext
  ) {}

  show(materialInfo: MaterialInfo): void {
    if (this.panel) {
      this.panel.reveal(vscode.ViewColumn.Two);
    } else {
      this.panel = this.createPanel();
    }

    this.currentMaterial = materialInfo;
    this.updateContent(materialInfo);
  }

  private createPanel(): vscode.WebviewPanel {
    const panel = vscode.window.createWebviewPanel(
      'materialInspector',
      'Material Inspector',
      vscode.ViewColumn.Two,
      {
        enableScripts: true,
        retainContextWhenHidden: true,
        localResourceRoots: [
          vscode.Uri.joinPath(this.extensionUri, 'media'),
        ],
      }
    );

    panel.webview.html = this.getHtmlContent(panel.webview);

    panel.onDidDispose(() => {
      this.panel = undefined;
    });

    // Handle messages from webview
    panel.webview.onDidReceiveMessage(
      async (message) => {
        switch (message.command) {
          case 'exportMaterialInfo':
            await this.exportMaterialInfo();
            break;
          case 'openTexture':
            await this.openTextureFile(message.path);
            break;
        }
      },
      undefined,
      this.context.subscriptions
    );

    return panel;
  }

  private updateContent(materialInfo: MaterialInfo): void {
    if (!this.panel) return;

    this.panel.webview.postMessage({
      command: 'showMaterial',
      material: materialInfo,
    });
  }

  private async exportMaterialInfo(): Promise<void> {
    if (!this.currentMaterial) return;

    const json = JSON.stringify(this.currentMaterial, null, 2);

    const uri = await vscode.window.showSaveDialog({
      defaultUri: vscode.Uri.file('material-info.json'),
      filters: {
        'JSON': ['json'],
      },
    });

    if (uri) {
      await vscode.workspace.fs.writeFile(
        uri,
        Buffer.from(json, 'utf8')
      );
      vscode.window.showInformationMessage(`Material info exported to ${uri.fsPath}`);
    }
  }

  private async openTextureFile(texturePath: string): Promise<void> {
    const uri = vscode.Uri.file(texturePath);
    await vscode.commands.executeCommand('vscode.open', uri);
  }

  private getHtmlContent(webview: vscode.Webview): string {
    const scriptUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'materialInspector.js')
    );

    const styleUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'materialInspector.css')
    );

    return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <meta http-equiv="Content-Security-Policy" content="default-src 'none'; style-src ${webview.cspSource} 'unsafe-inline'; script-src ${webview.cspSource}; img-src ${webview.cspSource} data:;">
  <link href="${styleUri}" rel="stylesheet">
  <title>Material Inspector</title>
</head>
<body>
  <div id="inspector-container">
    <div id="no-material">
      <p>No material selected</p>
    </div>
    <div id="material-content" style="display: none;">
      <div class="section">
        <h2 id="material-name">Material Name</h2>
        <span id="material-type" class="badge">Type</span>
      </div>

      <div class="section">
        <h3>Properties</h3>
        <div id="properties-grid" class="properties-grid"></div>
      </div>

      <div class="section">
        <h3>Textures</h3>
        <div id="textures-list" class="textures-list"></div>
      </div>

      <div class="section">
        <h3>Memory Usage</h3>
        <div id="memory-stats"></div>
      </div>

      <div class="actions">
        <button id="export-btn">Export Material Info</button>
      </div>
    </div>
  </div>
  <script src="${scriptUri}"></script>
</body>
</html>`;
  }

  dispose(): void {
    if (this.panel) {
      this.panel.dispose();
    }
  }
}
```

### Webview Script for Inspector

```typescript
// media/materialInspector.js
(function() {
  const vscode = acquireVsCodeApi();

  window.addEventListener('message', event => {
    const message = event.data;

    switch (message.command) {
      case 'showMaterial':
        displayMaterial(message.material);
        break;
    }
  });

  function displayMaterial(material) {
    const noMaterial = document.getElementById('no-material');
    const content = document.getElementById('material-content');

    noMaterial.style.display = 'none';
    content.style.display = 'block';

    // Update header
    document.getElementById('material-name').textContent = material.name;
    document.getElementById('material-type').textContent = material.type;

    // Display properties
    displayProperties(material.properties);

    // Display textures
    displayTextures(material.textures);

    // Display memory stats
    displayMemoryStats(material.textures);
  }

  function displayProperties(properties) {
    const grid = document.getElementById('properties-grid');
    grid.innerHTML = '';

    for (const [key, value] of Object.entries(properties)) {
      const item = document.createElement('div');
      item.className = 'property-item';

      const label = document.createElement('label');
      label.textContent = formatPropertyName(key);

      const valueElem = document.createElement('span');
      valueElem.className = 'property-value';
      valueElem.textContent = formatPropertyValue(key, value);

      // Add color swatch for color properties
      if (key === 'color' || key === 'emissive') {
        const swatch = document.createElement('span');
        swatch.className = 'color-swatch';
        swatch.style.backgroundColor = value;
        valueElem.prepend(swatch);
      }

      item.appendChild(label);
      item.appendChild(valueElem);
      grid.appendChild(item);
    }
  }

  function displayTextures(textures) {
    const list = document.getElementById('textures-list');
    list.innerHTML = '';

    if (textures.length === 0) {
      list.innerHTML = '<p class="empty-state">No textures</p>';
      return;
    }

    textures.forEach(texture => {
      const card = document.createElement('div');
      card.className = 'texture-card';

      const header = document.createElement('div');
      header.className = 'texture-header';

      const type = document.createElement('h4');
      type.textContent = formatTextureName(texture.type);

      const size = document.createElement('span');
      size.className = 'texture-size';
      size.textContent = `${texture.width}x${texture.height}`;

      header.appendChild(type);
      header.appendChild(size);

      const details = document.createElement('div');
      details.className = 'texture-details';

      const detailsHTML = `
        <div class="detail-row">
          <span>Path:</span>
          <a href="#" class="texture-link" data-path="${texture.path}">${getFileName(texture.path)}</a>
        </div>
        <div class="detail-row">
          <span>Format:</span>
          <span>${texture.format}</span>
        </div>
        <div class="detail-row">
          <span>Memory:</span>
          <span>${texture.memoryMB.toFixed(2)} MB</span>
        </div>
        <div class="detail-row">
          <span>Wrap:</span>
          <span>${texture.wrapS} / ${texture.wrapT}</span>
        </div>
        <div class="detail-row">
          <span>Filter:</span>
          <span>${texture.filter}</span>
        </div>
      `;

      details.innerHTML = detailsHTML;

      card.appendChild(header);
      card.appendChild(details);
      list.appendChild(card);
    });

    // Add click handlers for texture links
    document.querySelectorAll('.texture-link').forEach(link => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        const path = e.target.getAttribute('data-path');
        vscode.postMessage({
          command: 'openTexture',
          path: path,
        });
      });
    });
  }

  function displayMemoryStats(textures) {
    const stats = document.getElementById('memory-stats');

    const totalMemory = textures.reduce((sum, tex) => sum + tex.memoryMB, 0);
    const textureCount = textures.length;

    stats.innerHTML = `
      <div class="stat-item">
        <span class="stat-label">Total Textures:</span>
        <span class="stat-value">${textureCount}</span>
      </div>
      <div class="stat-item">
        <span class="stat-label">Total Memory:</span>
        <span class="stat-value">${totalMemory.toFixed(2)} MB</span>
      </div>
      <div class="stat-item">
        <span class="stat-label">Average Size:</span>
        <span class="stat-value">${textureCount > 0 ? (totalMemory / textureCount).toFixed(2) : 0} MB</span>
      </div>
    `;
  }

  function formatPropertyName(key) {
    return key
      .replace(/([A-Z])/g, ' $1')
      .replace(/^./, str => str.toUpperCase());
  }

  function formatPropertyValue(key, value) {
    if (typeof value === 'boolean') {
      return value ? 'Yes' : 'No';
    }
    if (typeof value === 'number') {
      return value.toFixed(3);
    }
    return String(value);
  }

  function formatTextureName(type) {
    const names = {
      'map': 'Albedo Map',
      'normalMap': 'Normal Map',
      'metalnessMap': 'Metalness Map',
      'roughnessMap': 'Roughness Map',
      'aoMap': 'Ambient Occlusion Map',
    };
    return names[type] || type;
  }

  function getFileName(path) {
    return path.split(/[\\/]/).pop();
  }

  // Export button
  document.getElementById('export-btn').addEventListener('click', () => {
    vscode.postMessage({ command: 'exportMaterialInfo' });
  });
})();
```

### Material Data Extractor

```typescript
// materialExtractor.ts
import * as THREE from 'three';
import { MaterialInfo, TextureInfo } from './materialInspectorPanel';

export class MaterialExtractor {
  extractMaterialInfo(material: THREE.Material): MaterialInfo {
    const info: MaterialInfo = {
      name: material.name || 'Unnamed Material',
      type: material.type,
      properties: {},
      textures: [],
    };

    // Extract StandardMaterial properties
    if (material instanceof THREE.MeshStandardMaterial) {
      info.properties = {
        color: '#' + material.color.getHexString(),
        metalness: material.metalness,
        roughness: material.roughness,
        emissive: '#' + material.emissive.getHexString(),
        emissiveIntensity: material.emissiveIntensity,
        transparent: material.transparent,
        opacity: material.opacity,
        side: this.getSideName(material.side),
      };

      // Extract textures
      if (material.map) {
        info.textures.push(this.extractTextureInfo('map', material.map));
      }
      if (material.normalMap) {
        info.textures.push(this.extractTextureInfo('normalMap', material.normalMap));
      }
      if (material.metalnessMap) {
        info.textures.push(this.extractTextureInfo('metalnessMap', material.metalnessMap));
      }
      if (material.roughnessMap) {
        info.textures.push(this.extractTextureInfo('roughnessMap', material.roughnessMap));
      }
      if (material.aoMap) {
        info.textures.push(this.extractTextureInfo('aoMap', material.aoMap));
      }
    }

    return info;
  }

  private extractTextureInfo(type: string, texture: THREE.Texture): TextureInfo {
    const width = texture.image?.width || 0;
    const height = texture.image?.height || 0;

    // Calculate memory usage (RGBA = 4 bytes per pixel)
    const bytesPerPixel = 4;
    const memoryBytes = width * height * bytesPerPixel;
    const memoryMB = memoryBytes / (1024 * 1024);

    return {
      type: type as any,
      path: (texture.userData?.path as string) || 'unknown',
      width,
      height,
      format: this.getFormatName(texture.format),
      memoryMB,
      wrapS: this.getWrapName(texture.wrapS),
      wrapT: this.getWrapName(texture.wrapT),
      filter: this.getFilterName(texture.minFilter),
    };
  }

  private getSideName(side: THREE.Side): string {
    switch (side) {
      case THREE.FrontSide: return 'Front';
      case THREE.BackSide: return 'Back';
      case THREE.DoubleSide: return 'Double';
      default: return 'Unknown';
    }
  }

  private getFormatName(format: THREE.PixelFormat): string {
    const formats: Record<number, string> = {
      [THREE.RGBAFormat]: 'RGBA',
      [THREE.RGBFormat]: 'RGB',
      [THREE.RedFormat]: 'Red',
      [THREE.RGFormat]: 'RG',
    };
    return formats[format] || 'Unknown';
  }

  private getWrapName(wrap: THREE.Wrapping): string {
    switch (wrap) {
      case THREE.RepeatWrapping: return 'Repeat';
      case THREE.ClampToEdgeWrapping: return 'Clamp';
      case THREE.MirroredRepeatWrapping: return 'Mirror';
      default: return 'Unknown';
    }
  }

  private getFilterName(filter: THREE.TextureFilter): string {
    const filters: Record<number, string> = {
      [THREE.LinearFilter]: 'Linear',
      [THREE.NearestFilter]: 'Nearest',
      [THREE.LinearMipmapLinearFilter]: 'Linear Mipmap Linear',
      [THREE.LinearMipmapNearestFilter]: 'Linear Mipmap Nearest',
      [THREE.NearestMipmapLinearFilter]: 'Nearest Mipmap Linear',
      [THREE.NearestMipmapNearestFilter]: 'Nearest Mipmap Nearest',
    };
    return filters[filter] || 'Unknown';
  }
}
```
