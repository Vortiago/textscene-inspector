### 1. TscnRenderer Integration

**File**: `packages/textscene-core/src/core/TscnRenderer.ts`

```typescript
import { getHierarchyRegistry } from './HierarchyRegistry.js';
import { ReconciliationEngine } from './ReconciliationEngine.js';
import type { SceneGraph } from './SceneGraph.js';
import type { Patch } from './ReconciliationEngine.js';

export class TscnRenderer {
  private currentGraph: SceneGraph | null = null;
  private reconciler = new ReconciliationEngine();

  async loadScene(rootScenePath: string): Promise<void> {
    const registry = getHierarchyRegistry();
    const graph = await registry.getOrCreateHierarchy(rootScenePath);

    // Initial render (no old graph to diff)
    await this.initialRender(graph);
    this.currentGraph = graph;

    // Subscribe to hierarchy updates
    this.resourceRegistry.getEventBus().on('hierarchy', 'updated',
      (scene: string, newGraph: SceneGraph) => {
        if (scene === rootScenePath) {
          this.reconcile(newGraph);
        }
      }
    );

    // Register panel with registry
    registry.registerPanel(this.panelId, rootScenePath);
  }

  private async initialRender(graph: SceneGraph): Promise<void> {
    // Render all nodes from flattened array
    for (const sceneNode of graph.flattenedNodes) {
      await this.nodeLifecycle.addNode(sceneNode.path, sceneNode.data, this.sceneData);
    }
  }

  private reconcile(newGraph: SceneGraph): void {
    if (!this.currentGraph) return;

    const patches = this.reconciler.diff(this.currentGraph, newGraph);

    // Apply patches
    patches.forEach(patch => this.applyPatch(patch));

    this.currentGraph = newGraph;
  }

  private async applyPatch(patch: Patch): Promise<void> {
    switch (patch.type) {
      case 'add':
        await this.nodeLifecycle.addNode(patch.nodePath, patch.node!.data, this.sceneData);
        break;

      case 'remove':
        this.nodeLifecycle.removeNode(patch.nodePath);
        break;

      case 'update-transform':
        this.selectiveUpdate.updateTransform(patch.nodePath, patch.data.transform);
        break;

      case 'update-material':
        this.selectiveUpdate.updateMaterial(patch.nodePath, patch.data.materialOverride);
        break;

      case 'update-visibility':
        this.selectiveUpdate.updateVisibility(patch.nodePath, patch.data.visible);
        break;

      case 'update-full':
        this.nodeLifecycle.removeNode(patch.nodePath);
        await this.nodeLifecycle.addNode(patch.nodePath, patch.node!.data, this.sceneData);
        break;
    }
  }

  dispose(): void {
    // Unregister panel
    const registry = getHierarchyRegistry();
    registry.unregisterPanel(this.panelId);

    // ... rest of cleanup
  }
}
```
