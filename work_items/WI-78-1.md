# WI-78.1: Immutable SceneGraph Model

**Parent**: WI-78 (Immutable Scene Hierarchies)
**Status**: Not Started
**Complexity**: Moderate

## Goal

Create immutable SceneGraph data structure with Copy-on-Write builder pattern.

## Problem

Current `TscnScene` is mutable:
```typescript
// BAD: TscnNode.children mutated during external scene loading
node.children.push(externalNode);  // Mutation!
```

**Issues:**
- Can't compare old vs new (data changed in-place)
- Can't keep history (no snapshots)
- Race conditions (multiple readers/writers)
- Hard to debug (data changes unexpectedly)

## Solution

**Immutable data structures** with **Copy-on-Write updates**.

### Immutable SceneGraph

```typescript
/**
 * Immutable scene graph representing resolved scene hierarchy.
 * Once created, never modified. Updates create new instances.
 */
interface SceneGraph {
  readonly rootScene: string;                        // Root scene path
  readonly scenes: ReadonlyMap<string, ParsedScene>; // All scenes in hierarchy
  readonly flattenedNodes: ReadonlyArray<SceneNode>; // All nodes (flattened)
  readonly dependencies: DependencyGraph;            // Scene dependency tracking
  readonly version: number;                          // Incrementing version (debugging)
  readonly timestamp: number;                        // Creation timestamp
}

/**
 * Parsed scene (immutable).
 */
interface ParsedScene {
  readonly path: string;                                 // Scene file path
  readonly nodes: ReadonlyArray<TscnNode>;               // Original nodes (not mutated)
  readonly externalScenes: ReadonlyArray<ExternalSceneRef>; // External scene references
  readonly internalResources: ReadonlyArray<TscnInternalResource>;
  readonly externalResources: ReadonlyArray<TscnExternalResource>;
}

/**
 * Scene node with full path and source tracking.
 */
interface SceneNode {
  readonly path: string;      // Full path (e.g., "Main/Hallway/Door")
  readonly name: string;      // Node name (e.g., "Door")
  readonly data: TscnNode;    // Original TSCN node data (immutable)
  readonly source: string;    // Source scene file (e.g., "res://door.tscn")
  readonly parent: string | null; // Parent node path (null for roots)
}

/**
 * External scene reference (before/after resolution).
 */
interface ExternalSceneRef {
  readonly nodeName: string;          // Node name for this instance
  readonly scenePath: string;         // External scene file path
  readonly resolvedScene?: ParsedScene; // Resolved scene (if loaded)
  readonly resolutionStatus: 'pending' | 'resolved' | 'failed';
}
```

**Key Properties:**
- All fields are `readonly` (TypeScript compiler enforces)
- Attempts to mutate = compile-time error
- Can be safely shared between panels (no mutations)
- Old versions coexist with new versions (snapshots)

### Copy-on-Write Builder

```typescript
/**
 * Builder for creating SceneGraphs with Copy-on-Write semantics.
 * Use to update existing graph without mutating it.
 */
class SceneGraphBuilder {
  private rootScene: string = '';
  private scenes = new Map<string, ParsedScene>();
  private dependencies: DependencyGraph = new DependencyGraph();
  private baseVersion: number = 0;

  /**
   * Start from existing graph (Copy-on-Write).
   */
  from(graph: SceneGraph): SceneGraphBuilder {
    this.rootScene = graph.rootScene;
    this.scenes = new Map(graph.scenes); // Shallow copy (scenes are immutable)
    this.dependencies = graph.dependencies; // Immutable, can share
    this.baseVersion = graph.version;
    return this;
  }

  /**
   * Set root scene path.
   */
  setRootScene(path: string): SceneGraphBuilder {
    this.rootScene = path;
    return this;
  }

  /**
   * Add scene to hierarchy.
   */
  addScene(scene: ParsedScene): SceneGraphBuilder {
    this.scenes.set(scene.path, scene);
    return this;
  }

  /**
   * Update existing scene (replaces old version).
   */
  updateScene(path: string, scene: ParsedScene): SceneGraphBuilder {
    if (!this.scenes.has(path)) {
      throw new Error(`Scene not found: ${path}`);
    }
    this.scenes.set(path, scene);
    return this;
  }

  /**
   * Remove scene from hierarchy.
   */
  removeScene(path: string): SceneGraphBuilder {
    this.scenes.delete(path);
    return this;
  }

  /**
   * Set dependency graph.
   */
  setDependencies(dependencies: DependencyGraph): SceneGraphBuilder {
    this.dependencies = dependencies;
    return this;
  }

  /**
   * Build immutable SceneGraph.
   * Old graph (if any) remains unchanged.
   */
  build(): SceneGraph {
    if (!this.rootScene) {
      throw new Error('Root scene not set');
    }

    // Flatten all nodes from all scenes
    const flattenedNodes = this.flattenNodes();

    return {
      rootScene: this.rootScene,
      scenes: new Map(this.scenes), // Freeze map
      flattenedNodes: Object.freeze([...flattenedNodes]), // Freeze array
      dependencies: this.dependencies,
      version: this.baseVersion + 1,
      timestamp: Date.now()
    };
  }

  /**
   * Flatten all scenes into single node array.
   */
  private flattenNodes(): SceneNode[] {
    const nodes: SceneNode[] = [];

    for (const scene of this.scenes.values()) {
      this.flattenSceneNodes(scene, '', scene.path, nodes);
    }

    return nodes;
  }

  /**
   * Recursively flatten scene nodes.
   */
  private flattenSceneNodes(
    scene: ParsedScene,
    parentPath: string,
    source: string,
    output: SceneNode[]
  ): void {
    for (const node of scene.nodes) {
      const nodePath = parentPath ? `${parentPath}/${node.name}` : node.name;

      output.push({
        path: nodePath,
        name: node.name,
        data: node,
        source: source,
        parent: parentPath || null
      });

      // Recurse for children
      if (node.children.length > 0) {
        for (const child of node.children) {
          const childPath = `${nodePath}/${child.name}`;
          output.push({
            path: childPath,
            name: child.name,
            data: child,
            source: source,
            parent: nodePath
          });

          // Note: Deep recursion handled by calling flattenSceneNodes with child
        }
      }
    }
  }
}
```

**Usage Example:**
```typescript
// Initial creation
const graph1 = new SceneGraphBuilder()
  .setRootScene('res://main.tscn')
  .addScene(mainScene)
  .addScene(hallwayScene)
  .build();

// Update (Copy-on-Write)
const graph2 = new SceneGraphBuilder()
  .from(graph1)  // Start from existing
  .updateScene('res://hallway.tscn', updatedHallwayScene)
  .build();

// graph1 unchanged, graph2 has update
// Both can coexist in memory
```

### Step 3: Add Node Flattening Logic

**Private method**: `flattenNodes()`

**Algorithm:**
1. Walk all scenes in `this.scenes`
2. For each scene, walk its nodes recursively
3. Build node path (e.g., "Main/Hallway/Door")
4. Track source scene for each node
5. Handle external scene references (WI-78.2 will populate)
6. Return flat array of SceneNode

**Output:**
```typescript
[
  { path: 'Main', name: 'Main', data: {...}, source: 'res://main.tscn', parent: null },
  { path: 'Main/Hallway', name: 'Hallway', data: {...}, source: 'res://main.tscn', parent: 'Main' },
  { path: 'Main/Hallway/Door', name: 'Door', data: {...}, source: 'res://door.tscn', parent: 'Main/Hallway' },
  // ... all nodes flattened
]
```

**Test Cases:**

```typescript
import { describe, it, expect } from 'vitest';
import { SceneGraphBuilder } from './SceneGraphBuilder.js';
import type { ParsedScene } from './SceneGraph.js';

describe('SceneGraphBuilder', () => {
  it('builds empty graph', () => {
    const graph = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .build();

    expect(graph.rootScene).toBe('res://test.tscn');
    expect(graph.scenes.size).toBe(0);
    expect(graph.flattenedNodes.length).toBe(0);
    expect(graph.version).toBe(1);
  });

  it('adds scene to graph', () => {
    const scene: ParsedScene = {
      path: 'res://test.tscn',
      nodes: [{ name: 'Root', type: 'Node3D', children: [], properties: {} }],
      externalScenes: [],
      internalResources: [],
      externalResources: []
    };

    const graph = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .addScene(scene)
      .build();

    expect(graph.scenes.get('res://test.tscn')).toBe(scene);
    expect(graph.flattenedNodes.length).toBe(1);
    expect(graph.flattenedNodes[0].path).toBe('Root');
  });

  it('flattens nested nodes', () => {
    const scene: ParsedScene = {
      path: 'res://test.tscn',
      nodes: [{
        name: 'Root',
        type: 'Node3D',
        children: [
          { name: 'Child1', type: 'Node3D', children: [], properties: {} },
          { name: 'Child2', type: 'Node3D', children: [], properties: {} }
        ],
        properties: {}
      }],
      externalScenes: [],
      internalResources: [],
      externalResources: []
    };

    const graph = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .addScene(scene)
      .build();

    expect(graph.flattenedNodes.length).toBe(3);
    expect(graph.flattenedNodes[0].path).toBe('Root');
    expect(graph.flattenedNodes[1].path).toBe('Root/Child1');
    expect(graph.flattenedNodes[2].path).toBe('Root/Child2');
  });

  it('preserves old graph when updating (Copy-on-Write)', () => {
    const scene1: ParsedScene = {
      path: 'res://test.tscn',
      nodes: [{ name: 'Root', type: 'Node3D', children: [], properties: {} }],
      externalScenes: [],
      internalResources: [],
      externalResources: []
    };

    const graph1 = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .addScene(scene1)
      .build();

    const scene2: ParsedScene = {
      ...scene1,
      nodes: [{ name: 'Root', type: 'Node3D', children: [], properties: { modified: true } }]
    };

    const graph2 = new SceneGraphBuilder()
      .from(graph1)
      .updateScene('res://test.tscn', scene2)
      .build();

    // graph1 unchanged
    expect(graph1.scenes.get('res://test.tscn')).toBe(scene1);
    expect(graph1.version).toBe(1);

    // graph2 has update
    expect(graph2.scenes.get('res://test.tscn')).toBe(scene2);
    expect(graph2.version).toBe(2);

    // Both coexist
    expect(graph1).not.toBe(graph2);
  });

  it('throws error when updating non-existent scene', () => {
    const graph = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .build();

    expect(() => {
      new SceneGraphBuilder()
        .from(graph)
        .updateScene('res://missing.tscn', {} as ParsedScene);
    }).toThrow('Scene not found: res://missing.tscn');
  });

  it('throws error when building without root scene', () => {
    expect(() => {
      new SceneGraphBuilder().build();
    }).toThrow('Root scene not set');
  });

  it('increments version on each build', () => {
    const graph1 = new SceneGraphBuilder()
      .setRootScene('res://test.tscn')
      .build();

    const graph2 = new SceneGraphBuilder()
      .from(graph1)
      .build();

    const graph3 = new SceneGraphBuilder()
      .from(graph2)
      .build();

    expect(graph1.version).toBe(1);
    expect(graph2.version).toBe(2);
    expect(graph3.version).toBe(3);
  });
});
```
