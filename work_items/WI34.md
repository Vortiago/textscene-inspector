# WI-34: Scene Complexity Metrics

## Overview

Implement comprehensive scene complexity analysis to help developers understand performance characteristics of their TSCN scenes. Display metrics such as node counts, polygon counts, draw calls, texture memory usage, and shader complexity, providing actionable insights for optimization.

## Implementation Requirements

From TODO.md Phase 9: Enhanced Visualization:

- Count total nodes in scene tree
- Calculate total polygon/triangle count
- Estimate draw calls based on materials and meshes
- Calculate total texture memory usage
- Identify performance bottlenecks
- Display metrics in dedicated panel
- Show per-node breakdown
- Highlight expensive nodes (high poly count, complex shaders)
- Compare against performance budgets
- Generate optimization suggestions
- Export metrics as report

### Scene Metrics Analyzer

```typescript
// sceneMetricsAnalyzer.ts
import * as THREE from 'three';

export interface SceneMetrics {
  nodes: NodeMetrics;
  geometry: GeometryMetrics;
  rendering: RenderingMetrics;
  memory: MemoryMetrics;
  performance: PerformanceMetrics;
}

export interface NodeMetrics {
  total: number;
  byType: Record<string, number>;
  depth: number;
  heaviest: NodeInfo[];
}

export interface GeometryMetrics {
  totalTriangles: number;
  totalVertices: number;
  meshes: number;
  uniqueGeometries: number;
  largestMesh: MeshInfo;
}

export interface RenderingMetrics {
  estimatedDrawCalls: number;
  uniqueMaterials: number;
  uniqueShaders: number;
  lights: number;
  shadows: boolean;
}

export interface MemoryMetrics {
  textures: TextureMemoryInfo;
  geometries: GeometryMemoryInfo;
  total: number; // MB
}

export interface PerformanceMetrics {
  score: number; // 0-100
  bottlenecks: string[];
  suggestions: string[];
}

export interface NodeInfo {
  name: string;
  type: string;
  triangles: number;
  path: string;
}

export interface MeshInfo {
  name: string;
  triangles: number;
  vertices: number;
}

export interface TextureMemoryInfo {
  count: number;
  totalMB: number;
  byResolution: Record<string, number>;
}

export interface GeometryMemoryInfo {
  count: number;
  totalMB: number;
}

export class SceneMetricsAnalyzer {
  private scene: THREE.Scene;
  private renderer: THREE.WebGLRenderer;

  constructor(scene: THREE.Scene, renderer: THREE.WebGLRenderer) {
    this.scene = scene;
    this.renderer = renderer;
  }

  analyze(): SceneMetrics {
    return {
      nodes: this.analyzeNodes(),
      geometry: this.analyzeGeometry(),
      rendering: this.analyzeRendering(),
      memory: this.analyzeMemory(),
      performance: this.analyzePerformance(),
    };
  }

  private analyzeNodes(): NodeMetrics {
    const byType: Record<string, number> = {};
    let maxDepth = 0;
    const heaviestNodes: NodeInfo[] = [];

    const analyzeNode = (node: THREE.Object3D, depth: number, path: string) => {
      // Count by type
      const typeName = node.type || 'Object3D';
      byType[typeName] = (byType[typeName] || 0) + 1;

      // Track depth
      maxDepth = Math.max(maxDepth, depth);

      // Track heavy nodes (meshes with high triangle count)
      if (node instanceof THREE.Mesh && node.geometry) {
        const triangles = this.getTriangleCount(node.geometry);
        if (triangles > 1000) {
          heaviestNodes.push({
            name: node.name || 'Unnamed',
            type: node.type,
            triangles,
            path,
          });
        }
      }

      // Recurse
      node.children.forEach((child, index) => {
        analyzeNode(child, depth + 1, `${path}/${child.name || index}`);
      });
    };

    let totalNodes = 0;
    this.scene.traverse(() => totalNodes++);

    analyzeNode(this.scene, 0, 'Scene');

    // Sort heaviest nodes
    heaviestNodes.sort((a, b) => b.triangles - a.triangles);

    return {
      total: totalNodes,
      byType,
      depth: maxDepth,
      heaviest: heaviestNodes.slice(0, 10), // Top 10
    };
  }

  private analyzeGeometry(): GeometryMetrics {
    let totalTriangles = 0;
    let totalVertices = 0;
    let meshCount = 0;
    const uniqueGeometries = new Set<THREE.BufferGeometry>();
    let largestMesh: MeshInfo = { name: '', triangles: 0, vertices: 0 };

    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh && object.geometry) {
        meshCount++;
        uniqueGeometries.add(object.geometry);

        const triangles = this.getTriangleCount(object.geometry);
        const vertices = this.getVertexCount(object.geometry);

        totalTriangles += triangles;
        totalVertices += vertices;

        if (triangles > largestMesh.triangles) {
          largestMesh = {
            name: object.name || 'Unnamed',
            triangles,
            vertices,
          };
        }
      }
    });

    return {
      totalTriangles,
      totalVertices,
      meshes: meshCount,
      uniqueGeometries: uniqueGeometries.size,
      largestMesh,
    };
  }

  private analyzeRendering(): RenderingMetrics {
    const uniqueMaterials = new Set<THREE.Material>();
    const uniqueShaders = new Set<string>();
    let lightCount = 0;
    let hasShadows = false;

    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh && object.material) {
        const materials = Array.isArray(object.material)
          ? object.material
          : [object.material];

        materials.forEach((mat) => {
          uniqueMaterials.add(mat);
          uniqueShaders.add(mat.type);
        });
      }

      if (object instanceof THREE.Light) {
        lightCount++;
        if ((object as any).castShadow) {
          hasShadows = true;
        }
      }
    });

    // Estimate draw calls
    // Each unique combination of geometry + material = 1 draw call
    const geometryMaterialPairs = new Set<string>();
    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh) {
        const geoId = object.geometry.uuid;
        const materials = Array.isArray(object.material)
          ? object.material
          : [object.material];

        materials.forEach((mat) => {
          geometryMaterialPairs.add(`${geoId}_${mat.uuid}`);
        });
      }
    });

    return {
      estimatedDrawCalls: geometryMaterialPairs.size,
      uniqueMaterials: uniqueMaterials.size,
      uniqueShaders: uniqueShaders.size,
      lights: lightCount,
      shadows: hasShadows,
    };
  }

  private analyzeMemory(): MemoryMetrics {
    const textures = this.analyzeTextureMemory();
    const geometries = this.analyzeGeometryMemory();

    return {
      textures,
      geometries,
      total: textures.totalMB + geometries.totalMB,
    };
  }

  private analyzeTextureMemory(): TextureMemoryInfo {
    const uniqueTextures = new Set<THREE.Texture>();
    const byResolution: Record<string, number> = {};

    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh && object.material) {
        const materials = Array.isArray(object.material)
          ? object.material
          : [object.material];

        materials.forEach((mat: any) => {
          // Check common texture properties
          ['map', 'normalMap', 'metalnessMap', 'roughnessMap', 'aoMap'].forEach((prop) => {
            if (mat[prop]) {
              uniqueTextures.add(mat[prop]);
            }
          });
        });
      }
    });

    let totalMemory = 0;
    uniqueTextures.forEach((texture) => {
      if (texture.image) {
        const width = texture.image.width || 0;
        const height = texture.image.height || 0;
        const bytesPerPixel = 4; // RGBA
        const bytes = width * height * bytesPerPixel;
        const mb = bytes / (1024 * 1024);

        totalMemory += mb;

        const res = `${width}x${height}`;
        byResolution[res] = (byResolution[res] || 0) + mb;
      }
    });

    return {
      count: uniqueTextures.size,
      totalMB: totalMemory,
      byResolution,
    };
  }

  private analyzeGeometryMemory(): GeometryMemoryInfo {
    const uniqueGeometries = new Set<THREE.BufferGeometry>();
    let totalMemory = 0;

    this.scene.traverse((object) => {
      if (object instanceof THREE.Mesh && object.geometry) {
        if (!uniqueGeometries.has(object.geometry)) {
          uniqueGeometries.add(object.geometry);

          // Estimate memory usage
          let bytes = 0;
          Object.values(object.geometry.attributes).forEach((attr) => {
            bytes += attr.array.byteLength;
          });
          if (object.geometry.index) {
            bytes += object.geometry.index.array.byteLength;
          }

          totalMemory += bytes / (1024 * 1024);
        }
      }
    });

    return {
      count: uniqueGeometries.size,
      totalMB: totalMemory,
    };
  }

  private analyzePerformance(): PerformanceMetrics {
    const bottlenecks: string[] = [];
    const suggestions: string[] = [];

    const metrics = {
      nodes: this.analyzeNodes(),
      geometry: this.analyzeGeometry(),
      rendering: this.analyzeRendering(),
      memory: this.analyzeMemory(),
    };

    // Check triangle count
    if (metrics.geometry.totalTriangles > 1000000) {
      bottlenecks.push('High triangle count');
      suggestions.push('Consider using LOD (Level of Detail) for distant objects');
    }

    // Check draw calls
    if (metrics.rendering.estimatedDrawCalls > 100) {
      bottlenecks.push('High draw call count');
      suggestions.push('Merge meshes with the same material to reduce draw calls');
    }

    // Check texture memory
    if (metrics.memory.textures.totalMB > 500) {
      bottlenecks.push('High texture memory usage');
      suggestions.push('Use texture compression or reduce texture resolutions');
    }

    // Check unique materials
    if (metrics.rendering.uniqueMaterials > 50) {
      bottlenecks.push('Many unique materials');
      suggestions.push('Share materials between objects where possible');
    }

    // Check scene depth
    if (metrics.nodes.depth > 10) {
      bottlenecks.push('Deep scene hierarchy');
      suggestions.push('Flatten scene hierarchy to improve traversal performance');
    }

    // Calculate score (0-100, higher is better)
    let score = 100;
    score -= Math.min(30, metrics.geometry.totalTriangles / 100000 * 10);
    score -= Math.min(20, metrics.rendering.estimatedDrawCalls / 10);
    score -= Math.min(20, metrics.memory.total / 50);
    score -= Math.min(15, metrics.rendering.uniqueMaterials / 5);
    score -= Math.min(15, metrics.nodes.depth * 1.5);

    score = Math.max(0, Math.min(100, score));

    return {
      score: Math.round(score),
      bottlenecks,
      suggestions,
    };
  }

  private getTriangleCount(geometry: THREE.BufferGeometry): number {
    if (geometry.index) {
      return geometry.index.count / 3;
    }
    return geometry.attributes.position.count / 3;
  }

  private getVertexCount(geometry: THREE.BufferGeometry): number {
    return geometry.attributes.position.count;
  }
}
```

### Metrics Display Panel

```typescript
// metricsPanel.ts
import * as vscode from 'vscode';
import { SceneMetrics } from './sceneMetricsAnalyzer';

export class MetricsPanel {
  private panel: vscode.WebviewPanel | undefined;

  constructor(
    private readonly extensionUri: vscode.Uri,
    private readonly context: vscode.ExtensionContext
  ) {}

  show(metrics: SceneMetrics): void {
    if (this.panel) {
      this.panel.reveal(vscode.ViewColumn.Two);
    } else {
      this.panel = this.createPanel();
    }

    this.updateMetrics(metrics);
  }

  private createPanel(): vscode.WebviewPanel {
    const panel = vscode.window.createWebviewPanel(
      'sceneMetrics',
      'Scene Complexity Metrics',
      vscode.ViewColumn.Two,
      {
        enableScripts: true,
        retainContextWhenHidden: true,
      }
    );

    panel.webview.html = this.getHtmlContent(panel.webview);

    panel.onDidDispose(() => {
      this.panel = undefined;
    });

    return panel;
  }

  private updateMetrics(metrics: SceneMetrics): void {
    if (!this.panel) return;

    this.panel.webview.postMessage({
      command: 'showMetrics',
      metrics,
    });
  }

  private getHtmlContent(webview: vscode.Webview): string {
    const scriptUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'metricsPanel.js')
    );

    const styleUri = webview.asWebviewUri(
      vscode.Uri.joinPath(this.extensionUri, 'media', 'metricsPanel.css')
    );

    return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <link href="${styleUri}" rel="stylesheet">
  <title>Scene Metrics</title>
</head>
<body>
  <div id="metrics-container">
    <div class="score-section">
      <div class="score-circle">
        <span id="score-value">--</span>
      </div>
      <h2>Performance Score</h2>

        <h3>Nodes</h3>
        <div id="node-metrics"></div>
      </div>

      <div class="metric-card">
        <h3>Geometry</h3>
        <div id="geometry-metrics"></div>
      </div>

      <div class="metric-card">
        <h3>Rendering</h3>
        <div id="rendering-metrics"></div>
      </div>

      <div class="metric-card">
        <h3>Memory</h3>
        <div id="memory-metrics"></div>
      </div>
    </div>

    <div class="section" id="bottlenecks-section">
      <h3>Performance Bottlenecks</h3>
      <ul id="bottlenecks-list"></ul>
    </div>

    <div class="section" id="suggestions-section">
      <h3>Optimization Suggestions</h3>
      <ul id="suggestions-list"></ul>

      <table id="heavy-nodes-table">
        <thead>
          <tr>
            <th>Name</th>
            <th>Type</th>
            <th>Triangles</th>
            <th>Path</th>
          </tr>
        </thead>
        <tbody></tbody>
      </table>
    </div>
  </div>
  <script src="${scriptUri}"></script>
</body>
</html>`;
  }

  dispose(): void {
    if (this.panel) {
      this.panel.dispose();
    }
  }
}
```
