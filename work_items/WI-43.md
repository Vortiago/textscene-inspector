# WI-43: Performance Optimization for Large Scenes

## Overview

Implement comprehensive performance optimizations to handle production-scale Godot scenes with thousands of nodes efficiently. Focus on rendering performance, UI responsiveness, memory management, and progressive loading strategies to ensure smooth experiences even with complex scenes.

### Virtual Scrolling for Scene Tree

```typescript
// virtualSceneTree.ts
export interface VirtualScrollConfig {
  itemHeight: number;
  overscan: number;
  containerHeight: number;
}

export class VirtualSceneTree {
  private visibleRange = { start: 0, end: 0 };
  private scrollTop = 0;
  private flattenedNodes: TscnNode[] = [];

  constructor(
    private container: HTMLElement,
    private config: VirtualScrollConfig
  ) {
    this.setupScrollListener();
  }

  setNodes(rootNode: TscnNode): void {
    this.flattenedNodes = this.flattenTree(rootNode);
    this.updateVisibleRange();
    this.render();
  }

  private flattenTree(node: TscnNode, depth = 0): FlatNode[] {
    const result: FlatNode[] = [{ node, depth, expanded: true }];

    if (node.children) {
      for (const child of node.children) {
        result.push(...this.flattenTree(child, depth + 1));
      }
    }

    return result;
  }

  private setupScrollListener(): void {
    this.container.addEventListener('scroll', () => {
      this.scrollTop = this.container.scrollTop;
      this.updateVisibleRange();
      this.render();
    });
  }

  private updateVisibleRange(): void {
    const { itemHeight, overscan, containerHeight } = this.config;

    const start = Math.max(
      0,
      Math.floor(this.scrollTop / itemHeight) - overscan
    );

    const visibleCount = Math.ceil(containerHeight / itemHeight);
    const end = Math.min(
      this.flattenedNodes.length,
      start + visibleCount + overscan * 2
    );

    this.visibleRange = { start, end };
  }

  private render(): void {
    const { itemHeight } = this.config;
    const { start, end } = this.visibleRange;

    // Set total height for scrollbar
    const totalHeight = this.flattenedNodes.length * itemHeight;
    this.container.style.height = `${totalHeight}px`;

    // Clear existing content
    const viewport = this.container.querySelector('.viewport');
    if (!viewport) return;

    viewport.innerHTML = '';

    // Render only visible items
    for (let i = start; i < end; i++) {
      const flatNode = this.flattenedNodes[i];
      const item = this.createTreeItem(flatNode, i);
      viewport.appendChild(item);
    }

    // Position viewport
    viewport.style.transform = `translateY(${start * itemHeight}px)`;
  }

  private createTreeItem(flatNode: FlatNode, index: number): HTMLElement {
    const div = document.createElement('div');
    div.className = 'tree-item';
    div.style.height = `${this.config.itemHeight}px`;
    div.style.paddingLeft = `${flatNode.depth * 20}px`;
    div.dataset.index = index.toString();

    const hasChildren = flatNode.node.children && flatNode.node.children.length > 0;

    div.innerHTML = `
      ${hasChildren ? '<span class="expand-icon">▶</span>' : '<span class="spacer"></span>'}
      <span class="node-name">${flatNode.node.name}</span>
      <span class="node-type">${flatNode.node.type}</span>
    `;

    return div;
  }

  toggleExpand(index: number): void {
    const flatNode = this.flattenedNodes[index];
    flatNode.expanded = !flatNode.expanded;

    // Rebuild flattened tree
    this.flattenedNodes = this.rebuildFlattenedTree();
    this.updateVisibleRange();
    this.render();
  }

  private rebuildFlattenedTree(): FlatNode[] {
    // Rebuild tree considering expanded/collapsed states
    const result: FlatNode[] = [];

    const traverse = (node: TscnNode, depth: number, parentExpanded: boolean) => {
      const isExpanded = this.isNodeExpanded(node);
      result.push({ node, depth, expanded: isExpanded });

      if (isExpanded && parentExpanded && node.children) {
        for (const child of node.children) {
          traverse(child, depth + 1, true);
        }
      }
    };

    traverse(this.rootNode, 0, true);
    return result;
  }
}

interface FlatNode {
  node: TscnNode;
  depth: number;
  expanded: boolean;
}
```

### Lazy Property Loading

```typescript
// lazyPropertyLoader.ts
export class LazyPropertyLoader {
  private propertyCache: Map<string, Map<string, any>> = new Map();
  private loadQueue: Set<string> = new Set();
  private isProcessing = false;

  async loadProperties(nodePath: string, node: TscnNode): Promise<Map<string, any>> {
    // Check cache first
    if (this.propertyCache.has(nodePath)) {
      return this.propertyCache.get(nodePath)!;
    }

    // Add to queue
    this.loadQueue.add(nodePath);

    // Process queue
    if (!this.isProcessing) {
      await this.processQueue();
    }

    return this.propertyCache.get(nodePath) || new Map();
  }

  private async processQueue(): Promise<void> {
    this.isProcessing = true;

    while (this.loadQueue.size > 0) {
      const batch = Array.from(this.loadQueue).slice(0, 10);
      this.loadQueue.clear();

      await Promise.all(
        batch.map(nodePath => this.loadNodeProperties(nodePath))
      );

      // Allow UI to update
      await this.nextFrame();
    }

    this.isProcessing = false;
  }

  private async loadNodeProperties(nodePath: string): Promise<void> {
    const node = this.findNodeByPath(nodePath);
    if (!node) return;

    const properties = new Map<string, any>();

    for (const [key, value] of node.properties) {
      // Parse complex properties on-demand
      properties.set(key, this.parsePropertyValue(value));
    }

    this.propertyCache.set(nodePath, properties);
  }

  private parsePropertyValue(value: string): any {
    // Lazy parsing of complex values
    if (value.startsWith('Vector')) {
      return this.parseVector(value);
    } else if (value.startsWith('Transform3D')) {
      return this.parseTransform(value);
    }

    return value;
  }

  private nextFrame(): Promise<void> {
    return new Promise(resolve => requestAnimationFrame(() => resolve()));
  }

  clearCache(): void {
    this.propertyCache.clear();
  }

  getCacheSize(): number {
    return this.propertyCache.size;
  }
}
```

### Frustum Culling and LOD

```typescript
// sceneOptimizer.ts
import * as THREE from 'three';

export interface OptimizationMetrics {
  totalObjects: number;
  visibleObjects: number;
  culledObjects: number;
  drawCalls: number;
  triangles: number;
  fps: number;
}

export class SceneOptimizer {
  private frustum = new THREE.Frustum();
  private projScreenMatrix = new THREE.Matrix4();
  private lodLevels: Map<string, THREE.LOD> = new Map();

  constructor(
    private scene: THREE.Scene,
    private camera: THREE.Camera,
    private renderer: THREE.WebGLRenderer
  ) {}

  optimizeScene(): void {
    this.applyFrustumCulling();
    this.applyLOD();
    this.mergeSimilarGeometry();
    this.enableInstancedMeshes();
  }

  private applyFrustumCulling(): void {
    // Update frustum from camera
    this.camera.updateMatrixWorld();
    this.projScreenMatrix.multiplyMatrices(
      this.camera.projectionMatrix,
      this.camera.matrixWorldInverse
    );
    this.frustum.setFromProjectionMatrix(this.projScreenMatrix);

    // Cull objects outside frustum
    this.scene.traverse(object => {
      if (object instanceof THREE.Mesh) {
        // Compute bounding sphere if not present
        if (!object.geometry.boundingSphere) {
          object.geometry.computeBoundingSphere();
        }

        const boundingSphere = object.geometry.boundingSphere!;
        const worldCenter = new THREE.Vector3();
        object.localToWorld(worldCenter.copy(boundingSphere.center));

        const sphere = new THREE.Sphere(worldCenter, boundingSphere.radius);
        object.visible = this.frustum.intersectsSphere(sphere);
      }
    });
  }

  private applyLOD(): void {
    // Create LOD groups for complex meshes
    this.scene.traverse(object => {
      if (object instanceof THREE.Mesh && object.geometry.attributes.position.count > 1000) {
        const lod = this.createLODGroup(object);
        if (lod) {
          // Replace original mesh with LOD group
          object.parent?.add(lod);
          object.parent?.remove(object);
          this.lodLevels.set(object.name, lod);
        }
      }
    });
  }

  private createLODGroup(mesh: THREE.Mesh): THREE.LOD | null {
    const lod = new THREE.LOD();
    lod.position.copy(mesh.position);
    lod.rotation.copy(mesh.rotation);
    lod.scale.copy(mesh.scale);

    // Level 0: Full detail
    lod.addLevel(mesh.clone(), 0);

    // Level 1: Medium detail (simplified geometry)
    const mediumMesh = this.simplifyGeometry(mesh, 0.5);
    if (mediumMesh) {
      lod.addLevel(mediumMesh, 50);
    }

    // Level 2: Low detail
    const lowMesh = this.simplifyGeometry(mesh, 0.25);
    if (lowMesh) {
      lod.addLevel(lowMesh, 100);
    }

    return lod;
  }

  private simplifyGeometry(mesh: THREE.Mesh, ratio: number): THREE.Mesh | null {
    // Simple decimation - keep every Nth vertex
    const geometry = mesh.geometry;
    const positions = geometry.attributes.position.array;
    const normals = geometry.attributes.normal?.array;

    const targetCount = Math.floor(positions.length * ratio);
    const step = Math.ceil(1 / ratio);

    const newPositions: number[] = [];
    const newNormals: number[] = [];

    for (let i = 0; i < positions.length; i += step * 3) {
      newPositions.push(positions[i], positions[i + 1], positions[i + 2]);
      if (normals) {
        newNormals.push(normals[i], normals[i + 1], normals[i + 2]);
      }
    }

    const simplifiedGeometry = new THREE.BufferGeometry();
    simplifiedGeometry.setAttribute(
      'position',
      new THREE.Float32BufferAttribute(newPositions, 3)
    );

    if (newNormals.length > 0) {
      simplifiedGeometry.setAttribute(
        'normal',
        new THREE.Float32BufferAttribute(newNormals, 3)
      );
    }

    return new THREE.Mesh(simplifiedGeometry, mesh.material);
  }

  private mergeSimilarGeometry(): void {
    // Group meshes by material
    const geometryGroups = new Map<THREE.Material, THREE.BufferGeometry[]>();

    this.scene.traverse(object => {
      if (object instanceof THREE.Mesh && object.userData.mergeable) {
        const material = object.material as THREE.Material;
        if (!geometryGroups.has(material)) {
          geometryGroups.set(material, []);
        }
        geometryGroups.get(material)!.push(object.geometry);
      }
    });

    // Merge geometries with same material
    for (const [material, geometries] of geometryGroups) {
      if (geometries.length > 1) {
        const mergedGeometry = this.mergeGeometries(geometries);
        const mergedMesh = new THREE.Mesh(mergedGeometry, material);
        this.scene.add(mergedMesh);

        // Remove original meshes
        this.scene.traverse(object => {
          if (object instanceof THREE.Mesh && object.material === material) {
            object.parent?.remove(object);
          }
        });
      }
    }
  }

  private mergeGeometries(geometries: THREE.BufferGeometry[]): THREE.BufferGeometry {
    // Use BufferGeometryUtils for efficient merging
    const merged = new THREE.BufferGeometry();

    const positions: number[] = [];
    const normals: number[] = [];
    const uvs: number[] = [];

    for (const geometry of geometries) {
      const pos = geometry.attributes.position.array;
      const norm = geometry.attributes.normal?.array;
      const uv = geometry.attributes.uv?.array;

      positions.push(...Array.from(pos));
      if (norm) normals.push(...Array.from(norm));
      if (uv) uvs.push(...Array.from(uv));
    }

    merged.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    if (normals.length > 0) {
      merged.setAttribute('normal', new THREE.Float32BufferAttribute(normals, 3));
    }
    if (uvs.length > 0) {
      merged.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
    }

    return merged;
  }

  private enableInstancedMeshes(): void {
    // Find duplicate meshes and convert to instanced rendering
    const meshMap = new Map<string, THREE.Mesh[]>();

    this.scene.traverse(object => {
      if (object instanceof THREE.Mesh && object.userData.instanceable) {
        const key = this.getMeshSignature(object);
        if (!meshMap.has(key)) {
          meshMap.set(key, []);
        }
        meshMap.get(key)!.push(object);
      }
    });

    // Create instanced meshes for duplicates
    for (const [key, meshes] of meshMap) {
      if (meshes.length > 10) {
        this.createInstancedMesh(meshes);
      }
    }
  }

  private getMeshSignature(mesh: THREE.Mesh): string {
    return `${mesh.geometry.uuid}_${(mesh.material as THREE.Material).uuid}`;
  }

  private createInstancedMesh(meshes: THREE.Mesh[]): void {
    const reference = meshes[0];
    const count = meshes.length;

    const instancedMesh = new THREE.InstancedMesh(
      reference.geometry,
      reference.material,
      count
    );

    // Set transforms for each instance
    const matrix = new THREE.Matrix4();
    meshes.forEach((mesh, index) => {
      mesh.updateMatrix();
      instancedMesh.setMatrixAt(index, mesh.matrix);
      mesh.parent?.remove(mesh);
    });

    this.scene.add(instancedMesh);
  }

  getMetrics(): OptimizationMetrics {
    const info = this.renderer.info;

    let visibleObjects = 0;
    let culledObjects = 0;

    this.scene.traverse(object => {
      if (object instanceof THREE.Mesh) {
        if (object.visible) visibleObjects++;
        else culledObjects++;
      }
    });

    return {
      totalObjects: visibleObjects + culledObjects,
      visibleObjects,
      culledObjects,
      drawCalls: info.render.calls,
      triangles: info.render.triangles,
      fps: 0, // Updated by performance monitor
    };
  }
}
```

### Performance Monitor

```typescript
// performanceMonitor.ts
export interface PerformanceStats {
  fps: number;
  frameTime: number;
  memoryUsage: number;
  renderTime: number;
  updateTime: number;
}

export class PerformanceMonitor {
  private stats: PerformanceStats = {
    fps: 60,
    frameTime: 16.67,
    memoryUsage: 0,
    renderTime: 0,
    updateTime: 0,
  };

  private frameCount = 0;
  private lastTime = performance.now();
  private frameTimes: number[] = [];

  update(): void {
    const now = performance.now();
    const delta = now - this.lastTime;

    this.frameTimes.push(delta);
    if (this.frameTimes.length > 60) {
      this.frameTimes.shift();
    }

    this.frameCount++;

    if (delta >= 1000) {
      this.stats.fps = Math.round((this.frameCount * 1000) / delta);
      this.stats.frameTime = this.getAverageFrameTime();
      this.frameCount = 0;
      this.lastTime = now;
    }

    // Memory usage (if available)
    if ((performance as any).memory) {
      const memory = (performance as any).memory;
      this.stats.memoryUsage = memory.usedJSHeapSize / 1048576; // MB
    }
  }

  private getAverageFrameTime(): number {
    const sum = this.frameTimes.reduce((a, b) => a + b, 0);
    return sum / this.frameTimes.length;
  }

  getStats(): PerformanceStats {
    return { ...this.stats };
  }

  measureRenderTime(renderFn: () => void): void {
    const start = performance.now();
    renderFn();
    this.stats.renderTime = performance.now() - start;
  }

  measureUpdateTime(updateFn: () => void): void {
    const start = performance.now();
    updateFn();
    this.stats.updateTime = performance.now() - start;
  }

  isPerformanceDegraded(): boolean {
    return this.stats.fps < 30 || this.stats.frameTime > 33.33;
  }

  getRecommendations(): string[] {
    const recommendations: string[] = [];

    if (this.stats.fps < 30) {
      recommendations.push('Consider enabling "Simplify View" mode');
      recommendations.push('Reduce visible node count');
    }

    if (this.stats.memoryUsage > 500) {
      recommendations.push('High memory usage detected');
      recommendations.push('Clear property cache');
    }

    if (this.stats.renderTime > 16) {
      recommendations.push('Enable frustum culling');
      recommendations.push('Use LOD for distant objects');
    }

    return recommendations;
  }
}
```

```typescript
// progressiveSceneLoader.ts
export class ProgressiveSceneLoader {
  private loadedNodeCount = 0;
  private totalNodeCount = 0;

  async loadScene(
    sceneData: TscnData,
    onProgress?: (progress: number) => void
  ): Promise<void> {
    this.totalNodeCount = this.countNodes(sceneData.rootNode);
    this.loadedNodeCount = 0;

    await this.loadNodesInBatches(sceneData.rootNode, 50, onProgress);
  }

  private async loadNodesInBatches(
    node: TscnNode,
    batchSize: number,
    onProgress?: (progress: number) => void
  ): Promise<void> {
    const queue: TscnNode[] = [node];

    while (queue.length > 0) {
      const batch = queue.splice(0, batchSize);

      for (const currentNode of batch) {
        await this.loadNode(currentNode);
        this.loadedNodeCount++;

        if (currentNode.children) {
          queue.push(...currentNode.children);
        }

        if (onProgress) {
          const progress = this.loadedNodeCount / this.totalNodeCount;
          onProgress(progress);
        }
      }

      // Yield to UI
      await this.nextFrame();
    }
  }

  private async loadNode(node: TscnNode): Promise<void> {
    // Load node-specific resources
    if (node.type === 'MeshInstance3D') {
      await this.loadMeshResources(node);
    }
  }

  private async loadMeshResources(node: TscnNode): Promise<void> {
    // Lazy load mesh geometry and materials
    await Promise.resolve(); // Placeholder for async loading
  }

  private countNodes(node: TscnNode): number {
    let count = 1;
    if (node.children) {
      for (const child of node.children) {
        count += this.countNodes(child);
      }
    }
```
