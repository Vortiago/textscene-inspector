# WI-38: Cross-Reference Analysis

## Overview

Implement workspace-wide reference tracking for TSCN scenes, enabling developers to find all usages of a scene (as instances or in code), understand dependencies, and safely refactor scene hierarchies. Support "Find All References" functionality similar to code navigation, with both scene-to-scene and code-to-scene reference tracking.

## Implementation Requirements

From TODO.md Phase 10: Advanced Navigation:

- "Find All References" command for scenes
- Show all scenes using a specific scene as instance
- Show all C# scripts referencing a scene path
- Workspace-wide dependency graph
- Support safe refactoring with usage information
- Track both ExtResource references and code references
- Generate visual dependency graphs
- Provide quick navigation to references

## Implementation Notes

### Cross-Reference Tracker

```typescript
// crossReferenceTracker.ts
import * as vscode from 'vscode';
import * as path from 'path';

export interface SceneReference {
  sourceUri: vscode.Uri;
  sourceType: 'scene' | 'code';
  referencePath: string;
  location: vscode.Location;
  context?: string;
}

export interface DependencyNode {
  scenePath: string;
  referencedBy: SceneReference[];
  references: string[];
  type: 'scene' | 'script' | 'resource';
}

export class CrossReferenceTracker {
  private sceneIndex: Map<string, DependencyNode> = new Map();
  private referenceCache: Map<string, SceneReference[]> = new Map();

  async indexWorkspace(): Promise<void> {
    const workspaceFolders = vscode.workspace.workspaceFolders;
    if (!workspaceFolders) return;

    // Find all .tscn files
    const tscnFiles = await vscode.workspace.findFiles(
      '**/*.tscn',
      '**/node_modules/**'
    );

    // Find all .cs files
    const csFiles = await vscode.workspace.findFiles(
      '**/*.cs',
      '**/node_modules/**'
    );

    // Index scene-to-scene references
    await Promise.all(
      tscnFiles.map(uri => this.indexSceneReferences(uri))
    );

    // Index code-to-scene references
    await Promise.all(
      csFiles.map(uri => this.indexCodeReferences(uri))
    );
  }

  private async indexSceneReferences(uri: vscode.Uri): Promise<void> {
    const content = await vscode.workspace.fs.readFile(uri);
    const text = Buffer.from(content).toString('utf8');

    const scenePath = this.normalizeScenePath(uri);
    const node = this.getOrCreateNode(scenePath);

    // Parse ExtResource references
    const extResourceRegex = /\[ext_resource[^\]]*path="([^"]+\.tscn)"[^\]]*\]/g;
    let match;

    while ((match = extResourceRegex.exec(text)) !== null) {
      const referencedPath = match[1];
      const absolutePath = this.resolveRelativePath(uri, referencedPath);

      node.references.push(absolutePath);

      // Add reverse reference
      const referencedNode = this.getOrCreateNode(absolutePath);
      const location = this.createLocation(uri, text, match.index);

      referencedNode.referencedBy.push({
        sourceUri: uri,
        sourceType: 'scene',
        referencePath: scenePath,
        location,
        context: this.extractContext(text, match.index),
      });
    }

    // Parse instance references
    const instanceRegex = /instance=ExtResource\("([^"]+)"\)/g;
    while ((match = instanceRegex.exec(text)) !== null) {
      // Link to external resource by ID
      const resourceId = match[1];
      const referencedPath = this.resolveResourceId(text, resourceId);

      if (referencedPath) {
        const absolutePath = this.resolveRelativePath(uri, referencedPath);
        const referencedNode = this.getOrCreateNode(absolutePath);
        const location = this.createLocation(uri, text, match.index);

        referencedNode.referencedBy.push({
          sourceUri: uri,
          sourceType: 'scene',
          referencePath: scenePath,
          location,
          context: this.extractContext(text, match.index),
        });
      }
    }
  }

  private async indexCodeReferences(uri: vscode.Uri): Promise<void> {
    const content = await vscode.workspace.fs.readFile(uri);
    const text = Buffer.from(content).toString('utf8');

    // Find scene path references in C# code
    const patterns = [
      // GD.Load<PackedScene>("res://...")
      /GD\.Load<PackedScene>\("(res:\/\/[^"]+\.tscn)"\)/g,
      // ResourceLoader.Load<PackedScene>("res://...")
      /ResourceLoader\.Load<PackedScene>\("(res:\/\/[^"]+\.tscn)"\)/g,
      // [SceneTree("res://...")]
      /\[SceneTree\("(res:\/\/[^"]+\.tscn)"\)\]/g,
      // Any string literal containing .tscn
      /"(res:\/\/[^"]+\.tscn)"/g,
    ];

    for (const pattern of patterns) {
      let match;
      while ((match = pattern.exec(text)) !== null) {
        const scenePath = match[1];
        const absolutePath = this.godotPathToAbsolute(scenePath);

        const node = this.getOrCreateNode(absolutePath);
        const location = this.createLocation(uri, text, match.index);

        node.referencedBy.push({
          sourceUri: uri,
          sourceType: 'code',
          referencePath: uri.fsPath,
          location,
          context: this.extractContext(text, match.index),
        });
      }
    }
  }

  findReferences(scenePath: string): SceneReference[] {
    const normalizedPath = this.normalizeScenePath(
      vscode.Uri.file(scenePath)
    );
    const node = this.sceneIndex.get(normalizedPath);
    return node ? node.referencedBy : [];
  }

  findDependencies(scenePath: string): string[] {
    const normalizedPath = this.normalizeScenePath(
      vscode.Uri.file(scenePath)
    );
    const node = this.sceneIndex.get(normalizedPath);
    return node ? node.references : [];
  }

  generateDependencyGraph(): DependencyGraph {
    const nodes: GraphNode[] = [];
    const edges: GraphEdge[] = [];

    for (const [scenePath, depNode] of this.sceneIndex) {
      nodes.push({
        id: scenePath,
        label: path.basename(scenePath),
        type: depNode.type,
        referenceCount: depNode.referencedBy.length,
      });

      depNode.references.forEach(ref => {
        edges.push({
          from: scenePath,
          to: ref,
          type: 'dependency',
        });
      });
    }

    return { nodes, edges };
  }

  private getOrCreateNode(scenePath: string): DependencyNode {
    if (!this.sceneIndex.has(scenePath)) {
      this.sceneIndex.set(scenePath, {
        scenePath,
        referencedBy: [],
        references: [],
        type: scenePath.endsWith('.tscn') ? 'scene' : 'script',
      });
    }
    return this.sceneIndex.get(scenePath)!;
  }

  private normalizeScenePath(uri: vscode.Uri): string {
    const workspaceFolder = vscode.workspace.getWorkspaceFolder(uri);
    if (workspaceFolder) {
      return path.relative(workspaceFolder.uri.fsPath, uri.fsPath);
    }
    return uri.fsPath;
  }

  private resolveRelativePath(baseUri: vscode.Uri, relativePath: string): string {
    if (relativePath.startsWith('res://')) {
      return this.godotPathToAbsolute(relativePath);
    }

    const baseDir = path.dirname(baseUri.fsPath);
    const absolute = path.resolve(baseDir, relativePath);
    return this.normalizeScenePath(vscode.Uri.file(absolute));
  }

  private godotPathToAbsolute(godotPath: string): string {
    const workspaceFolders = vscode.workspace.workspaceFolders;
    if (!workspaceFolders) return godotPath;

    // res:// maps to workspace root
    const relativePath = godotPath.replace('res://', '');
    const absolutePath = path.join(
      workspaceFolders[0].uri.fsPath,
      relativePath
    );

    return this.normalizeScenePath(vscode.Uri.file(absolutePath));
  }

  private resolveResourceId(tscnContent: string, resourceId: string): string | null {
    const regex = new RegExp(
      `\\[ext_resource[^\\]]*id="${resourceId}"[^\\]]*path="([^"]+)"[^\\]]*\\]`
    );
    const match = regex.exec(tscnContent);
    return match ? match[1] : null;
  }

  private createLocation(
    uri: vscode.Uri,
    text: string,
    offset: number
  ): vscode.Location {
    const lines = text.substring(0, offset).split('\n');
    const line = lines.length - 1;
    const character = lines[lines.length - 1].length;

    return new vscode.Location(
      uri,
      new vscode.Position(line, character)
    );
  }

  private extractContext(text: string, offset: number): string {
    const lines = text.split('\n');
    const textBeforeMatch = text.substring(0, offset);
    const lineNumber = textBeforeMatch.split('\n').length - 1;

    const contextLines = lines.slice(
      Math.max(0, lineNumber - 1),
      Math.min(lines.length, lineNumber + 2)
    );

    return contextLines.join('\n').trim();
  }
}

interface DependencyGraph {
  nodes: GraphNode[];
  edges: GraphEdge[];
}

interface GraphNode {
  id: string;
  label: string;
  type: 'scene' | 'script' | 'resource';
  referenceCount: number;
}

interface GraphEdge {
  from: string;
  to: string;
  type: 'dependency' | 'reference';
}
```

### Reference Provider

```typescript
// sceneReferenceProvider.ts
import * as vscode from 'vscode';
import { CrossReferenceTracker } from './crossReferenceTracker';

export class SceneReferenceProvider implements vscode.ReferenceProvider {
  constructor(private tracker: CrossReferenceTracker) {}

  provideReferences(
    document: vscode.TextDocument,
    position: vscode.Position,
    context: vscode.ReferenceContext,
    token: vscode.CancellationToken
  ): vscode.ProviderResult<vscode.Location[]> {
    // Check if we're in a .tscn file
    if (document.languageId !== 'tscn') {
      return [];
    }

    // Get scene path
    const scenePath = document.uri.fsPath;
    const references = this.tracker.findReferences(scenePath);

    return references.map(ref => ref.location);
  }
}
```

### Commands Integration

```typescript
// referenceCommands.ts
import * as vscode from 'vscode';
import { CrossReferenceTracker, SceneReference } from './crossReferenceTracker';

export class ReferenceCommands {
  constructor(private tracker: CrossReferenceTracker) {}

  registerCommands(context: vscode.ExtensionContext): void {
    // Find all references command
    context.subscriptions.push(
      vscode.commands.registerCommand(
        'tscn.findAllReferences',
        async (uri?: vscode.Uri) => {
          await this.findAllReferences(uri);
        }
      )
    );

    // Show dependency graph
    context.subscriptions.push(
      vscode.commands.registerCommand(
        'tscn.showDependencyGraph',
        async () => {
          await this.showDependencyGraph();
        }
      )
    );

    // Find scene instances
    context.subscriptions.push(
      vscode.commands.registerCommand(
        'tscn.findSceneInstances',
        async (uri?: vscode.Uri) => {
          await this.findSceneInstances(uri);
        }
      )
    );

    // Rebuild reference index
    context.subscriptions.push(
      vscode.commands.registerCommand(
        'tscn.rebuildReferenceIndex',
        async () => {
          await this.rebuildIndex();
        }
      )
    );
  }

  private async findAllReferences(uri?: vscode.Uri): Promise<void> {
    const targetUri = uri || vscode.window.activeTextEditor?.document.uri;
    if (!targetUri) {
      vscode.window.showErrorMessage('No file selected');
      return;
    }

    const references = this.tracker.findReferences(targetUri.fsPath);

    if (references.length === 0) {
      vscode.window.showInformationMessage(
        `No references found for ${path.basename(targetUri.fsPath)}`
      );
      return;
    }

    // Show in references view
    const locations = references.map(ref => ref.location);
    await vscode.commands.executeCommand(
      'editor.action.showReferences',
      targetUri,
      new vscode.Position(0, 0),
      locations
    );
  }

  private async findSceneInstances(uri?: vscode.Uri): Promise<void> {
    const targetUri = uri || vscode.window.activeTextEditor?.document.uri;
    if (!targetUri) return;

    const references = this.tracker.findReferences(targetUri.fsPath);
    const sceneReferences = references.filter(
      ref => ref.sourceType === 'scene'
    );

    if (sceneReferences.length === 0) {
      vscode.window.showInformationMessage('Scene is not instanced anywhere');
      return;
    }

    // Show quick pick
    const items = sceneReferences.map(ref => ({
      label: path.basename(ref.sourceUri.fsPath),
      description: ref.sourceUri.fsPath,
      detail: ref.context,
      uri: ref.sourceUri,
      location: ref.location,
    }));

    const selected = await vscode.window.showQuickPick(items, {
      placeHolder: `Found ${sceneReferences.length} scene instance(s)`,
    });

    if (selected) {
      await vscode.window.showTextDocument(selected.uri, {
        selection: selected.location.range,
      });
    }
  }

  private async showDependencyGraph(): Promise<void> {
    const graph = this.tracker.generateDependencyGraph();

    const panel = vscode.window.createWebviewPanel(
      'tscnDependencyGraph',
      'Scene Dependency Graph',
      vscode.ViewColumn.Two,
      {
        enableScripts: true,
      }
    );

    panel.webview.html = this.getDependencyGraphHtml(graph);
  }

  private getDependencyGraphHtml(graph: DependencyGraph): string {
    // Generate DOT format for visualization
    let dot = 'digraph SceneDependencies {\n';
    dot += '  rankdir=LR;\n';
    dot += '  node [shape=box, style=rounded];\n\n';

    // Add nodes with styling based on type
    graph.nodes.forEach(node => {
      const color = node.type === 'scene' ? 'lightblue' : 'lightgreen';
      const label = `${node.label}\\n(${node.referenceCount} refs)`;
      dot += `  "${node.id}" [label="${label}", fillcolor="${color}", style="filled,rounded"];\n`;
    });

    dot += '\n';

    // Add edges
    graph.edges.forEach(edge => {
      dot += `  "${edge.from}" -> "${edge.to}";\n`;
    });

    dot += '}\n';

    return `
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Scene Dependency Graph</title>
  <script src="https://unpkg.com/viz.js@2.1.2/viz.js"></script>
  <script src="https://unpkg.com/viz.js@2.1.2/full.render.js"></script>
  <style>
    body {
      margin: 0;
      padding: 20px;
      font-family: var(--vscode-font-family);
      background-color: var(--vscode-editor-background);
      color: var(--vscode-editor-foreground);
    }
    #graph {
      width: 100%;
      height: calc(100vh - 40px);
      overflow: auto;
      border: 1px solid var(--vscode-panel-border);
    }
    #graph svg {
      max-width: 100%;
      height: auto;
    }
  </style>
</head>
<body>
  <div id="graph"></div>
  <script>
    const dot = ${JSON.stringify(dot)};

    const viz = new Viz();
    viz.renderSVGElement(dot)
      .then(element => {
        document.getElementById('graph').appendChild(element);
      })
      .catch(error => {
        console.error('Error rendering graph:', error);
        document.getElementById('graph').innerHTML =
          '<pre>' + error + '</pre>';
      });
  </script>
</body>
</html>`;
  }

  private async rebuildIndex(): Promise<void> {
    await vscode.window.withProgress(
      {
        location: vscode.ProgressLocation.Notification,
        title: 'Rebuilding reference index',
        cancellable: false,
      },
      async progress => {
        progress.report({ increment: 0 });
        await this.tracker.indexWorkspace();
        progress.report({ increment: 100 });
      }
    );

    vscode.window.showInformationMessage('Reference index rebuilt');
  }
}
```

### File Watcher for Auto-Indexing

```typescript
// referenceIndexWatcher.ts
import * as vscode from 'vscode';
import { CrossReferenceTracker } from './crossReferenceTracker';

export class ReferenceIndexWatcher {
  private watchers: vscode.FileSystemWatcher[] = [];

  constructor(private tracker: CrossReferenceTracker) {}

  activate(context: vscode.ExtensionContext): void {
    // Watch .tscn files
    const tscnWatcher = vscode.workspace.createFileSystemWatcher(
      '**/*.tscn'
    );

    tscnWatcher.onDidCreate(uri => this.onFileChanged(uri));
    tscnWatcher.onDidChange(uri => this.onFileChanged(uri));
    tscnWatcher.onDidDelete(uri => this.onFileDeleted(uri));

    this.watchers.push(tscnWatcher);

    // Watch .cs files
    const csWatcher = vscode.workspace.createFileSystemWatcher('**/*.cs');

    csWatcher.onDidCreate(uri => this.onFileChanged(uri));
    csWatcher.onDidChange(uri => this.onFileChanged(uri));
    csWatcher.onDidDelete(uri => this.onFileDeleted(uri));

    this.watchers.push(csWatcher);

    context.subscriptions.push(...this.watchers);
  }

  private async onFileChanged(uri: vscode.Uri): Promise<void> {
    // Debounce to avoid excessive re-indexing
    if (uri.fsPath.endsWith('.tscn')) {
      await this.tracker.indexSceneReferences(uri);
    } else if (uri.fsPath.endsWith('.cs')) {
      await this.tracker.indexCodeReferences(uri);
    }
  }

  private onFileDeleted(uri: vscode.Uri): void {
    // Remove from index
    const normalizedPath = this.tracker['normalizeScenePath'](uri);
    this.tracker['sceneIndex'].delete(normalizedPath);
  }
}
```

## Performance Considerations

### Indexing Strategy

1. **Initial Index**: Build full index on extension activation
2. **Incremental Updates**: Re-index only changed files via file watchers
3. **Lazy Loading**: Load reference details on-demand
4. **Caching**: Cache parsed references in memory

### Optimization Techniques

```typescript
// Debounced re-indexing
class DebouncedIndexer {
  private pending = new Map<string, NodeJS.Timeout>();

  scheduleReindex(uri: vscode.Uri, delay = 500): void {
    const key = uri.toString();

    if (this.pending.has(key)) {
      clearTimeout(this.pending.get(key)!);
    }

    this.pending.set(
      key,
      setTimeout(() => {
        this.tracker.indexFile(uri);
        this.pending.delete(key);
      }, delay)
    );
  }
}
```
