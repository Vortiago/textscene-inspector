# WI-28: C# Class → Scene Type Matching

## Overview

Implement type validation for Godot GetNode<T>() calls, ensuring the generic type parameter matches the actual node type in the scene. This feature prevents runtime type cast errors by validating at development time that a node retrieved with GetNode<T>() is actually of type T in the scene file.

## Implementation Requirements

From TODO.md Phase 7: C# Integration:

- Extract generic type parameter from GetNode<T>() calls
- Determine actual node type from scene file
- Compare C# type with Godot node type
- Support Godot-to-C# type mapping (Node3D → Node3D, Label → Label, etc.)
- Handle inheritance (Node is base of all node types)
- Create diagnostics for type mismatches
- Suggest correct type in error message
- Support custom C# classes that inherit from Godot types
- Handle script-attached nodes (nodes with custom C# scripts)

### Custom Scripts

Nodes with attached C# scripts have dual typing:

```ini
[node name="Player" type="CharacterBody3D" parent="."]
script = ExtResource("player.cs")
```

```csharp
// player.cs
public partial class Player : CharacterBody3D
{
    // Custom player logic
}
```

Valid GetNode calls:
```csharp
GetNode<Player>("Player");         // ✓ Script type
GetNode<CharacterBody3D>("Player"); // ✓ Base Godot type
GetNode<Node3D>("Player");          // ✓ Ancestor type
GetNode<Label>("Player");           // ✗ Unrelated type
```

## Implementation Notes

### Type Extraction

**From C# GetNode Call:**

```typescript
interface GetNodeCallInfo {
  genericType: string;  // Type parameter
  path: string;         // Node path
  range: vscode.Range;  // Location in file
}

function extractGetNodeCalls(
  document: vscode.TextDocument
): GetNodeCallInfo[] {
  const regex = /GetNode<(\w+)>\s*\(\s*"([^"]+)"\s*\)/g;
  const calls: GetNodeCallInfo[] = [];

  const text = document.getText();
  let match;

  while ((match = regex.exec(text)) !== null) {
    const position = document.positionAt(match.index);
    const endPosition = document.positionAt(match.index + match[0].length);

    calls.push({
      genericType: match[1],
      path: match[2],
      range: new vscode.Range(position, endPosition),
    });
  }

  return calls;
}
```

**From Scene File:**

```typescript
interface NodeTypeInfo {
  nodeName: string;
  godotType: string;      // Built-in type (e.g., "Node3D")
  scriptType?: string;    // C# script type (e.g., "Player")
  scriptPath?: string;    // Path to .cs file
}

function getNodeTypeInfo(
  node: TscnNode,
  resourceRegistry: ResourceRegistry
): NodeTypeInfo {
  const info: NodeTypeInfo = {
    nodeName: node.name,
    godotType: node.type,
  };

  // Check for attached script
  if (node.properties.script) {
    const scriptId = parseExtResourceId(node.properties.script);
    const scriptResource = resourceRegistry.get(scriptId);

    if (scriptResource) {
      info.scriptPath = scriptResource.path;
      info.scriptType = extractClassNameFromScript(scriptResource.path);
    }
  }

  return info;
}
```

### Type Compatibility Checking

**Basic Compatibility:**

```typescript
function isTypeCompatible(
  requestedType: string,
  actualNodeType: NodeTypeInfo
): { compatible: boolean; reason?: string } {
  // Exact match with script type
  if (actualNodeType.scriptType === requestedType) {
    return { compatible: true };
  }

  // Exact match with Godot type
  if (actualNodeType.godotType === requestedType) {
    return { compatible: true };
  }

  // Check inheritance
  if (isInheritanceCompatible(requestedType, actualNodeType.godotType)) {
    return { compatible: true };
  }

  return {
    compatible: false,
    reason: `Type mismatch: requested '${requestedType}' but node is '${actualNodeType.godotType}'${
      actualNodeType.scriptType ? ` with script '${actualNodeType.scriptType}'` : ''
    }`,
  };
}
```

**Inheritance Checking:**

```typescript
const godotTypeHierarchy: Record<string, string[]> = {
  Node: ['Object'],
  Node2D: ['Node', 'Object'],
  Node3D: ['Node', 'Object'],
  Control: ['CanvasItem', 'Node2D', 'Node', 'Object'],
  Label: ['Control', 'CanvasItem', 'Node2D', 'Node', 'Object'],
  MeshInstance3D: ['GeometryInstance3D', 'VisualInstance3D', 'Node3D', 'Node', 'Object'],
  Camera3D: ['Node3D', 'Node', 'Object'],
  Light3D: ['VisualInstance3D', 'Node3D', 'Node', 'Object'],
  DirectionalLight3D: ['Light3D', 'VisualInstance3D', 'Node3D', 'Node', 'Object'],
  SpotLight3D: ['Light3D', 'VisualInstance3D', 'Node3D', 'Node', 'Object'],
  // ... add more as needed
};

function isInheritanceCompatible(
  requestedType: string,
  actualType: string
): boolean {
  // Requested type is the actual type
  if (requestedType === actualType) {
    return true;
  }

  // Check if requested type is an ancestor of actual type
  const ancestors = godotTypeHierarchy[actualType] || [];
  return ancestors.includes(requestedType);
}
```

### Script Class Name Extraction

**From C# File:**

```typescript
async function extractClassNameFromScript(
  scriptPath: string
): Promise<string | null> {
  const uri = vscode.Uri.file(scriptPath);

  try {
    const document = await vscode.workspace.openTextDocument(uri);
    const text = document.getText();

    // Extract class name from C# code
    const classRegex = /public\s+partial\s+class\s+(\w+)\s*:\s*\w+/;
    const match = classRegex.exec(text);

    return match ? match[1] : null;
  } catch {
    return null;
  }
}
```

### Diagnostic Generation

**Type Mismatch Diagnostic:**

```typescript
function createTypeMismatchDiagnostic(
  range: vscode.Range,
  requestedType: string,
  actualType: NodeTypeInfo
): vscode.Diagnostic {
  const message = `Type mismatch: GetNode<${requestedType}>() called on node of type '${actualType.godotType}'${
    actualType.scriptType ? ` (script: ${actualType.scriptType})` : ''
  }`;

  const diagnostic = new vscode.Diagnostic(
    range,
    message,
    vscode.DiagnosticSeverity.Error
  );

  diagnostic.source = 'TSCN Type Checker';
  diagnostic.code = 'type-mismatch';

  // Suggest correct type
  const correctType = actualType.scriptType || actualType.godotType;
  diagnostic.message += `\n\nConsider using: GetNode<${correctType}>()`;

  return diagnostic;
}
```

## Code Examples

### Type Validator

```typescript
// typeValidator.ts
import * as vscode from 'vscode';
import type { TscnNode } from '@tscn-renderer/core';

export class GetNodeTypeValidator {
  private diagnosticCollection: vscode.DiagnosticCollection;
  private sceneCache: Map<string, any> = new Map();

  constructor(diagnosticCollection: vscode.DiagnosticCollection) {
    this.diagnosticCollection = diagnosticCollection;
  }

  async validateDocument(document: vscode.TextDocument): Promise<void> {
    if (!document.fileName.endsWith('.cs')) {
      return;
    }

    // Find scene file
    const sceneUri = await this.findSceneForScript(document.uri);
    if (!sceneUri) {
      this.diagnosticCollection.delete(document.uri);
      return;
    }

    // Load scene
    const scene = await this.loadScene(sceneUri);
    if (!scene) {
      return;
    }

    // Extract GetNode calls
    const getNodeCalls = this.extractGetNodeCalls(document);

    // Validate types
    const diagnostics: vscode.Diagnostic[] = [];

    for (const call of getNodeCalls) {
      // Find node in scene
      const node = this.findNodeByPath(call.path, scene.rootNode);
      if (!node) {
        // Path validation (handled by WI-26)
        continue;
      }

      // Get node type info
      const nodeType = await this.getNodeTypeInfo(node);

      // Check type compatibility
      const compatible = this.isTypeCompatible(call.genericType, nodeType);

      if (!compatible.compatible) {
        const diagnostic = this.createTypeMismatchDiagnostic(
          call.range,
          call.genericType,
          nodeType,
          compatible.reason
        );

        diagnostics.push(diagnostic);
      }
    }

    this.diagnosticCollection.set(document.uri, diagnostics);
  }

  private extractGetNodeCalls(
    document: vscode.TextDocument
  ): GetNodeCallInfo[] {
    const calls: GetNodeCallInfo[] = [];
    const text = document.getText();
    const regex = /GetNode<(\w+)>\s*\(\s*"([^"]+)"\s*\)/g;

    let match;
    while ((match = regex.exec(text)) !== null) {
      const position = document.positionAt(match.index);
      const endPosition = document.positionAt(match.index + match[0].length);

      calls.push({
        genericType: match[1],
        path: match[2],
        range: new vscode.Range(position, endPosition),
      });
    }

    return calls;
  }

  private async getNodeTypeInfo(node: TscnNode): Promise<NodeTypeInfo> {
    const info: NodeTypeInfo = {
      nodeName: node.name,
      godotType: node.type,
    };

    // Check for attached script
    if (node.properties.script) {
      const scriptId = this.parseExtResourceId(node.properties.script);
      // In a real implementation, resolve script from resource registry
      // For now, simplified:
      info.scriptType = 'CustomScriptType';  // TODO: Extract from actual script
    }

    return info;
  }

  private isTypeCompatible(
    requestedType: string,
    actualNodeType: NodeTypeInfo
  ): { compatible: boolean; reason?: string } {
    // Exact match with script type
    if (actualNodeType.scriptType === requestedType) {
      return { compatible: true };
    }

    // Exact match with Godot type
    if (actualNodeType.godotType === requestedType) {
      return { compatible: true };
    }

    // Check inheritance
    const ancestors = this.getTypeAncestors(actualNodeType.godotType);
    if (ancestors.includes(requestedType)) {
      return { compatible: true };
    }

    return {
      compatible: false,
      reason: `Type '${requestedType}' is not compatible with '${actualNodeType.godotType}'`,
    };
  }

  private getTypeAncestors(type: string): string[] {
    const hierarchy: Record<string, string[]> = {
      Node: ['Object'],
      Node2D: ['Node', 'Object'],
      Node3D: ['Node', 'Object'],
      MeshInstance3D: ['GeometryInstance3D', 'VisualInstance3D', 'Node3D', 'Node', 'Object'],
      Label: ['Control', 'CanvasItem', 'Node2D', 'Node', 'Object'],
      Camera3D: ['Node3D', 'Node', 'Object'],
      DirectionalLight3D: ['Light3D', 'VisualInstance3D', 'Node3D', 'Node', 'Object'],
    };

    return hierarchy[type] || [];
  }

  private createTypeMismatchDiagnostic(
    range: vscode.Range,
    requestedType: string,
    actualType: NodeTypeInfo,
    reason?: string
  ): vscode.Diagnostic {
    const correctType = actualType.scriptType || actualType.godotType;

    let message = `Type mismatch: GetNode<${requestedType}>() expects type '${requestedType}', `;
    message += `but node is of type '${actualType.godotType}'`;

    if (actualType.scriptType) {
      message += ` with script '${actualType.scriptType}'`;
    }

    message += `\n\nSuggestion: Use GetNode<${correctType}>() instead`;

    const diagnostic = new vscode.Diagnostic(
      range,
      message,
      vscode.DiagnosticSeverity.Error
    );

    diagnostic.source = 'TSCN Type Checker';
    diagnostic.code = 'getnode-type-mismatch';

    return diagnostic;
  }
```

## Testing Strategy

```typescript
describe('GetNode Type Validation', () => {
  it('should accept exact type match', () => {
    const nodeType: NodeTypeInfo = {
      nodeName: 'Camera',
      godotType: 'Camera3D',
    };

    const result = isTypeCompatible('Camera3D', nodeType);
    expect(result.compatible).toBe(true);
  });

  it('should accept ancestor type', () => {
    const nodeType: NodeTypeInfo = {
      nodeName: 'Camera',
      godotType: 'Camera3D',
    };

    const result = isTypeCompatible('Node3D', nodeType);  // Ancestor
    expect(result.compatible).toBe(true);
  });

  it('should reject incompatible type', () => {
    const nodeType: NodeTypeInfo = {
      nodeName: 'Camera',
      godotType: 'Camera3D',

    expect(result.compatible).toBe(false);
```
