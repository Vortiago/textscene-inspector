# WI-31: Undo/Redo Support

## Overview

Implement comprehensive undo/redo functionality for all interactive editing operations in the VS Code TSCN editor. This includes property changes, gizmo transformations, and any other modifications made through the custom editor UI. By integrating with VS Code's native undo/redo stack, users get familiar Ctrl+Z/Ctrl+Y keyboard shortcuts and consistent behavior across the editor.

## Implementation Requirements

From TODO.md Phase 8: Interactive Editing:

- Track all edits made through custom editor UI
- Integrate with VS Code's undo/redo stack
- Support keyboard shortcuts (Ctrl+Z, Ctrl+Y / Cmd+Z, Cmd+Shift+Z)
- Maintain edit history with descriptive labels
- Handle complex edits (multi-property changes)
- Sync undo/redo with preview updates
- Support edits from multiple sources (property panel, gizmo, text editor)
- Maintain document state consistency
- Clear undo stack when document closes
- Optional: Group related edits into single undo operation

**Undo Stack Management:**

VS Code automatically manages the undo stack:
```
onDidChangeCustomDocument(editA);  // Push A onto stack
onDidChangeCustomDocument(editB);  // Push B onto stack
onDidChangeCustomDocument(editC);  // Push C onto stack

// User presses Ctrl+Z
// → editC.undo() is called

// User presses Ctrl+Z again
// → editB.undo() is called

// User presses Ctrl+Y
// → editB.redo() is called
```

## Implementation Notes

### Edit History Manager

```typescript
// editHistoryManager.ts
import * as vscode from 'vscode';

export interface Edit {
  id: string;
  label: string;
  timestamp: number;
  undo: () => void | Promise<void>;
  redo: () => void | Promise<void>;
}

export interface EditGroup {
  label: string;
  edits: Edit[];
}

export class EditHistoryManager {
  private readonly _onDidChangeCustomDocument = new vscode.EventEmitter<
    vscode.CustomDocumentEditEvent<TscnDocument>
  >();
  readonly onDidChangeCustomDocument = this._onDidChangeCustomDocument.event;

  private pendingEditGroup: EditGroup | null = null;
  private editGroupTimeout?: NodeJS.Timeout;

  constructor(
    private readonly document: TscnDocument,
    private readonly updateCallback: (edit: Edit) => Promise<void>
  ) {}

  /**
   * Record a single edit operation
   */
  recordEdit(label: string, undo: () => void | Promise<void>, redo: () => void | Promise<void>): void {
    const edit: Edit = {
      id: this.generateEditId(),
      label,
      timestamp: Date.now(),
      undo,
      redo,
    };

    if (this.pendingEditGroup) {
      // Add to current group
      this.pendingEditGroup.edits.push(edit);
      this.resetGroupTimeout();
    } else {
      // Create standalone edit
      this.fireEditEvent(label, undo, redo);
    }
  }

  /**
   * Start grouping edits together (e.g., dragging gizmo)
   */
  startEditGroup(label: string): void {
    if (this.pendingEditGroup) {
      this.commitEditGroup();
    }

    this.pendingEditGroup = {
      label,
      edits: [],
    };
  }

  /**
   * Stop grouping and commit the group as a single undo operation
   */
  commitEditGroup(): void {
    if (!this.pendingEditGroup) return;

    const group = this.pendingEditGroup;
    this.pendingEditGroup = null;

    if (this.editGroupTimeout) {
      clearTimeout(this.editGroupTimeout);
      this.editGroupTimeout = undefined;
    }

    if (group.edits.length === 0) return;

    // Create composite undo/redo functions
    const undo = async () => {
      // Undo in reverse order
      for (let i = group.edits.length - 1; i >= 0; i--) {
        await group.edits[i].undo();
      }
    };

    const redo = async () => {
      // Redo in forward order
      for (const edit of group.edits) {
        await edit.redo();
      }
    };

    this.fireEditEvent(group.label, undo, redo);
  }

  private resetGroupTimeout(): void {
    if (this.editGroupTimeout) {
      clearTimeout(this.editGroupTimeout);
    }

    // Auto-commit group after 1 second of inactivity
    this.editGroupTimeout = setTimeout(() => {
      this.commitEditGroup();
    }, 1000);
  }

  private fireEditEvent(
    label: string,
    undo: () => void | Promise<void>,
    redo: () => void | Promise<void>
  ): void {
    this._onDidChangeCustomDocument.fire({
      document: this.document,
      label,
      undo: async () => {
        await undo();
        await this.updateCallback({ id: '', label, timestamp: 0, undo, redo });
      },
      redo: async () => {
        await redo();
        await this.updateCallback({ id: '', label, timestamp: 0, undo, redo });
      },
    });
  }

  private generateEditId(): string {
    return `edit_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  dispose(): void {
    if (this.editGroupTimeout) {
      clearTimeout(this.editGroupTimeout);
    }
    this._onDidChangeCustomDocument.dispose();
  }
}
```

```typescript
// tscnEditorProvider.ts
import * as vscode from 'vscode';
import { TscnDocument } from './tscnDocument';
import { EditHistoryManager } from './editHistoryManager';

export class TscnEditorProvider implements vscode.CustomEditorProvider<TscnDocument> {
  private readonly editManagers = new Map<TscnDocument, EditHistoryManager>();

  static register(context: vscode.ExtensionContext): vscode.Disposable {
    const provider = new TscnEditorProvider(context);
    const providerRegistration = vscode.window.registerCustomEditorProvider(
      'tscn.editor',
      provider,
      {
        webviewOptions: {
          retainContextWhenHidden: true,
        },
        supportsMultipleEditorsPerDocument: false,
      }
    );

    return providerRegistration;
  }

  constructor(private readonly context: vscode.ExtensionContext) {}

  async openCustomDocument(
    uri: vscode.Uri,
    openContext: vscode.CustomDocumentOpenContext,
    token: vscode.CancellationToken
  ): Promise<TscnDocument> {
    const document = await TscnDocument.create(uri, openContext.backupId);

    // Create edit manager
    const editManager = new EditHistoryManager(document, async (edit) => {
      // Update preview when undo/redo occurs
      await this.updatePreviewAfterEdit(document, edit);
    });

    this.editManagers.set(document, editManager);

    return document;
  }

  async resolveCustomEditor(
    document: TscnDocument,
    webviewPanel: vscode.WebviewPanel,
    token: vscode.CancellationToken
  ): Promise<void> {
    webviewPanel.webview.options = {
      enableScripts: true,
    };

    webviewPanel.webview.html = this.getHtmlForWebview(webviewPanel.webview);

    // Setup message handling
    webviewPanel.webview.onDidReceiveMessage(async (message) => {
      await this.handleWebviewMessage(document, webviewPanel, message);
    });

    // Cleanup on dispose
    webviewPanel.onDidDispose(() => {
      const editManager = this.editManagers.get(document);
      if (editManager) {
        editManager.dispose();
        this.editManagers.delete(document);
      }
    });
  }

  private async handleWebviewMessage(
    document: TscnDocument,
    panel: vscode.WebviewPanel,
    message: any
  ): Promise<void> {
    const editManager = this.editManagers.get(document);
    if (!editManager) return;

    switch (message.command) {
      case 'propertyChanged':
        await this.handlePropertyChange(document, editManager, message);
        break;

      case 'transformChanged':
        await this.handleTransformChange(document, editManager, message);
        break;

      case 'startTransform':
        // Gizmo drag started - start grouping edits
        editManager.startEditGroup(message.label || 'Transform node');
        break;

      case 'endTransform':
        // Gizmo drag ended - commit grouped edits
        editManager.commitEditGroup();
        break;
    }
  }

  private async handlePropertyChange(
    document: TscnDocument,
    editManager: EditHistoryManager,
    message: any
  ): Promise<void> {
    const { nodePath, property, value, oldValue } = message;

    // Store old document state
    const oldContent = document.documentData;

    // Apply change
    const newContent = this.applyPropertyChange(oldContent, nodePath, property, value);
    document.updateContent(newContent);

    // Record edit for undo/redo
    editManager.recordEdit(
      `Update ${property}`,
      async () => {
        // Undo: restore old content
        document.updateContent(oldContent);
        await this.saveDocument(document);
      },
      async () => {
        // Redo: reapply new content
        document.updateContent(newContent);
        await this.saveDocument(document);
      }
    );

    // Save document
    await this.saveDocument(document);
  }

  private async handleTransformChange(
    document: TscnDocument,
    editManager: EditHistoryManager,
    message: any
  ): Promise<void> {
    const { nodePath, property, value, oldValue } = message;

    const oldContent = document.documentData;
    const newContent = this.applyTransformChange(oldContent, nodePath, property, value);
    document.updateContent(newContent);

    // Record edit (will be grouped if in edit group)
    editManager.recordEdit(
      `Transform ${nodePath}`,
      async () => {
        document.updateContent(oldContent);
        await this.saveDocument(document);
      },
      async () => {
        document.updateContent(newContent);
        await this.saveDocument(document);
      }
    );

    await this.saveDocument(document);
  }
```

### Webview Communication for Undo/Redo

```javascript
// media/preview.js
(function() {
  const vscode = acquireVsCodeApi();
  let isTransforming = false;

  // Listen for undo/redo from extension
  window.addEventListener('message', event => {
    const message = event.data;

    switch (message.command) {
      case 'documentUpdated':
        // Document was updated via undo/redo
        reloadScene(message.documentContent);
        break;

      case 'propertyUpdated':
        // Update specific property without full reload
        updateNodeProperty(message.nodePath, message.property, message.value);
        break;
    }
  });

  // Property editing
  function onPropertyChange(nodePath, property, value, oldValue) {
    vscode.postMessage({
      command: 'propertyChanged',
      nodePath,
      property,
      value,
      oldValue,
    });
  }

  // Transform gizmo
  gizmoManager.addEventListener('dragging-changed', (event) => {
    if (event.value) {
      // Start transform - begin edit group
      vscode.postMessage({
        command: 'startTransform',
        label: `Transform ${currentNodePath}`,
      });
      isTransforming = true;
    } else {
      // End transform - commit edit group
      vscode.postMessage({
        command: 'endTransform',
      });
      isTransforming = false;
    }
  });

  gizmoManager.onChange((event) => {
    vscode.postMessage({
      command: 'transformChanged',
      nodePath: event.nodePath,
      property: event.property,
      value: event.value.toArray(),
      oldValue: event.oldValue.toArray(),
    });
  });

  function reloadScene(content) {
    // Parse and reload entire scene
    const parser = new TscnParser();
    const scene = parser.parse(content);
    renderScene(scene);
  }

  function updateNodeProperty(nodePath, property, value) {
    // Update specific node property without full reload
    const object = nodeCache.get(nodePath);
    if (object) {
      applyPropertyToObject(object, property, value);
    }
  }
})();
```
