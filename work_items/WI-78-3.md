# WI-78.3: Hierarchy Registry

**Parent**: WI-78 (Immutable Scene Hierarchies)
**Status**: Not Started
**Complexity**: Simple
**Dependencies**: WI-78.1 (SceneGraph), WI-78.2 (SceneResolver), WI-76 (Event Bus)

## Goal

Create singleton registry that stores SceneGraphs, coordinates panels, and emits update events.

## Architecture

```
HierarchyRegistry (Singleton)
  ├─ Stores SceneGraph per root scene
  ├─ Tracks panels using each hierarchy
  ├─ Emits events via WI-76 ResourceEventBus
  └─ Coordinates scene resolution

Event Flow:
  File watcher detects change
    ↓
  hierarchyRegistry.updateHierarchy('main.tscn')
    ↓
  SceneResolver creates new immutable SceneGraph
    ↓
  Registry stores new version
    ↓
  eventBus.emit('hierarchy', 'updated', 'main.tscn', newGraph)
    ↓
  All panels subscribed to 'main.tscn' receive event
    ↓
  Each panel reconciles independently
```

## Implementation

**File**: `packages/textscene-core/src/core/HierarchyRegistry.ts`

```typescript
import { ResourceEventBus } from '../resources/ResourceEventBus.js';
import { SceneResolver } from './SceneResolver.js';
import type { SceneGraph } from './SceneGraph.js';
import type { ResourceProvider } from './ResourceProvider.js';
import { logger } from '../utils/logger.js';

/**
 * Panel information tracked by registry.
 */
interface PanelInfo {
  panelId: string;
  rootScene: string;
  registeredAt: number;
}

/**
 * Singleton registry for scene hierarchies.
 * Coordinates multi-panel updates via event bus.
 */
export class HierarchyRegistry {
  private hierarchies = new Map<string, SceneGraph>();
  private panels = new Map<string, PanelInfo>();
  private resolver: SceneResolver;

  constructor(
    private resourceProvider: ResourceProvider,
    private eventBus: ResourceEventBus
  ) {
    this.resolver = new SceneResolver(resourceProvider);
  }

  /**
   * Get or create hierarchy for root scene.
   */
  async getOrCreateHierarchy(rootScene: string): Promise<SceneGraph> {
    if (this.hierarchies.has(rootScene)) {
      logger.info(`[HierarchyRegistry] Returning cached hierarchy: ${rootScene}`);
      return this.hierarchies.get(rootScene)!;
    }

    logger.info(`[HierarchyRegistry] Creating new hierarchy: ${rootScene}`);
    const graph = await this.resolver.resolve(rootScene);
    this.hierarchies.set(rootScene, graph);
    return graph;
  }

  /**
   * Update hierarchy (re-resolve scenes).
   * Emits 'hierarchy:updated' event to all panels.
   */
  async updateHierarchy(rootScene: string): Promise<void> {
    logger.info(`[HierarchyRegistry] Updating hierarchy: ${rootScene}`);

    // Clear resolver cache for this scene
    this.resolver.clearCache(rootScene);

    // Re-resolve
    const newGraph = await this.resolver.resolve(rootScene);
    this.hierarchies.set(rootScene, newGraph);

    // Emit event
    this.eventBus.emit('hierarchy', 'updated', rootScene, newGraph);

    logger.info(`[HierarchyRegistry] ✅ Updated hierarchy: ${rootScene} (v${newGraph.version})`);
  }

  /**
   * Register panel using a hierarchy.
   */
  registerPanel(panelId: string, rootScene: string): void {
    this.panels.set(panelId, {
      panelId,
      rootScene,
      registeredAt: Date.now()
    });
    logger.info(`[HierarchyRegistry] Registered panel ${panelId} for ${rootScene}`);
  }

  /**
   * Unregister panel (cleanup).
   */
  unregisterPanel(panelId: string): void {
    const info = this.panels.get(panelId);
    this.panels.delete(panelId);

    if (info) {
      logger.info(`[HierarchyRegistry] Unregistered panel ${panelId}`);

      // If no panels using this hierarchy, could clean up (optional)
      const stillUsed = Array.from(this.panels.values())
        .some(p => p.rootScene === info.rootScene);

      if (!stillUsed) {
        logger.info(`[HierarchyRegistry] No panels using ${info.rootScene}, keeping in cache`);
        // Note: Keep in cache for fast re-open
      }
    }
  }

  /**
   * Get panels using a specific root scene.
   */
  getPanelsForScene(rootScene: string): string[] {
    return Array.from(this.panels.values())
      .filter(p => p.rootScene === rootScene)
      .map(p => p.panelId);
  }

  /**
   * Get all hierarchies.
   */
  getAllHierarchies(): ReadonlyMap<string, SceneGraph> {
    return this.hierarchies;
  }

  /**
   * Clear all hierarchies (for testing).
   */
  clearAll(): void {
    this.hierarchies.clear();
    this.panels.clear();
    this.resolver.clearCache();
    logger.info('[HierarchyRegistry] Cleared all hierarchies');
  }
}

/**
 * Singleton instance.
 * Import and use in TscnRenderer, VSCode extension, etc.
 */
export let hierarchyRegistry: HierarchyRegistry | null = null;

/**
 * Initialize singleton with dependencies.
 */
export function initializeHierarchyRegistry(
  resourceProvider: ResourceProvider,
  eventBus: ResourceEventBus
): void {
  hierarchyRegistry = new HierarchyRegistry(resourceProvider, eventBus);
}

/**
 * Get singleton instance (throws if not initialized).
 */
export function getHierarchyRegistry(): HierarchyRegistry {
  if (!hierarchyRegistry) {
    throw new Error('HierarchyRegistry not initialized. Call initializeHierarchyRegistry() first.');
  }
  return hierarchyRegistry;
}
```

## Testing

**File**: `packages/textscene-core/src/core/HierarchyRegistry.test.ts`

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { HierarchyRegistry } from './HierarchyRegistry.js';
import { ResourceEventBus } from '../resources/ResourceEventBus.js';
import type { ResourceProvider } from './ResourceProvider.js';

describe('HierarchyRegistry', () => {
  let registry: HierarchyRegistry;
  let mockProvider: ResourceProvider;
  let mockEventBus: ResourceEventBus;

  beforeEach(() => {
    mockProvider = { loadFile: vi.fn() } as any;
    mockEventBus = new ResourceEventBus();
    registry = new HierarchyRegistry(mockProvider, mockEventBus);
  });

  it('creates new hierarchy on first access', async () => {
    mockProvider.loadFile = vi.fn().mockResolvedValue('[gd_scene]\n[node name="Root" type="Node3D"]');

    const graph = await registry.getOrCreateHierarchy('res://test.tscn');

    expect(graph.rootScene).toBe('res://test.tscn');
    expect(mockProvider.loadFile).toHaveBeenCalledTimes(1);
  });

  it('returns cached hierarchy on second access', async () => {
    mockProvider.loadFile = vi.fn().mockResolvedValue('[gd_scene]\n[node name="Root" type="Node3D"]');

    await registry.getOrCreateHierarchy('res://test.tscn');
    await registry.getOrCreateHierarchy('res://test.tscn');

    expect(mockProvider.loadFile).toHaveBeenCalledTimes(1);
  });

  it('emits event when hierarchy updated', async () => {
    mockProvider.loadFile = vi.fn().mockResolvedValue('[gd_scene]\n[node name="Root" type="Node3D"]');

    const events: any[] = [];
    mockEventBus.on('hierarchy', 'updated', (scene, graph) => {
      events.push({ scene, graph });
    });

    await registry.getOrCreateHierarchy('res://test.tscn');
    await registry.updateHierarchy('res://test.tscn');

    expect(events.length).toBe(1);
    expect(events[0].scene).toBe('res://test.tscn');
    expect(events[0].graph).toBeDefined();
  });

  it('tracks panel registrations', () => {
    registry.registerPanel('panel1', 'res://main.tscn');
    registry.registerPanel('panel2', 'res://main.tscn');
    registry.registerPanel('panel3', 'res://other.tscn');

    const panelsForMain = registry.getPanelsForScene('res://main.tscn');
    expect(panelsForMain).toEqual(['panel1', 'panel2']);

    const panelsForOther = registry.getPanelsForScene('res://other.tscn');
    expect(panelsForOther).toEqual(['panel3']);
  });

  it('unregisters panels', () => {
    registry.registerPanel('panel1', 'res://main.tscn');
    registry.registerPanel('panel2', 'res://main.tscn');

    registry.unregisterPanel('panel1');

    const panels = registry.getPanelsForScene('res://main.tscn');
    expect(panels).toEqual(['panel2']);
  });

  it('handles multiple hierarchies', async () => {
    mockProvider.loadFile = vi.fn().mockResolvedValue('[gd_scene]\n[node name="Root" type="Node3D"]');

    await registry.getOrCreateHierarchy('res://scene1.tscn');
    await registry.getOrCreateHierarchy('res://scene2.tscn');

    const hierarchies = registry.getAllHierarchies();
    expect(hierarchies.size).toBe(2);
    expect(hierarchies.has('res://scene1.tscn')).toBe(true);
    expect(hierarchies.has('res://scene2.tscn')).toBe(true);
  });
});
```
